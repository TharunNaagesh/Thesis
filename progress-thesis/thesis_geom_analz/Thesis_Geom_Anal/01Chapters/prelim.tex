\chapter*{Introduction}
\label{chap:introduction}

\section*{Motivating the problem}

A basic problem for vector fields is to decompose a given field into
simpler components. The classical example is the \emph{Helmholtz
decomposition}: on a suitable domain, a vector field splits into an
irrotational part and a solenoidal part. In $\bbR^3$ this reads
\[
    F = \nabla\varphi + \nabla\times A,
\]
with $\varphi$ a scalar potential and $A$ a vector potential. The
gradient term is curl-free and the curl term is divergence-free. Such
decompositions are used throughout fluid mechanics, electromagnetism,
and the theory of partial differential equations; see
\cite{CantarellaDeTurckGluck2002,Temam1977}.
\\\\ The form of the decomposition depends on the domain, and in particular
on its topology. On domains of simple topology it suffices to use a
gradient and a curl term. On general domains there are fields that are
simultaneously curl-free and divergence-free, and these cannot be
produced by either term. Such fields are harmonic, and their existence
is governed by the topology of the domain.

The flat torus $\mathbb{T}^n$ illustrates the point. The constant
vector fields on $\mathbb{T}^n$ are both curl-free and divergence-free,
but a nonzero constant field is not the gradient of a periodic
function. Hence the space of harmonic fields is nontrivial, and a
decomposition into a gradient and a curl term alone cannot be complete.
\\\\ In the language of differential forms the picture is as follows. A
$k$-form $\omega$ is \emph{exact} if $\omega=d\alpha$ for some
$(k-1)$-form $\alpha$, and \emph{coexact} if $\omega=\delta\beta$ for
some $(k+1)$-form $\beta$, where $\delta$ is the codifferential. It is
\emph{harmonic} if it lies in the kernel of the Hodge Laplacian
\[
    \Delta_H := d\delta+\delta d .
\]
On a compact Riemannian manifold $M$ without boundary, the Hodge
decomposition theorem states that every smooth $k$-form splits
orthogonally as
\[
    \Omega^k(M)
    =
    d\Omega^{k-1}(M)
    \oplus
    \delta\Omega^{k+1}(M)
    \oplus
    \mathcal{H}^k(M),
    \qquad
    \mathcal{H}^k(M)
    :=
    \{\omega\in\Omega^k(M):\Delta_H\omega=0\}.
\]
The space $\mathcal{H}^k(M)$ is finite-dimensional and is isomorphic
to the de Rham cohomology group $H^k_{\mathrm{dR}}(M)$; equivalently,
every de Rham class has a unique harmonic representative. This ties
the topology of $M$ to the kernel of an elliptic operator; see
\cite{Hodge1952,Warner1983,BottTu1982}.

The existence of harmonic representatives is therefore both a
topological and an analytic statement, and its proof requires
existence, uniqueness, and regularity for solutions of an elliptic
equation. This leads naturally to the study of the Hodge Laplacian on
Sobolev spaces of forms.
\\\\The aim of this work is to establish the Hodge--Morrey decomposition
of differential forms on bounded domains, together with the analytic
framework needed for the decomposition and regularity. We shall take up this endeavor in the following steps:

\begin{itemize}
    \item[--] \emph{The calculus of smooth forms}, together with its boundary
    calculus. This supplies the algebraic identities (wedge, interior
    product, Hodge star, \(d\), \(\delta\)) and, more importantly, the
    integration-by-parts formula that relates \(d\) and \(\delta\) through
    a boundary term. 
    \item[--] \emph{Sobolev spaces of forms}, together with the trace theory
    for the tangential and normal boundary conditions. This supplies the
    function space in which the variational problem is posed, and it
    supplies the density theorem that transfers the identities of the
    first stage from \(C^\infty\) forms to \(W^{1,2}\) forms.
    \item[--] \emph{The Gaffney inequality} which states that every differential form $\omega$ satisfying either
the vanishing tangential or normal boundary condition on $\partial\Omega$
obeys
\[
    \|\omega\|_{W^{1,2}(\Omega)}^2
    \leq C\left(
        \|d\omega\|_{L^2(\Omega)}^2
        +\|\delta\omega\|_{L^2(\Omega)}^2
        +\|\omega\|_{L^2(\Omega)}^2
    \right),
\]
for some constant $C>0$ independent of $\omega$. This is the inequality that gives us compactness: it says that the graph norm of \((d,\delta)\) controls the
    full \(W^{1,2}\) norm, and hence that the quadratic form
    \(\|d\omega\|^2+\|\delta\omega\|^2+\|\omega\|^2\) is coercive on the
    boundary-conditioned Sobolev spaces.
    \item[--] \emph{Compactness and weak compactness}. The Gaffney, along with the Rellich--Kondarachov theorem gives us the required compactness which gives existence of minimizers for the variational
    problem (Eberlein--\v{S}mulian).
\end{itemize}


\chapter{Exterior forms and exterior algebra}

\section{Exterior Algebra: Definitions and Results}

\defn{Dual Space $V^*$}{Let $V$ be a vector space over $\bbR$. The set of all linear maps $f:V\to \bbR$ is called the \emph{dual space} of $V$ and is denoted by $V^*$ or $\Lambda^1(V)$.}
\begin{lemma}
    Let $e_1,e_2,\cdots,e_n$ be a basis for $V$. Then the vectors $\xi^1,\xi^2,\cdots,\xi^n \in \Lambda^1(V)$ defined by $\xi^i(e_j)=\delta_{ij}$ forms a basis for $\Lambda^1(V)$.
\end{lemma}

\defn{Multilinear Maps (Tensors)}{$$f:\underbrace{V \times V \times \cdots \times V}_{k \text{ copies}} \to \bbR$$ is a \emph{k-tensor} if it is linear in each coordinate. More precisely, maps from $\underbrace{V\times V\times \cdots \times V}_{k \text{ copies}}$ into $\bbR$ are called a \emph{covariant} tensor.
\\\\ \textbf{Note:} The space of all tensors forms a vector space.}
\begin{lemma}\label{basis for tensors}
    For the space of all $k-$tensors from $V$, $\{dx^{i_1}\otimes dx^{i_2} \otimes \cdots \otimes dx^{i_k}: (i_1,i_2,\cdots,i_k)\in \{1,2,\cdots,n\}^k \}$ forms a basis (Here we have adopted the notational choice of writing $dx^i$ in place $\xi^i$).
\end{lemma}

\defn{$\sigma f$}{Let $f$ be a $k-$ tensor and $\sigma \in Sym(k)$. We define $$\sigma f (v_1,v_2,\cdots,v_k):= f(v_{\sigma(1)},v_{\sigma(2)},\cdots,v_{\sigma(k)}) $$}

\subsubsection{$\sigma f$ as a group action}
Let $X=$ set of all $k-$ tensors. Define $\rho: sym(k)\times X\to X$ by $\rho(\sigma, f):=\sigma f$. We see that $\rho$ is a group action since 
\begin{enumerate}
    \item $\rho(1,f)=f$
    \item $\rho(\sigma_1, \rho(\sigma_2,f))=\rho(\sigma_1\sigma_2,f)$
\end{enumerate}

\defn{Alternating Tensors}{A $k-$tensor is alternating if for every $\sigma \in sym(k)$, $$f(v_{\sigma(1)},v_{\sigma(2)},\cdots,v_{\sigma(k)})=\operatorname{sign}(\sigma) f(v_1,v_2,\cdots,v_k)$$ or $$\sigma f\equiv \operatorname{sign}(\sigma) f$$ }

\defn{Symmetric Tensors}{A $k-$tensor is symmetric if for every $\sigma\in sym(k)$, $$f(v_{\sigma(1)},v_{\sigma(2)},\cdots,v_{\sigma(k)})=f(v_1,v_2,\cdots,v_k)$$ or $$\sigma f\equiv f $$}

\defn{Symmetrization Operator $Sf$}{$$S:\{k-\text{tensors}\}\to \{k-\text{tensors}\}$$ given by $Sf(v_1,v_2,\cdots,v_k)=\sum_{\sigma \in sym(k)} f(v_{\sigma(1)},v_{\sigma(2)},\cdots,v_{\sigma(k)})$ or $$Sf\equiv \sum_{\sigma \in sym(k)} \sigma f $$ is called the \emph{symmetrization operator}}

\defn{Alternation or Antisymmetrizer Operator $Af$}{$$A:\{k-\text{tensors}\}\to \{k-\text{tensors}\}$$ given by $Af(v_1,v_2,\cdots,v_k)=\sum_{\sigma \in sym(k)} \operatorname{sign}(\sigma) f(v_{\sigma(1)},v_{\sigma(2)},\cdots,v_{\sigma(k)})$ or $$Af\equiv \sum_{\sigma \in sym(k)} \operatorname{sign}(\sigma)\sigma f $$ is called the \emph{alternation operator}}

\thmp{}{If $f$ is a $k-$tensor, then 
\begin{enumerate}
    \item $Sf$ is a symmetric $k-$tensor.
    \item $Af$ is an alternating $k-$tensor.
\end{enumerate}
}{
1) $Sf:=\sum_{\sigma \in sym(k)} \sigma f$. Let $\tau \in sym(k)$. $$\tau Sf= \sum_{\sigma \in sym(k)} (\tau \sigma)f = Sf$$ (since $\tau \sigma$ runs over all of $sym(k)$ once and exactly once).
\\\\ 2) $Af:=\sum_{\sigma\in sym(k)}\operatorname{sign}(\sigma)f$. Let $\tau \in sym(k)$. $$\tau Af= \sum_{\sigma \in sym(k)} \operatorname{sign}(\sigma \tau) \operatorname{sign}(\tau) (\tau \sigma)f=$$ $$ \operatorname{sign}(\tau) Af$$. 
 }
\lemp{}{\begin{enumerate}
    \item If $f$ is an alternating $k-$ tensor ($\sigma f=\operatorname{sign}(\sigma)f$) then $Af=k!f$.
    \item If $f$ is an alternating $k-$ tensor then $Sf=0$.
    \item If $f$ is a symmetric $k-$tensor ($\sigma f=f$) then $Af=0$.
    \item If $f$ is a symmetric $k-$tensor then $Sf=k! f$
\end{enumerate}}{
1) $$Af:=\sum_{\sigma \in sym(k)} \operatorname{sign}(\sigma) \operatorname{sign}(\sigma)f=\sum_{\sigma \in sym(k)} f = k!f$$
\\\\ 2) If $f$ is alternating, then $$Sf=\sum_{\sigma \in sym(k)} \operatorname{sign}(\sigma) f=0 $$
\\\\ 3) If $f$ is symmetric, then $$A(f):=\sum_{\sigma \in sym(k)}\operatorname{sign}(\sigma) f=0 $$
\\\\ 4) If $f$ is symmetric, then $$Sf:=\sum_{\sigma \in sym(k)} \sigma f = k! f$$
}

\subsubsection{Tensor Product, Alternating Forms and the Wedge Product}
\defn{Tensor Product}{Let $f$ be a $k-$tensor and $g$ be a $l-$tensor. We define the \textbf{tensor product} $f\otimes g$ to be the $(k+l)-$tensor given by $$f\otimes g (v_1,v_2,\cdots, v_k, v_{k+1},v_{k+2},\cdots, v_{k+l}):=f(v_1,v_2,\cdots,v_k) \cdot g(v_{k+1},v_{k+2},\cdots,v_{k+l}) $$}
\textbf{Note:} The tensor product is \emph{bilinear} (linear in each argument) and \emph{associative} ($(f\otimes g)\otimes h=f\otimes(g\otimes h)$) but not \emph{commutative}.

\defn{Wedge Product}{Let $f$ be an alternating $k-$tensor ($f\in \Lambda^k(V)$) and $g$ an alternating $l-$tensor ($g\in \Lambda^l(V)$). We define $$f \wedge g:= \frac{1}{k!l!} A(f\otimes g)$$ or written explicitly $$f\wedge g(v_1,\cdots,v_{k+l}):=\frac{1}{k!l!} \sum_{\sigma \in sym(k+l)} \operatorname{sign}(\sigma) f(v_{\sigma(1)},\cdots,v_{\sigma(k)})g(v_{\sigma(k+1)},\cdots,v_{\sigma(k+l)}) $$ }
\defn{$Sym(k,l)$}{Given natural numbers $k$ and $l$, we define $Sym(k,l)$ as follows: $$Sym(k,l):=\{\sigma\in sym(k+l): \sigma(1)<\sigma(2)<\cdots<\sigma(k) \text{ and }\sigma(k+1)<\sigma(k+2)<\cdots<\sigma(k+l)\} $$}
\thmp{(Equivalent Expressions for the Wedge Product)}{The two expressions \begin{enumerate}
    \item $$f\wedge g(v_1,\cdots,v_{k+l}):=\frac{1}{k!l!} \sum_{\sigma \in sym(k+l)} \operatorname{sign}(\sigma) f(v_{\sigma(1)},\cdots,v_{\sigma(k)})g(v_{\sigma(k+1)},\cdots,v_{\sigma(k+l)}) $$
    \item $$f\wedge g(v_1,\cdots,v_{l+l})=\sum_{\sigma\in sym(k,l)}\operatorname{sign}(\sigma) f(v_{\sigma(1)},\cdots,v_{\sigma(k)})g(v_{\sigma(k+1)},\cdots,v_{\sigma(k+l)}) $$
\end{enumerate} are equivalent.}{Consider $G=sym(k+l)$ and consider the subgroup $$H:=\{\sigma \in sym(k+l): \sigma|_{\{1,\cdots,k\}}\in \{1,\cdots,k\}\text{ and } \sigma|_{\{k+1,\cdots,l+l\}}\in \{k+1,\cdots,k+l\}\}$$ (One can view every element $\sigma \in H$ as $\pi_1 \times \pi_2$ where $\pi_1 \in sym\{1,\cdots,k\}$ and $\pi_2 \in sym\{k+1,\cdots,k_l\}$).
\\\\ Let $\alpha,\beta \in G$ and define an equivalence relation $\alpha \sim \beta $ if and only if there exists $h \in H$ such that $\alpha h=\beta $
\\\\ Let $S_{\sigma}$ be the expression $\operatorname{sign}(\sigma) \sigma(f\otimes g)$. 
\\\\ Suppose $\alpha \sim \beta$ or $\alpha h=\beta$ where $h=\pi_1 \times \pi_2$. 
\\\\ $$S_{\alpha}=\operatorname{sign}(\alpha) f(v_{\alpha(1)},\cdots,v_{\alpha(k)})g(v_{\alpha(k+1)},\cdots,v_{\alpha(k+l)})$$
and $$S_{\beta}=\operatorname{sign}(\beta) f(v_{\beta(1)},\cdots,v_{\beta(k)})g(v_{\beta(k+1)},\cdots,v_{\beta(k+l)})$$ which is equal to $$ \operatorname{sign}(\alpha)\operatorname{sign}(\pi_1)\operatorname{sign}(\pi_2) f(v_{\alpha(h(1))},\cdots,v_{\alpha(h(k))})g(v_{\alpha(h(k+1))},\cdots,v_{\alpha(h(k+l))}) $$ $$=\operatorname{sign}(\alpha) \operatorname{sign}(\pi_1)^2 \operatorname{sign}(\pi_2)^2 f(v_{\alpha(1)},\cdots,v_{\alpha(k)})g(v_{\alpha(k+1)},\cdots,v_{\alpha(k+l)})=S_{\alpha} $$ So whenever $\alpha\sim \beta$, $S_{\alpha}=S_{\beta}$.
\\\\ We can also see that, via rearrangement, for every $\sigma \in sym(k+l)$ there exists an $\alpha \in sym(k,l)$ and a $h\in H$ such that $\sigma h=\alpha$ or $\sigma \sim \alpha$.
\\\\ The first expression can now be written as $$\frac{1}{k!l!}\sum_{\sigma\in sym(k,l)}\sum_{\alpha\in [\sigma]} S_{\alpha} $$

but for elements of the same equivalence class, $S_{\alpha}$ are all the same and is equal to $S_{\sigma}$, so we have $$\frac{1}{k!l!} \sum_{\sigma\in sym(k,l)}S_{\sigma} (k!l!)=\sum_{\sigma \in sym(k,l)} \operatorname{sign}(\sigma) \sigma (f\otimes g)  $$ (The $k!l!$ comes from the fact that $|H|=k!l!$ and so the size of each coset is $k!l!$) and the equality is proved.}

\thmp{}{For $f\in \Lambda^k(V)$ and $g\in \Lambda^l(V)$ $$f\wedge g=(-1)^{kl}g \wedge f$$}{$$f\wedge g=\sum_{\sigma \in sym(k,l)} \operatorname{sign}(\sigma) \sigma(f\otimes g)$$ Let $\tau$ be the element of $sym(k+l)$ given by $\tau(1)=k+1, \tau(2)=k+2,\cdots \tau(l)=k+l$ and $\tau(l+1)=1, \tau(l+2)=2,\cdots \tau(l+k)=k$ (We are moving the block $\{k+1,k+2,\cdots,k+l\}$ back, behind the block $\{1,2\cdots k\}$, which amounts to a total of $kl$ swaps/transpositions).
Note that $\tau$ acts as a correspondence between $sym(k,l)$ and $sym(l,k)$ since for every $\sigma \in sym(k,l)$, we have $\sigma(\tau(l+1))<\sigma(\tau(l+2))<\cdots<\sigma(\tau(l+k))$ and $\sigma(\tau(1))<\sigma(\tau(2))<\cdots<\sigma(\tau(l))$ so $\sigma(\tau)$ is in $sym(l,k)$. 
$$f\wedge g(v_1,\cdots,v_{k+l})=\sum_{\sigma \in sym(k,l)} \operatorname{sign}(\sigma) f(v_{\sigma(1)},\cdots,v_{\sigma(k)})g(v_{\sigma(k+1)},\cdots,v_{\sigma(k+l)})$$ $$=\sum_{\sigma \in sym(k,l)} \operatorname{sign}(\tau\sigma)\operatorname{sign}(\tau)g(v_{\sigma(\tau(1))},\cdots,v_{\sigma(\tau(l))}) f(v_{\sigma(\tau(l+1))},\cdots,v_{\sigma(\tau(l+k))}) $$
$$=\sum_{\gamma \in sym(l,k)} \operatorname{sign}(\gamma) (-1)^{kl}g(v_{\gamma(1)},\cdots,v_{\gamma(l)})f(v_{\gamma(l+1)},\cdots,v_{\gamma(l+k)})=(-1)^{kl} g\wedge f$$}

\lemp{}{Let $f\in \Lambda^k(V)$ and $g\in \Lambda^l(V)$, then: 
\begin{enumerate}
    \item $A(Af\otimes G)=k!A(f\otimes g)$
    \item $A(f \otimes Ag)=l! A(f\otimes g)$
\end{enumerate}
}{1) $$A(Af\otimes g)=\sum_{\sigma \in sym(k+l)} \operatorname{sign}(\sigma) \sigma(Af\otimes g)$$
$$Af:= \sum_{\substack{\tau\in sym(k+l) \\ \tau \text{ preserves }\{k+1,\cdots,k+l\}}}\operatorname{sign}(\tau) \tau (f) $$
and $$ A(Af \otimes g):=\sum_{\sigma \in sym(k+l)} \operatorname{sign}(\sigma) (Af(X_{\sigma(1)},\cdots,X_{\sigma(k)})g(X_{\sigma(k+1)},\cdots,X_{\sigma(k+l)}))$$
$$=\sum_{\sigma \in sym(k+l)} \operatorname{sign}(\sigma) \bigg{(} \sum_{\substack{\tau \in sym(k+l)\\ \tau \text{ preserves } \\ \{k+1,\cdots,k+l\}}} \operatorname{sign}(\tau) f(X_{\sigma \tau (1)},\cdots,X_{\sigma \tau(k)}) \bigg{)}g(X_{\sigma(k+1)},\cdots,X_{\sigma(k+l)}))$$

$$=\sum_{\sigma \in sym(k+l)}\sum_{\substack{\tau \in sym(k+l)\\ \tau \text{ fixes } \\ \{k+1,\cdots,k+l\}}} \operatorname{sign}(\sigma \tau) f(X_{\sigma \tau (1)},\cdots,X_{\sigma \tau (k)})g(X_{\sigma\tau (k+1)},\cdots,X_{\sigma\tau (k+l)}) $$
The number of such $\tau$ that fixes $\{k+1,\cdots,k+l\}$ are precisely $k!$, and for each such $\tau$, $\sigma \tau$ runs over all of $sym(k+l)$ once and exactly once. Hence we have $$A(Af\otimes g)=k! \sum_{\gamma \in sym(k+l)}\operatorname{sign}(\gamma) \gamma (f\otimes g)=k! A(f\otimes g) $$
2) The 2nd part is proved similarly.
}
\thmp{\label{associativity_of_wedge}(Associativity of Wedge Product)}{For any $f\in \Lambda^k(V),g\in \Lambda^l(V), h\in \Lambda^m(V)$ $$f\wedge (g\wedge h)=(f\wedge g)\wedge h $$}{$$f\wedge (g\wedge h):= \frac{1}{k!(l+m)!} A(f\otimes (g\wedge h))=\frac{1}{k!(l+m)!} A(f\otimes (\frac{1}{l!m!}A(g\otimes h) ))  $$ $$= \frac{1}{k!l!m!(l+m)!} A(f\otimes A(g\otimes h))=\frac{1}{k!l!m!} A(f\otimes g\otimes h) $$ and likewise $$(f\wedge g)\wedge h=\frac{1}{(k+l)!m!}A((f\wedge g)\otimes h)=\frac{1}{(k+l)k!l!m!}A(A(f\otimes g)\otimes h) $$ $$=\frac{1}{k!l!m!} (k+l)!A(f\otimes g\otimes h)=\frac{1}{k!l!m!}A(f\otimes g\otimes h) $$}

\textbf{Note:} \emph{The proof of \ref{associativity_of_wedge} shows that if $f_{l_1}$ is an $l_1-$form, $f_{l_2}$ is an $l_2-$form and so on, $f_{l_1}\wedge f_{l_2}\wedge \cdots \wedge f_{l_k}$ is given by}

\begin{equation}\label{wedge_product_of_bunch_of_forms}
    f_{l_1}\wedge f_{l_2}\wedge \cdots \wedge f_{l_k}=\frac{1}{l_1 ! l_2 ! \cdots l_k!} A(f_{l_1}\otimes f_{l_2}\otimes \cdots \otimes f_{l_k}) 
\end{equation}
\subsubsection{Short Digression on Determinants}
\defn{Determinant (\emph{Leibniz Formula})}{Let $\{a_{ij}:1\leq i,j\leq n\}$ be a matrix. We define determinant of $A$ as $$\det(A):=\sum_{\sigma \in sym(n)}\operatorname{sign}(\sigma) a_{1,\sigma(1)}a_{2,\sigma(2)}\cdots a_{n,\sigma(n)}$$ (To be thought of as multiplying one element from each row chosen from distinct columns, and taking into account the sign of the permutation of the columns).}

\defn{Minor of $(i,j)-$entry}{Let $A$ be a matrix and let $a_{ij}$ be the $(i,j)-$entry of $A$. The \emph{minor} of $a_{ij}$ is the determinant of the matrix obtained by deleting the $i^{th}$ row and $j^{th}$ column from $A$. It is denoted $m_{ij}$. The matrix itself (after deleting $i^{th}$ row and $j^{th}$ column) is denoted $M_{ij}$. }

\defn{Laplace's Expansion}{Fix a row $i$. Then the Laplace's expansion gives $$\det(B)=\sum_{j=1}^n (-1)^{i+j} b_{ij} \det(M_{ij})$$}
\textbf{Note:} 
From \begin{equation}
    \sum_{\sigma \in sym(n)}\operatorname{sign}(\sigma) \prod_{q=1}^n a_{q,\sigma(q)}=\sum_{j=1}^n \sum_{\substack{\sigma \in sym(n)\\ \sigma: i \mapsto j}}\operatorname{sign}(\sigma) a_{i,j}  \prod_{\substack{q=1\\ q\neq i}}^{n} a_{q,\sigma(q)}
\end{equation}
we can see that the Laplace's expansion is a consequence of the Leibniz formula for determinants.

\subsubsection{Dimension of Alternating k-forms}
Let $E_1,E_2,\cdots, E_n$ be a basis for $V$. Let $\xi^1,\xi^2,\cdots,\xi^n \in V^*=\Lambda^k(V)$ be the dual basis to $E_1,\cdots E_n$. 

\thmp{(Basis for $\Lambda^k(V)$)}{The set $$\{\xi^{i_1}\wedge \xi^{i_2}\wedge \cdots \wedge \xi^{i_k}: 1\leq i_1<i_2<\cdots<i_k\leq n\}$$ forms a basis for $\Lambda^k(V)$.}{\textbf{Linear Independence:} Look at $$\sum_{1\leq i_1<\cdots<i_k\leq n} c_{i_1 i_2 \cdots i_k} \xi^{i_1}\wedge xi^{i_2}\wedge \cdots \wedge \xi^{i_k} \equiv 0$$ in $\Lambda^k(V)$. Hence, $$\sum_{1\leq i_1<\cdots<i_k\leq n} c_{i_1 i_2 \cdots i_k} \xi^{i_1}\wedge xi^{i_2}\wedge \cdots \wedge \xi^{i_k} (E_{j_1}, E_{j_2},\cdots ,E_{j_k})=$$ $$\sum_{1\leq i_1<\cdots<i_k\leq n} c_{i_1 i_2 \cdots i_k} det(\xi^{i_r}(E_{j_s}))=0 $$ $$\implies c_{j_1 j_2 \cdots j_k}=0 $$ for every increasing k-tuple $j_1,j_2,\cdots,j_k$ and hence $\{\xi^{i_1}\wedge \xi^{i_2}\wedge \cdots \wedge \xi^{i_k}:1\leq i_1<\cdots<i_k\leq n\}$ is linearly independent.
\\\\ \textbf{Span:} Let $f\in \Lambda^k(V)$. Let $x_r=\sum_{j_1=1}^n \xi^{j_1}(x_r) E_{j_1} $ for $r=1,2,\cdots,k$. $$f(x_1,x_2,\cdots,x_k)=f(\sum_{j_1=1}^n \xi^{j_1}(x_1) E_{j_1},\sum_{j_1=1}^n \xi^{j_1}(x_2) E_{j_1},\cdots,\sum_{j_1=1}^n \xi^{j_1}(x_k) E_{j_1})$$ $$=\sum_{j_1=1}^n \sum_{j_2=1}^n \cdots \sum_{j_k=1}^n \xi^{j_1}(x_1)\xi^{j_2}(x_2)\cdots \xi^{j_k}(x_k) f(E_{j_1},E_{j_2},\cdots,E_{j_k}) $$ If $j_r=j_s$ for some $r$ and $s$, note that the summand is $0$. We start with the observation that $\{dx^I: I \in \mathscr T_k\}$ not only forms a basis for $\Lambda^k(V)$, but is also an orthonormal set: $$\langle dx^I,dx^J\rangle=\begin{cases}

 \end{cases}$$
This is a natural consequence of the definition $\langle f,g\rangle=\sum_{I\in \mathscr T_k} f^Ig^I$. In the case of a Riemannian manifold with a given $g_p:T_pM\times T_pM\to \bbR$, there is a natural choice for $\langle f,g\rangle_g$ given by   \begin{equation}

\end{equation}

\end{corollary}

\subsubsection{\label{section:flat_map}The $\flat$ map}
For any general Manifold, the vector spaces $T_pM$ and $T_p^*M$ are merely of the same dimension, but there is no global way to identify $T_pM$ with $T_p^*M$. A metric changes this. 

Let $g_p : T_p M \times T_p M \to \bbR$ be an inner product on $T_pM$. For a given vector $X \in T_pM$, define $X^{\flat}\in \Lambda^1(T_p(M))$ given by \begin{equation}\label{flat_isom}X^{\flat}(Y)=g(X,Y)\end{equation} for $Y\in T_p(M)$. Therefore, $\flat: T_p(M)\to T_p^*(M)$.
This map is linear, and injective since $X^{\beta}(Y)=g(X,Y)=0, \text{ for every }Y\in T_p(M)$ implies $X=0$. Hence, $\flat:T_pM\to T^*_pM$ is an isomorphism.
\\\\ Let $X\equiv \sum_{i=1}^n X^i \frac{\partial}{\partial x^i}$ be a vector field. $g$ being an inner product, is a $2-$tensor on $T_p(M)$.  

By \ref{basis for tensors} we see then, that $$g\equiv \sum_{1\leq i,j\leq n} g_{ij}(p) dx_p^i \otimes dx_p^j:T_pM\times T_pM\to \bbR $$ and \ref{flat_isom} tells us that $$X^{\flat}_{p}= \sum_{1\leq i,j\leq n}g_{ij}(p)\sum_{s=1}^na^s(p)dx^i_p( \frac{\partial}{\partial x^s}\bigg.{|}_p) dx^j_p = \sum_{1\leq i\leq n}\sum_{1\leq j\leq n} g_{ij}(p)a^i(p)dx^j_p \in \Lambda^{1}(T_p(M))$$
If we assume that $g_{ij}=\delta_{ij}$ we get back \ref{one forms vs vectors}.
\\\\ \textbf{On Basis:} If $X\equiv \frac{\partial}{\partial x^q}$, then $X^{\flat}= g(X,())$ given by $$\sum_{i=1}^n \sum_{j=1}^n g_{ij} dx^i(\frac{\partial}{\partial x^q})dx^j=\sum_{j=1}^n g_{qj}dx^j$$
Therefore, $$\Big{(}\frac{\partial}{\partial x^i}\Big{)}^{\flat}\equiv \sum_{j=1}^n g_{ij} dx^j $$
If our metric $g$ were Euclidean ($g_{ij}=\delta_{ij}$) we would have that $$\Big{(}\frac{\partial}{\partial x^i}\Big{)}^{\flat}\equiv dx^i $$

\begin{proposition}\textbf{(Leibniz rule for interior product)}\label{prop:leibniz_rule_for_interior} Let $U\subset\bbR^n$ be open, $k\ge 1$, and let $f\in C^1(U;\Lambda^k)$ and

\[
\nabla_X f:=\sum_{|I|=k}X(f_I)\,dx^I
\quad\text{if } f=\sum_{|I|=k} f_I\,dx^I,
\qquad
\nabla_X Y:=\sum_{i=1}^n X(Y^i)\,e_i
\quad\text{if } Y=\sum_{i=1}^n Y^i e_i .
\]
Then, at every point of $U$,
\begin{equation}\label{eq:leibniz-interior}
\nabla_X(\imath_Y f)=\imath_{\nabla_X Y}f+\imath_Y(\nabla_X f).
\end{equation}
In particular, \eqref{eq:leibniz-interior} holds with $X=e_a$ for any
$a$ (and, since both sides depend only on the value of $X$ at the point, for any local frame $e_a$).    
\end{proposition}
\pf{By linearity, it is enough to consider $X=\partial_{x^i}$, $Y=g e_j$ (for some function $g$), and $f=h dx^I$ (for some function $h$). \begin{align*}

\end{align*}

\subsection{Hodge Star}
Let $f$ be a $k-$form given by $$f\equiv \sum_{I\in \mathscr T_k} f^I dx^I $$
We define the operation \emph{Hodge Star} as follows (defined first on basic $k-$forms, and then extended linearly for any $k-$form): 

$$\star \ dx^{i_1}\wedge dx^{i_2}\wedge \cdots \wedge dx^{i_k}={\operatorname{sign}(R(I,J))} dx^{j_1}\wedge dx^{j_2}\wedge \cdots \wedge dx^{j_{n-k}} $$ where \begin{enumerate}
    \item $I\equiv \{i_1,i_2,\cdots, i_k\}\in \mathscr{T}_k$ and $J\equiv \{j_1,j_2,\cdots,j_{n-k}\}$ being the complementary increasing index set to $I$ (i.e. $J\cup I\equiv \{1,2,\cdots,n\}$ where $J$ is also increasingly ordered).
    \item $R(I,J)\in sym(n)$ rearranges $\{i_1,i_2,\cdots,i_k, j_1, j_2,\cdots,j_{n-k}\}$ in its increasing order.
    \item For any $f\equiv \sum_{I\in \mathscr T_k} f^I dx^I$ we have $$\star \ f := \sum_{I\in \mathscr T_k} f^{I} \star dx^I$$
\end{enumerate}
An alternate way to define the \emph{Hodge Star}: If $f$ and $g$ are $k-$forms, then $\star \ g$ is defined as the unique $(n-k)-$form such that \begin{equation}\label{hodge_and_inner}f\wedge \star g=\langle f,g\rangle dx^1 \wedge dx^2 \wedge \cdots \wedge dx^n\end{equation} 
where the inner product is $$\langle f,g\rangle:=\sum_{I\in \mathscr{T}_k} f^I g^I$$

\lemp{}{Let $f\in \Omega(\Lambda^k)$. We then have \begin{equation}\label{hodge_double_star_identity}\star \star f=(-1)^{k(n-k)} f\end{equation} }{Since $\star$ is linear, it is enough to prove on basis elements. $$\star dx^I=\operatorname{sign}(R(I,J)) dx^J $$ where $J$ is the complement of $I$ relative to $\{1,2,\cdots,n\}$ and $R(I,J)$ is the permutation that rearranges $\{i_1,i_2,\cdots,i_k,j_1,j_2,\cdots,j_{n-k}\}\to \{1,2,\cdots,n\}$. 
$$\star \star dx^I=\operatorname{sign}(R(I,J)) \star dx^J= \operatorname{sign}(R(I,J)) \operatorname{sign}(R(J,I))dx^I $$
To compute the product of the signs we look at the map $\sigma$ that takes $(I,J)$, rearranges them to $\{1,2,\cdots,n\}$ and then rearranges them to $\{J,I\}$. This amounts to moving the first $k$ elements behind the last $n-k$ elements. This takes $k(n-k)$ transpositions, and hence the sign is $(-1)^{k(n-k)}$. Therefore, $$\star \star dx^I=(-1)^{k(n-k)} dx^I $$
and the result follows for a general $k-$form from linearity. 
}

\lemp{}{For $f,g\in \Omega(\Lambda^k)$ $$\langle f,g\rangle=\langle \star f,\star g\rangle$$}{Since $f\wedge \star g=\langle f,g\rangle dV$ (where $dV$ is the volume form, in the Euclidean case, $dx^1\wedge dx^2\wedge \cdots \wedge dx^n$), we have $\star f \wedge \star \star g= (-1)^{k(n-k)}\star f \wedge g= \langle \star f,\star g\rangle dV$. Recall that $f\wedge g=(-1)^{kl} g \wedge f$ for $f\in \Omega\Lambda^k$ and $g\in \Omega\Lambda^l$. Since $\star f \in \Lambda^{n-k}$ and $g\in \Lambda^k$, we have $\star f \wedge g= (-1)^{k(n-k)} g \wedge \star f$. This gives us $$(-1)^{k(n-k)}\star f \wedge g= g \wedge \star f= \langle \star f,\star g\rangle dV $$
But since $g \wedge \star f:= \langle g,f \rangle = \langle f,g\rangle $ (Real inner product), we have the result. 
}

\subsection{Hodge Star and Interior}
\lemp{}{\label{interior_in_terms_of_star_euclid}Let $\omega$ be a $k-$form and $\eta$ an $l-$form. We have the following identity: \begin{equation}\label{interior_in_terms_of_hodge}\omega \lrcorner \eta= (-1)^{(n-l)(l-k)} \star (\omega \wedge \star \eta) \end{equation}
If $\omega$ were a $1$ form, this formula reads: \begin{equation}
    \omega \lrcorner \eta=(-1)^{n(l-1)}\star(\omega \wedge \star \eta)
\end{equation} }{Let $\gamma$ be an arbitrary $(l-k)-$form. 
$$\langle \omega \wedge \gamma, \eta\rangle dV= \omega \wedge \gamma \wedge \star \eta $$
By the anti-commutation rule of the wedge, we have $$\langle \omega\wedge \gamma,\eta\rangle dV=(-1)^{k(l-k)} \gamma \wedge (\omega \wedge \star \eta) $$
Using the $\star \star$ identity, we can write $$(\omega \wedge \star \eta)= (-1)^{(k+n-l)(l-k)}\star \star (\omega\wedge \star \eta) $$ which gives us $$\langle \omega \wedge \gamma,\eta\rangle dV=(-1)^{k(l-k)}(-1)^{(k+n-l)(l-k)}\gamma \wedge \star \big{(}\star (\omega \wedge \star \eta)\big{)} $$

Since $\gamma$ was arbitrary, we have $$\omega\lrcorner\eta=(-1)^{(l-k)(n-l)}\star (\omega \wedge \star \eta) $$
}

\section{Pullback of Forms}
\label{sec:pullback}

Throughout this section $U\subseteq\R^n$ and $V\subseteq\R^m$ are open
and $F:U\to V$ is smooth. The definitions and results below are stated in
Euclidean coordinates.
\subsection{The differential of a smooth map}

For $p\in U$, the Fr\'echet derivative of $F$ at $p$ is the unique linear
map $DF_p:\R^n\to\R^m$ satisfying
\[
    F(p+h) = F(p) + DF_p(h) + o(|h|),
    \qquad |h|\to 0 .
\]
Equivalently, $DF_p$ is characterised by its action on derivations,
\[
    DF_p(v)(g) = v(g\circ F),
    \qquad v\in T_p U,\ g\in C^\infty_{F(p)}(V).
\]
Writing $F^j = y^j\circ F$ for the $j$-th coordinate function of $F$, the
coordinate expression is
\[
    DF_p\Bigl(\frac{\partial}{\partial x^i}\Big|_p\Bigr)
    =
    \sum_{j=1}^m
    \frac{\partial F^j}{\partial x^i}\Big|_p\,
    \frac{\partial}{\partial y^j}\Big|_{F(p)} ,
\]
and the matrix $(\partial F^j/\partial x^i|_p)$ is the Jacobian matrix of
$F$ at $p$.

\subsection{Pullback of a differential form}

\begin{definition}[Pullback]
\label{def:pullback}
Let $\omega$ be a $k$-form on $V$. The \emph{pullback} $F^*\omega$ is the
$k$-form on $U$ defined at $p\in U$ by
\[
    (F^*\omega)_p(v_1,\dots,v_k)
    =
    \omega_{F(p)}\bigl(DF_p(v_1),\dots,DF_p(v_k)\bigr),
    \qquad v_1,\dots,v_k\in T_p U .
\]
\end{definition}

The definition is pointwise and therefore linear; the content of the
notion lies in how pullback interacts with the algebraic and
differential structure of forms. The following four statements are the
ones used in the body of the thesis.

\begin{proposition}[Pullback of coordinate $1$-forms]
\label{prop:pullback-1-form}
For $1\le q\le m$,
\[
    F^*(dy^q)
    =
    \sum_{i=1}^n \frac{\partial F^q}{\partial x^i}\, dx^i
    =
    dF^q
    =
    d(F^*y^q).
\]
In particular, pullback of a coordinate function is composition,
$F^*y^q = y^q\circ F$, and pullback of its differential is the
differential of the composition.
\end{proposition}

\begin{proposition}[Pullback of $k$-forms]
\label{prop:pullback-k-form}
Let $I=(i_1<\cdots<i_k)$ be a multi-index and
$dy^I = dy^{i_1}\wedge\cdots\wedge dy^{i_k}$. Then
\[
    F^*(dy^I)_p
    =
    \sum_{1\le j_1<\cdots<j_k\le n}
    \det\Bigl(\bigl(\partial_{j_s}F^{i_r}|_p\bigr)_{1\le r,s\le k}\Bigr)
    \,
    dx^{j_1}_p\wedge\cdots\wedge dx^{j_k}_p ,
\]
equivalently,
\[
    F^*(dy^I) = dF^{i_1}\wedge\cdots\wedge dF^{i_k}.
\]
For a general $k$-form $\omega = \sum_I \omega^I\, dy^I$ on $V$,
\[
    F^*\omega = \sum_I (\omega^I\circ F)\, dF^I,
    \qquad
    dF^I := dF^{i_1}\wedge\cdots\wedge dF^{i_k}.
\]
\end{proposition}

\begin{corollary}[Change of variables for top forms]
\label{cor:change-of-variables-top}
If $m=n$ and $F:U\to V$ is a smooth diffeomorphism, then
\[
    F^*(dy^1\wedge\cdots\wedge dy^n)_p
    =
    \det\bigl((\partial_i F^j|_p)_{1\le i,j\le n}\bigr)\,
    dx^1_p\wedge\cdots\wedge dx^n_p .
\]
\end{corollary}

\begin{proposition}[Pullback respects the wedge product]
\label{prop:pullback-wedge}
For a $k$-form $\alpha$ and an $\ell$-form $\beta$ on $V$,
\[
    F^*(\alpha\wedge\beta) = F^*\alpha\wedge F^*\beta .
\]
\end{proposition}

\begin{proposition}[Pullback commutes with $d$]
\label{prop:pullback-commutes-with-d}
For every smooth $k$-form $\omega$ on $V$,
\[
    F^*(d\omega) = d(F^*\omega).
\]
\end{proposition}

\begin{remark}
\label{rem:pullback-chain-map}
Proposition~\ref{prop:pullback-commutes-with-d} says that pullback is a
chain map of the de Rham complex, i.e.\ the diagram
\[
    \begin{array}{ccc}
        \Omega^k(V) & \xrightarrow{\,F^*\,} & \Omega^k(U) \\[2pt]
        \big\downarrow{\scriptstyle d} & & \big\downarrow{\scriptstyle d} \\[2pt]
        \Omega^{k+1}(V) & \xrightarrow{\,F^*\,} & \Omega^{k+1}(U)
    \end{array}
\]
commutes. Together with Proposition~\ref{prop:pullback-wedge}, this makes
$F^*$ a homomorphism of differential graded algebras. These are the only
structural properties of pullback used in the thesis; in
section~\ref{chap:tangential-normal} the pullback is applied to transfer the
tangential and normal boundary conditions to admissible boundary
coordinates.
\end{remark}

The proofs of Propositions~\ref{prop:pullback-1-form}--\ref{prop:pullback-commutes-with-d}
and Corollary~\ref{cor:change-of-variables-top} are coordinate
computations using the determinant expansion of a $k$-form on a
$k$-tuple of vectors. They are reproduced in full in
Appendix~\ref{app:pullback}.

\section{The Codifferential $\delta$}
Before we begin with the section on the \emph{codifferential}, we shall recall the Stokes' theorem (see \ref{app:manifold_with_bd} in the appendix for a detailed development and proof):
\begin{theorem}
{\textbf{(Stokes)} Let $M$ be an oriented $n-$dimensional $C^2$ manifold with boundary, and $\beta$ be a $C^1$ class $n-1$ form that is compactly supported on $M$. Then \begin{equation}
    \int_M d\beta=\int_{\partial M}\beta
\end{equation} }
    
\end{theorem}

\subsection{Motivation}
Let $\alpha \in \Omega\Lambda^{k-1}$ and $\beta \in \Omega\Lambda^k$. Here we can invoke the definition of the Hodge star (\ref{hodge_and_inner}) to get $$\langle d\alpha,\beta\rangle vol=d\alpha \wedge (\star \beta) $$
$$\int_{\Omega} \langle d\alpha,\beta\rangle vol=\int_{\Omega} \Big{(}d\alpha \wedge (\star \beta)+(-1)^{k-1}\alpha \wedge d(\star \beta)+(-1)^k \alpha \wedge d(\star \beta) \Big{)}$$
Using the Antiderivation identity \ref{exterior_derivative_antiderivation}
we get 
\begin{equation}
\int_{\Omega} \langle d\alpha,\beta\rangle vol=\int_{\Omega} \Big{(}d(\alpha \wedge \star \beta)+(-1)^k \alpha \wedge d(\star \beta) \Big{)}\end{equation}
$$\int_{\Omega} \langle d\alpha,\beta\rangle vol= \Big{(}\int_{\partial\Omega} \alpha \wedge (\star \beta)+(-1)^k\int_{\Omega} \alpha \wedge d(\star \beta) \Big{)} $$
Since $\star \beta$ is a $n-k$ form and $d(\star \beta)$ is an $n-(k-1)$ form, we apply the definition of Hodge star again to replace $\alpha \wedge d(\star \beta)$ with $(-1)^{(n-(k-1))(k-1)}\langle \alpha,\star d\star \beta \rangle vol$ to get: 
\begin{equation}\label{diff_codiff_boundary_integral}\int_{\Omega} \langle d\alpha,\beta\rangle vol= \Big{(}\int_{\partial\Omega} \alpha \wedge (\star \beta)+(-1)^k(-1)^{(n-(k-1))(k-1)}\int_{\Omega} \langle \alpha, \star d(\star \beta) \rangle vol\Big{)} \end{equation}
Note that \eqref{diff_codiff_boundary_integral} is a formula that is independent of the definition for \emph{Codifferential} that will follow. This is important because different texts (cf. \cite{Jost2017} and \cite{CsatoDacorognaKneuss2012}) opt for different sign conventions for the Codifferential.

\defn{Codifferential}{\label{codiff}We define a map $\delta: \oplus_{k=0}^{\infty}\Omega\Lambda^k\to \oplus_{k=0}^{\infty}\Omega\Lambda^k$ given by \begin{equation}
\delta(\omega):=(-1)^{n(k-1)}\star d\star (\omega)
\end{equation} for a $k-$form $\omega$.}
Using \ref{codiff} in \eqref{diff_codiff_boundary_integral} we immediately see that \begin{equation}\label{diff_codiff_boundary_integral_final}
    \int_{\Omega} \langle d\alpha,\beta\rangle \mathrm{vol}=(-1)\int_{\Omega} \langle \alpha,\delta\beta\rangle\mathrm{vol} + \int_{\partial\Omega}\alpha\wedge \star \beta
\end{equation}
This is called the \emph{integration by parts formula} for forms.
\begin{remark}
1. Under the conventions of \ref{codiff}, the formula \ref{diff_codiff_boundary_integral_final} takes the above form. But if we propose that $\delta=(-1)^{nk+n+1}\star d \star$, or $(-1)^{k(n-k+1)}\star d\star$, we see different possible factors in \ref{diff_codiff_boundary_integral_final}.
\\2. If we assume that our set either has no boundary or that the forms vanish on the boundary, the formula \ref{diff_codiff_boundary_integral_final} yields \begin{equation}\label{diff_codiff_compact}
    \int_{\Omega} \langle d\alpha,\beta\rangle + \int_{\Omega}\langle \alpha,\delta \beta \rangle=0
\end{equation}
\end{remark}

\subsection{Coordinate formulas for the Codifferential}
\textbf{Case 1:}
Let $\omega=dx^I$. Where $I\in \mathscr T_k(\bbR^n)$. Then $\delta(\omega)=(-1)^{n(k-1)}\star d\star (dx^I)$ gives us $$(-1)^{n(k-1)} \operatorname{sgn}(R(I,I^c))\star (d (dx^{I^c})) =0$$
\\ \textbf{Case 2:}  Consider the case where $\omega=fdx^I$.
For ease sake, let $(-1)^{n(k-1)}:=\kappa$. We then have $\delta(\omega)=\kappa\star d\star (fdx^I)=\kappa\star d (f \mathrm{sgn}(R(I,I^c))dx^{I^c})$
which gives $$\kappa \mathrm{sgn}(R(I,I^c)) \star d(fdx^{I^c})= \kappa\mathrm{sgn}(R(I,I^c)) \star \Big{(}\sum_{q=1}^n \frac{df}{dx^q} dx^q\wedge dx^{I^c} \Big{)}  $$
since the summand is $0$ for when $q\not\in I$ we have 
\begin{equation}\label{formula_for_codiff_step}\delta(fdx^I)=\kappa\mathrm{sgn}(R(I,I^c)) \Big{(}\sum_{\gamma=1}^k \frac{df}{dx^{i_{\gamma}}} \star \big{(}dx^{i_{\gamma}}\wedge dx^{I^c}\big{)} \Big{)} \end{equation}

}

    Note that, if we adopted a different convention for $\delta$ (like $\delta=(-1)^{k(n-k+1)}$) we would then have the following formula instead of \eqref{formula_codiff_fdxI_case}:
    \begin{equation}\delta(fdx^I)=(-1)^n\Big{(}\sum_{\gamma=1}^k (-1)^{\gamma-1}\frac{\partial f}{\partial x^{i_{\gamma}}}dx^{I_{\widehat i_{\gamma}}} \Big{)} \end{equation}
\end{remark}
\noindent\textbf{General Case:}
Let $\omega=\sum_{I\in \mathscr T_k(\bbR^n)} f^I dx^I $
Then \begin{equation}\label{codiff_expansion_formula_general_kform}
    \delta(\sum_{I\in \mathscr T_k}f^I dx^I)=\sum_{I\in \mathscr T_k} \delta(f^I dx^I)=\sum_{I\in \mathscr T_k}\sum_{\gamma=1}^k (-1)^{\gamma-1}\frac{\partial f^I}{\partial x^{i_{\gamma}}} dx^{I_{\widehat i_{\gamma}}} 
\end{equation}

\subsubsection{Alternative Formula for the Codifferential:}

$$=\kappa \star \Big{(}\sum_{1\leq i_1<\cdots<i_k\leq n}\mathrm{sgn}(R(I,I^c))d\omega^{i_1\cdots i_k} \wedge dx^{I^c}  \Big{)}= \kappa \star \Big{(}\sum_{1\leq i_1<\cdots<i_k\leq n}\mathrm{sgn}(R(I,I^c))\big{(}\sum_{q=1}^n\frac{\partial \omega^{i_1i_2\cdots i_k}}{\partial x^q}dx^q\big{)} \wedge dx^{I^c}  \Big{)} $$

\section{Boundary Calculus and Tangential and Normal Components}\label{chap:tangential-normal}
This section draws primarily on \cite{CsatoDacorognaKneuss2012} (see Section 3.2), though we have modified and provided, whenever possible, simpler proofs of our own. In this section we will define the tangential and normal components of $f$ at the boundary, and various identites that will help us provide a consistent notion of boundary conditions for the PDEs that we shall consider in the sequel. Throughout this section, let $\Omega\subset \bbR^n$ is a bounded open $C^1$ set and $\nu(x)$ the exterior unit normal at $x\in \partial \Omega$. Let $f\in C^0(\overline{\Omega};\Lambda^k)$ be a $C^0$ $k-$form on the closure of $\Omega$. 
\subsection{Tangential and Normal components}
Let $\nu(x)$ be the exterior unit normal. Since the metric is Euclidean, we can identify $\nu(x)$, a vector field, with $\nu^{\flat}_{(x)}$, a $1-$form, in the usual sense: $$\nu(x)=\sum_{i=1}^n \nu_i e_i \xrightleftharpoons[\sharp \text{ map}]{\flat \text{ map}} \nu^{\flat}_{(x)}= \sum_{i=1}^n \nu_i dx^i $$ 

\defn{Tangential and Normal component}{\label{defn:1-tangential_and_normal}Let $f$ be a $k-$form. The tangential component of $f$ on $\partial \Omega$ is the $(k+1)-$form $$\nu^{\flat} \wedge f.$$
Likewise the normal component of $f$ on $\partial\Omega$ is the $(k-1)-$form $$\nu^{\flat}\lrcorner f=\imath_{\nu}(f) $$}

\defn{Admissible boundary coordinate system}{Let $U,V$ and $\Omega$ be open in $\bbR^n$. Let $\Omega$ be at least a $C^1$ set. A coordinate system $\phi:U\to V$ around a boundary point is said to be an \emph{admissible boundary coordinate system} if the following holds: 
\begin{enumerate}
    \item $\phi$ is $C^1$ diffeomorphism.
    \item The boundary is locally $\partial \bbH^n$, i.e.  $$\partial \Omega\cap V=\{\phi(y',0):(y',0)\in U\}$$
    \item For every $1\leq i\leq n$ and for all $(y',0)\in U$, $$\langle \partial_{y^i}\phi(y',0),\partial_{y^n}\phi(y',0)\rangle=\delta_{in} $$
\end{enumerate}
 }
\begin{remark}
    Suppose $\phi$ is such a boundary chart. Then for any $p\in \partial \Omega$, $p=\phi(y',0)$. Let $\gamma$ be a curve on $\partial \Omega$ passing $p$ at $t_0$. $\gamma(t)=\phi(y'(t),0)$ for a curve $y'(t)$ on $\bbR^{n-1}$. Taking the derivative of the curve at $t_0$, we note that (since $T_p(\partial \Omega)$ is $n-1$ dimensioned) $\partial_{y^1}\phi(y',0),\partial_{y^2}\phi(y',0),\cdots,\partial_{y^{n-1}}\phi(y',0)$ forms a basis for $T_p(\partial \Omega)$. From the 3rd condition of the definition, we see that the unit normal at $p$, $\nu(p)$ is $\pm\partial_{y^n}(\phi)$.
\end{remark}

\end{equation*}
is the unit normal at the boundary point $\phi(y',0)$. 
In other words, the pushforward of $\partial_{y^n}$ through $\phi$ gives us $\nu$, the unit normal at the boundary. 
\\\\\textbf{Observation 2:} Recall that for $1$-forms,
\begin{equation}
    (\phi^*\nu^\flat)_y(\overline{X})
    =\nu^\flat_{\phi(y)}(D\phi_y(\overline{X})).
\end{equation}
On basis vectors, we have
\begin{align}
    (\phi^*\nu^\flat)_y\left(\frac{\partial}{\partial y^i}\right)
    &=\nu^\flat_{\phi(y)}
    \left(D\phi_y\left(\frac{\partial}{\partial y^i}\right)\right)\\
    &=\left\langle
    \nu,D\phi_y\left(\frac{\partial}{\partial y^i}\right)
    \right\rangle \nonumber\\
    &=\left\langle
    D\phi_y\left(\frac{\partial}{\partial y^n}\right),
    D\phi_y\left(\frac{\partial}{\partial y^i}\right)
    \right\rangle \nonumber\\
    &=\delta_{in}. \nonumber
\end{align}
But this implies
\begin{equation*}
    {
    \phi^*\nu^\flat=dy^n
    \quad\text{on }U\cap\{y^n=0\}.
    }
\end{equation*}
Therefore, the pullback of the $1-$form associated with $\nu$ is the $1-$form $dy^n$
\\\\\textbf{Observation 3:} Let $\omega$ be any $k$-form on $V$. We have
\begin{align*}
    \phi^*(\imath_\nu\omega)_y(X_1,\ldots,X_{k-1})
    &=(\imath_\nu\omega)_{\phi(y)}
    (D\phi_yX_1,\ldots,D\phi_yX_{k-1})\\
    &=\omega_{\phi(y)}
    (\nu,D\phi_yX_1,\ldots,D\phi_yX_{k-1})\\
    &=\omega_{\phi(y)}
    \left(
    D\phi_y\frac{\partial}{\partial y^n},
    D\phi_yX_1,\ldots,D\phi_yX_{k-1}
    \right)\\
    &=(\phi^*\omega)_y
    \left(\frac{\partial}{\partial y^n},
    X_1,\ldots,X_{k-1}\right)\\
    &=\big(\imath_{\partial_{y^n}}\phi^*\omega\big)_y
    (X_1,\ldots,X_{k-1}).
\end{align*}
Therefore, the pullback of the interior of $\nu$ with $\omega$ is precisely the interior of the vector $x$ whose pushforward is $\nu$ (in this case, $x=\partial_{y^n}$) with the pullback of $\omega$. 
\\\\ With these observations in hand, the first identity follows directly:
\begin{align*}
    \phi^*\big(\imath_\nu(\nu^\flat\wedge f)\big)
    &=\imath_{\partial_{y^n}}
    \big(\phi^*(\nu^\flat\wedge f)\big)\\
    &=\imath_{\partial_{y^n}}
    \big(dy^n\wedge\phi^*(f)\big).
\end{align*}
The next identity proceeds similarly:
\begin{align*}
    \phi^*\big(\nu^\flat\wedge(\imath_\nu f)\big)
    &=\phi^*\nu^\flat\wedge\phi^*((\imath_\nu f))\\
    &=dy^n\wedge
    \big(\imath_{\partial_{y^n}}\phi^*(f)\big).
\end{align*}
}
The previous theorems allow us to give a relationship between the various notions of the tangential and normal boundary conditions. We have the following: 
\begin{corollary}\textbf{(Equivalent characterisations of tangential and normal boundary conditions)}
    $$\nu \wedge f=0 \iff tf=0 \iff t'f=0$$
    $$\nu \lrcorner f=0 \iff nf=0 \iff n'f=0.$$
(Recall that $t'f=0$ and $n'f=0$ are defined in \ref{defn:vanishing_tangential_and_normal_element}, $tf$ and $nf$ in \ref{defn:2-tangential_and_normal} and the usual notions of tangential and normal components in \ref{defn:1-tangential_and_normal}.)
\end{corollary}

\section{Function Spaces of Differential Forms}
\label{sec:function-spaces}
We take a brief detour from our prior development to define certain notions of function spaces, and fix some notations. This section can be viably skipped, without loss of continuity from the previous section.
\\\\ Throughout this section, let $\Omega\subset\bbR^n$ be an open set and let $0\leq k\leq n$. Since $\Omega$ is an open subset of $\bbR^n$, we identify $T_p\Omega\simeq\bbR^n$ for every $p\in\Omega$, so that every differential $k$-form may be viewed as a map
\[
    f:\Omega\longrightarrow\Lambda^k(\bbR^n).
\]
With respect to the standard basis of $\Lambda^k(\bbR^n)$, every $k$-form admits a unique expansion
\[
    f
    =
    \sum_{I\in\mathscr T_k} f^I\,dx^I
    =
    \sum_{1\leq i_1<\cdots<i_k\leq n}
    f^{i_1\cdots i_k}\,
    dx^{i_1}\wedge\cdots\wedge dx^{i_k},
\]
where each coefficient $f^I:\Omega\to\bbR$ is scalar-valued. Since
$\dim\Lambda^k(\bbR^n)=\binom nk$ is finite, a $k$-form is equivalently a
$\binom nk$-tuple of scalar functions, and all regularity notions below are defined coefficientwise.

\defn{$L^p$ differential forms}{Let $1\leq p\leq\infty$. We write $f\in L^p(\Omega;\Lambda^k)$ if and only if every coefficient satisfies $f^I\in L^p(\Omega)$ for $I\in\mathscr T_k$; equivalently,
\[
    L^p(\Omega;\Lambda^k)
    =
    L^p(\Omega;\bbR^{\binom nk}).
\]
The pointwise inner product and norm on $\Lambda^k(\bbR^n)$ are
\[
    \langle f,g\rangle
    =
    \sum_{I\in\mathscr T_k} f^I g^I,
    \qquad
    |f|^2
    =
    \langle f,f\rangle
    =
    \sum_{I\in\mathscr T_k} (f^I)^2,
\]
and the $L^p$-norm is
\[
    \|f\|_{L^p(\Omega;\Lambda^k)}
    =
    \Bigl(\int_\Omega |f(x)|^p\,dx\Bigr)^{1/p},
\]
with the usual modification when $p=\infty$.}

\defn{Sobolev differential forms}{Let $1\leq p<\infty$. We write $f\in W^{1,p}(\Omega;\Lambda^k)$ if and only if every coefficient satisfies $f^I\in W^{1,p}(\Omega)$ for $I\in\mathscr T_k$. Equivalently, all coefficients possess weak first-order partial derivatives lying in $L^p(\Omega)$. The pointwise gradient and its norm are
\[
    \nabla f
    =
    \sum_{i=1}^n\sum_{I\in\mathscr T_k}
    \frac{\partial f^I}{\partial x^i}\,
    dx^i\otimes dx^I,
    \qquad
    |\nabla f|^2
    =
    \sum_{i=1}^n\sum_{I\in\mathscr T_k}
    \Bigl(\frac{\partial f^I}{\partial x^i}\Bigr)^2,
\]
and the Sobolev norm is
\[
    \|f\|_{W^{1,p}(\Omega;\Lambda^k)}
    =
    \Bigl(
        \|f\|_{L^p(\Omega;\Lambda^k)}^p
        +
        \|\nabla f\|_{L^p(\Omega;\Lambda^k)}^p
    \Bigr)^{1/p}.
\]
In particular,
\[
    \|f\|_{W^{1,2}(\Omega;\Lambda^k)}^2
    =
    \|f\|_{L^2(\Omega;\Lambda^k)}^2
    +
    \|\nabla f\|_{L^2(\Omega;\Lambda^k)}^2.
\]}
Classical scalar Sobolev spaces are defined in section \ref{app:sobolev} in the appendix.
\defn{Classical regularity spaces}{For an integer $r\geq0$, we write $f\in C^r(\Omega;\Lambda^k)$ if and only if every coefficient lies in $C^r(\Omega)$. For $0<\alpha\leq1$, we write $f\in C^{r,\alpha}(\Omega;\Lambda^k)$ if and only if every coefficient lies in the H\"older space $C^{r,\alpha}(\Omega)$, equipped with the usual H\"older norm. All norms are defined coefficientwise. We define the classical scalar H\"older spaces in section \ref{app:holder} of the appendix. }

\defn{Spaces on the closure}{We write $f\in C^r(\overline\Omega;\Lambda^k)$ if every coefficient extends continuously to $\overline\Omega$ together with all derivatives of order at most $r$, and $f\in C^{r,\alpha}(\overline\Omega;\Lambda^k)$ if every coefficient extends to a function in $C^{r,\alpha}(\overline\Omega)$.}

\begin{remark}
The distinction between $C^{r,\alpha}(\Omega;\Lambda^k)$ and
$C^{r,\alpha}(\overline\Omega;\Lambda^k)$ is essential for boundary conditions. Elements of the former satisfy the H\"older condition only in the interior of $\Omega$, while elements of the latter admit continuous extensions, together with their derivatives up to order $r$, to the boundary $\partial\Omega$. Boundary conditions such as $\nu\wedge f=0$ or $\nu\lrcorner f=0$ are naturally stated in the latter setting.
\end{remark}

\begin{remark}
Since $\Lambda^k(\bbR^n)$ is finite-dimensional, all norms on it are equivalent. The particular choice of Euclidean norm used above is therefore immaterial: the $L^p$, Sobolev and H\"older norms defined here are equivalent on $\Lambda^k$-valued forms, and in particular the definition of $W^{1,p}(\Omega;\Lambda^k)$ does not depend on the choice of basis of $\Lambda^k(\bbR^n)$.
\end{remark}

\section{Tangential and normal components (cont.)}
\subsection{Tangential and normal boundary spaces}
\label{subsec:tangential-normal-spaces}

We now define the tangential and normal components of forms belonging to Sobolev spaces. Let $1\leq p\leq\infty$. If $f\in W^{1,p}(\Omega;\Lambda^k)$, then its boundary value
\[
    f\in L^p(\partial\Omega;\Lambda^k)
\]
is well defined in the sense of traces (see section~\ref{app:trace} in the appendix), and the trace map
\[
    \operatorname{Tr}:W^{1,p}(\Omega;\Lambda^k)
    \longrightarrow
    L^p(\partial\Omega;\Lambda^k)
\]
is bounded. Since the outward unit normal satisfies
$\nu\in C^0(\partial\Omega;\Lambda^1)$ for a $C^1$ boundary, the following
functions are well defined:
\[
    \nu\wedge f\in L^p(\partial\Omega;\Lambda^{k+1}),
    \qquad
    \nu\lrcorner f\in L^p(\partial\Omega;\Lambda^{k-1}),
\]
and we can now define the following spaces.

\begin{definition}[Spaces with vanishing tangential or normal component]
\label{def:WTP-WNP}
Let $r\geq0$ be an integer, $0\leq\alpha\leq1$, and $1\leq p\leq\infty$. The spaces with vanishing tangential component are
\[
    C^{r,\alpha}_T(\Omega;\Lambda^k)
    =
    \bigl\{f\in C^{r,\alpha}(\Omega;\Lambda^k):
    \nu\wedge f=0\text{ on }\partial\Omega\bigr\},
\]
\[
    W^{r+1,p}_T(\Omega;\Lambda^k)
    =
    \bigl\{f\in W^{r+1,p}(\Omega;\Lambda^k):
    \nu\wedge f=0\text{ on }\partial\Omega\bigr\}.
\]
Similarly, the spaces with vanishing normal component are
\[
    C^{r,\alpha}_N(\Omega;\Lambda^k)
    =
    \bigl\{f\in C^{r,\alpha}(\Omega;\Lambda^k):
    \nu\lrcorner f=0\text{ on }\partial\Omega\bigr\},
\]
\[
    W^{r+1,p}_N(\Omega;\Lambda^k)
    =
    \bigl\{f\in W^{r+1,p}(\Omega;\Lambda^k):
    \nu\lrcorner f=0\text{ on }\partial\Omega\bigr\}.
\]
\end{definition}

In the notation $W^{r+1,p}_T$ and $W^{r+1,p}_N$, the exponent $r+1$ is the
order of the Sobolev space, matched to the number of derivatives in
$C^{r,\alpha}$. The boundary conditions are understood in the trace sense:
for $f\in W^{r+1,p}(\Omega;\Lambda^k)$ we require
$\nu\wedge\operatorname{Tr} f=0$ or
$\nu\lrcorner\operatorname{Tr} f=0$ on $\partial\Omega$, and for
$f\in C^{r,\alpha}(\Omega;\Lambda^k)$ the trace is the ordinary
restriction.

\begin{remark}
The two boundary conditions are dual to each other under the Hodge star.
On a $C^1$ boundary, the unit normal $\nu$ satisfies
$\nu\lrcorner\nu=|\nu|^2=1$, and the Hodge star exchanges the exterior and
interior products with $\nu$ up to sign. Consequently, the condition
$\nu\wedge f=0$ on $\partial\Omega$ is equivalent to
$\nu\lrcorner(\star f)=0$ on $\partial\Omega$, so that
$\star$ maps $W^{1,p}_T(\Omega;\Lambda^k)$ isometrically onto
$W^{1,p}_N(\Omega;\Lambda^{n-k})$. We have seen an instance of this symmetry in the proof of \ref{thm:nu_wedge_f_nu_wedge_df} (in proving $(2)$ from $(1)$).This symmetry is used to simplify the proof of the Gaffney inequality, where it allows the tangential and
normal cases to be treated by a single argument.
\end{remark}

We state, without proof, a density result of considerable importance.

\begin{theorem}\label{thm:density_tangential_normal}
Let $r\geq 1$ be an integer, $1\leq p<\infty$, and
$\Omega\subset\bbR^n$ be a bounded open $C^{r+1}$ set.
Then the following two statements hold:
\begin{enumerate}
    \item[(i)] $C^r_T(\Omega;\Lambda^k)$ is dense in
    $W^{1,p}_T(\Omega;\Lambda^k)$.
    \item[(ii)] $C^r_N(\Omega;\Lambda^k)$ is dense in
    $W^{1,p}_N(\Omega;\Lambda^k)$.
\end{enumerate}
\end{theorem}
\subsubsection{Gauss-Green theorem}
We shall assume that $\Omega\subset\bbR^n$ is a bounded open
sufficiently regular set so that integration by parts can be performed.
In most cases, however, we shall require that $\Omega$ is at least
$C^2$. We present the Gauss--Green theorem first.

\begin{theorem}[Gauss--Green theorem]\label{thm:gauss_green}
Let $0\leq k\leq n$ and $f\in C^1(\overline{\Omega};\Lambda^k)$.
Then
\[
    \int_\Omega
    \frac{\partial f_{i_1\cdots i_k}}{\partial x_j}\,dx
    =
    \int_{\partial\Omega}
    \nu^j f_{i_1\cdots i_k}\,dS,
\]
for all $1\leq i_1<\cdots<i_k\leq n$ and $1\leq j\leq n$.
\\\\ As a consequence, we have the following identities:
\begin{equation}
    \int_{\Omega}df=\int_{\partial\Omega}\nu\wedge f \text{ and } \int_{\Omega}\delta f=\int_{\partial\Omega}\nu\lrcorner f.
\end{equation}
\end{theorem}
\pf{The first assertion is just the classical Gauss-Green formula. We shall prove the identities that follow it. 
Recall the coordinate formula \ref{coordinate_formula_for_df} for $df$:

\begin{equation}
    d\omega=\sum_{i=1}^n \sum_{I\in \mathscr T_k}\frac{\partial\omega^I}{\partial x^i} dx^i\wedge dx^I= \sum_{I\in \mathscr T_{k+1}}\sum_{\gamma=1}^{k+1}(-1)^{\gamma-1}\frac{\partial\omega^{I_{\hat i_{\gamma}}}}{\partial x^{i_{\gamma}}} dx^I
\end{equation}
From the classical formula, we have \begin{align*}
    \int_{\Omega} (df)_{i_1 i_2\cdots i_{k+1}}&=\int_{\Omega} \sum_{\gamma=1}^{k+1}(-1)^{\gamma-1} \frac{\partial f_{i_1 i_2 \cdots i_{\gamma-1}i_{\gamma+1}\cdots i_{k+1}}}{\partial i_{\gamma}}\\&=\int_{\partial\Omega}\sum_{\gamma=1}^{k-1} (-1)^{\gamma-1}\nu^{i_{\gamma}}f_{i_1\cdots i_{\gamma-1}i_{\gamma+1}\cdots i_{k+1}}=\int_{\partial\Omega}(\nu\wedge f)_{i_1 i_2 \cdots i_{k+1}}.
\end{align*}
The analog result for the codifferential is proved similarly, noting the coordinate identities: \ref{codiff_expansion_formula_general_kform} and \ref{eqn:coordinate_expression_for_interior}. 
}

\begin{theorem}[Integration-by-parts formula]
\label{thm:integration_by_parts}
Let $1\leq k\leq n$. Let
$f\in C^1(\overline{\Omega};\Lambda^{k-1})$ and
$g\in C^1(\overline{\Omega};\Lambda^k)$. Then
\[
    \int_\Omega \langle df,g\rangle
    +\int_\Omega \langle f,\delta g\rangle
    =
    \int_{\partial\Omega}
    \langle \nu^\flat\wedge f,g\rangle
    =
    \int_{\partial\Omega}
    \langle f,\imath_\nu g\rangle.
\]
\end{theorem}

\begin{remark}
By the density theorem, the above integration-by-parts formula
continues to hold for Sobolev differential forms, where the exterior
derivative and codifferential are defined coordinate-wise, with the
partial derivatives understood in the weak sense.
\end{remark} 
\pf{The last equality is trivial from the duality of interior and wedge. For the first, recall from our derivation \ref{diff_codiff_boundary_integral_final} that
    $$\int_{\Omega} \langle df,g\rangle+\int_{\Omega}\langle f,\delta g\rangle=\int_{\partial\Omega} f\wedge \star g. $$
Therefore, we need only show that $\int_{\partial\Omega} f\wedge \star g =\int_{\partial\Omega} \langle \nu \wedge f,g\rangle$. 
Indeed, an application of Stokes' theorem, the Gauss--Green formula and the definition of the Hodge star together gives us
\begin{align*}
    \int_{\partial\Omega}f\wedge \star g=\int_{\Omega} d(f\wedge \star g)=\int_{\partial\Omega} \nu \wedge f\wedge\star g=\int_{\partial\Omega} \langle \nu \wedge f,g\rangle
\end{align*} and we have the result.}
From the \emph{integration by parts} theorem \ref{thm:integration_by_parts}, we can also give an equivalent characterisation of the spaces
$W^{1,p}_T(\Omega;\Lambda^k)$ and
$W^{1,p}_N(\Omega;\Lambda^k)$ (and similarly for the Hölder spaces).
Indeed, the space $W^{1,p}_T(\Omega;\Lambda^k)$ consists precisely
of those $f\in W^{1,p}(\Omega;\Lambda^k)$ satisfying
\[
    \int_\Omega \langle df,\varphi\rangle
    +\int_\Omega \langle f,\delta\varphi\rangle
    =0,
    \qquad
    \forall\varphi\in C^\infty(\overline{\Omega};\Lambda^{k+1}).
\]
Similarly, the space $W^{1,p}_N(\Omega;\Lambda^k)$ consists precisely
of those $f\in W^{1,p}(\Omega;\Lambda^k)$ satisfying
\[
    \int_\Omega \langle \delta f,\varphi\rangle
    +\int_\Omega \langle f,d\varphi\rangle
    =0,
    \qquad
    \forall\varphi\in C^\infty(\overline{\Omega};\Lambda^{k-1}).
\]

\begin{corollary}\label{cor:orthogonality}
Let $1\leq k\leq n$ and $1<p,q<\infty$ be such that
$\frac{1}{p}+\frac{1}{q}=1$. Let $\Omega\subset\bbR^n$ be a
bounded open $C^2$ set. Suppose that
$f\in W^{1,p}(\Omega;\Lambda^{k-1})$ and
$g\in W^{1,q}(\Omega;\Lambda^{k+1})$ satisfy either
$\nu\wedge f=0$ or $\nu\lrcorner g=0$ on $\partial\Omega$.
Then
\[
    \int_\Omega \langle df,\delta g\rangle=0.
\]
\end{corollary}

\begin{proof}
We consider the two boundary conditions separately. In both cases,
the result follows by approximation and the integration-by-parts
formula.

\medskip
\noindent
\textbf{Case 1.} Suppose that $\nu\wedge f=0$ on $\partial\Omega$.
By the density theorem, there exists a sequence
$f_m\in C^1_T(\Omega;\Lambda^{k-1})$ such that
$f_m\to f$ in $W^{1,p}(\Omega;\Lambda^{k-1})$.
Furthermore, since smooth forms are dense in
$W^{1,q}(\Omega;\Lambda^{k+1})$, we may choose
$g_m\in C^\infty(\overline{\Omega};\Lambda^{k+1})$ such that
$g_m\to g$ in $W^{1,q}(\Omega;\Lambda^{k+1})$.

Applying the integration-by-parts formula to $f_m$ and
$\delta g_m$, and using $\delta^2=0$, we obtain
\[
\begin{aligned}
    \int_\Omega\langle df_m,\delta g_m\rangle
    &=\int_{\partial\Omega}
    \langle\nu\wedge f_m,\delta g_m\rangle\\
    &=0.
\end{aligned}
\]
Passing to the limit, by H\"older's inequality, gives
\[
    \int_\Omega\langle df,\delta g\rangle=0.
\]

\medskip
\noindent
\textbf{Case 2.} Suppose that $\nu\lrcorner g=0$ on
$\partial\Omega$. By the density theorem, there exists a sequence
$g_m\in C^1_N(\Omega;\Lambda^{k+1})$ such that
$g_m\to g$ in $W^{1,q}(\Omega;\Lambda^{k+1})$.
Also, choose
$f_m\in C^\infty(\overline{\Omega};\Lambda^{k-1})$ such that
$f_m\to f$ in $W^{1,p}(\Omega;\Lambda^{k-1})$.

Applying the integration-by-parts formula to $df_m$ and $g_m$,
we obtain
\[
\begin{aligned}
    \int_\Omega\langle df_m,\delta g_m\rangle
    &=\int_{\partial\Omega}
    \langle\nu\wedge df_m,g_m\rangle\\
    &=\int_{\partial\Omega}
    \langle df_m,\nu\lrcorner g_m\rangle\\
    &=0,
\end{aligned}
\]
where we have used $d^2f_m=0$ and the boundary condition
$\nu\lrcorner g_m=0$.

Once again, passing to the limit by H\"older's inequality yields
\[
    \int_\Omega\langle df,\delta g\rangle=0.
\]
This completes the proof.
\end{proof}

\subsection{The Gaffney inequality}
The main result of this section is the following inequality:
{\color{purple}
\begin{theorem}[Gaffney inequality]
\label{thm:gaffney}
Let $\Omega\subset\R^n$ be a bounded open $C^2$ set. Then there exists a
constant $C=C(\Omega)>0$ such that
\begin{equation}
    \|\omega\|_{W^{1,2}(\Omega)}
    \leq
    C\left(
        \|d\omega\|_{L^2(\Omega)}
        +\|\delta\omega\|_{L^2(\Omega)}
        +\|\omega\|_{L^2(\Omega)}
    \right)
\end{equation}
for every
\[
    \omega\in
    W^{1,2}_{T}(\Omega;\Lambda^k)\cup
    W^{1,2}_{N}(\Omega;\Lambda^k).
\]
\end{theorem}
}
This chapter heavily draws from the paper \cite{Csato_2018}, and parts of chapter $6$ of \cite{CsatoDacorognaKneuss2012}.
We begin with fixing some notation: Recall the multi-index notation set up in \ref{def:multi_index}. 

\begin{lemma}
    Let $n\geq 2$ and $1\leq k\leq n-1$ be integers. Then for any $I\in \mathscr T_{k+1}$ and $i,j \in I$, with $i\neq j$ we have \begin{equation}
\operatorname{sign}[i,I_{\hat{i}\hat{j}}]\operatorname{sign}[j,I_{\hat{i}\hat{j}}]=-\operatorname{sign}[i,I_{\hat{i}}]\operatorname{sign}[j,I_{\hat{j}}]
\end{equation}
And and for any $I\in \mathscr T_{k-1}$ and every $i,j \not\in I$ with $i\neq j$,\begin{equation}
    \operatorname{sign}[i,[jI]]\operatorname{sign}[j,[iI]]=-\operatorname{sign}[i,I]\operatorname{sign}[j,I]
\end{equation}

\end{lemma} 
The proof of the lemma is discussed in \ref{lem:multi_index}

\begin{lemma}
\label{lemma:pointwise_identity}
Let $U\subset\mathbb{R}^n$ be open and let
\[
    \omega\in C^1(U;\Lambda^k).
\]
Then, for every $x\in U$,
\[
\begin{aligned}\label{lem23_1}
    |d\omega|^2+|\delta\omega|^2-|\nabla\omega|^2
    &=
    \sum_{I\in\mathscr{T}_{k+1}}
    \sum_{\substack{i,j\in I\\i\neq j}}
    \operatorname{sgn}[i,\widehat{I}_i]\,
    \operatorname{sgn}[j,\widehat{I}_j]
    \left(
        \frac{\partial\omega_{\widehat{I}_i}}{\partial x^i}
        \frac{\partial\omega_{\widehat{I}_j}}{\partial x^j}
        -
        \frac{\partial\omega_{\widehat{I}_j}}{\partial x^i}
        \frac{\partial\omega_{\widehat{I}_i}}{\partial x^j}
    \right)
\end{aligned}
\tag{17}
\]
and
\[
\begin{aligned}\label{lem23_2}
    |d\omega|^2+|\delta\omega|^2-|\nabla\omega|^2
    &=
    \sum_{I\in\mathscr{T}_{k-1}}
    \sum_{\substack{i,j\notin I\\i<j}}
    \operatorname{sgn}[i,I]\,
    \operatorname{sgn}[j,I]
    \left(
        \frac{\partial\omega_{[iI]}}{\partial x^i}
        \frac{\partial\omega_{[jI]}}{\partial x^j}
        -
        \frac{\partial\omega_{[jI]}}{\partial x^i}
        \frac{\partial\omega_{[iI]}}{\partial x^j}
    \right).
\end{aligned}
\tag{18}
\]
\end{lemma}
\begin{remark}
    Here, the quantities $|d\omega|$, $|\delta \omega|$ and $|\nabla \omega|$ are \emph{pointwise} norms on the forms. 
\end{remark}

\begin{proof}
From the component formulas for $d$ and $\delta$,
\[
    d\omega
    =
    \sum_{I\in\mathscr{T}_{k+1}}
    \left(
        \sum_{i\in I}
        \operatorname{sgn}[i,\widehat{I}_i]\,
        \frac{\partial\omega_{\widehat{I}_i}}
        {\partial x^i}
    \right)dx^I
\]
and
\[
    \delta\omega
    =
    \sum_{I\in\mathscr{T}_{k-1}}
    \left(
        \sum_{i\notin I}
        \operatorname{sgn}[i,I]\,
        \frac{\partial\omega_{[iI]}}
        {\partial x^i}
    \right)dx^I.
\]

Expanding the squares gives
\[
\begin{aligned}
|d\omega|^2
&=
\sum_{I\in\mathscr{T}_{k+1}}
\sum_{i\in I}
\left(
    \frac{\partial\omega_{\widehat{I}_i}}
    {\partial x^i}
\right)^2
\\
&\quad+
2\sum_{I\in\mathscr{T}_{k+1}}
\sum_{\substack{i,j\in I\\i<j}}
\operatorname{sgn}[i,\widehat{I}_i]\,
\operatorname{sgn}[j,\widehat{I}_j]\,
\frac{\partial\omega_{\widehat{I}_i}}{\partial x^i}
\frac{\partial\omega_{\widehat{I}_j}}{\partial x^j},
\end{aligned}
\tag{19}
\]
while
\[
\begin{aligned}
|\delta\omega|^2
&=
\sum_{I\in\mathscr{T}_{k-1}}
\sum_{i\notin I}
\left(
    \frac{\partial\omega_{[iI]}}
    {\partial x^i}
\right)^2
\\
&\quad+
2\sum_{I\in\mathscr{T}_{k-1}}
\sum_{\substack{i,j\notin I\\i<j}}
\operatorname{sgn}[i,I]\,
\operatorname{sgn}[j,I]\,
\frac{\partial\omega_{[iI]}}{\partial x^i}
\frac{\partial\omega_{[jI]}}{\partial x^j}.
\end{aligned}
\tag{20}
\]

The diagonal terms in these two expressions together give (after first suitably converting the sum over $\mathscr T_{k-1}$ to a sum over $\mathscr T_{k+1}$ by noting that for every $I_0$ in $\mathscr T_{k-1}$, $i$ and $j$ not in $I_0$, we have a \emph{unique} $J=[I_0,i,j]\in \mathscr T_{k+1}$)
\[
    |\nabla\omega|^2
    =
    \sum_{I\in\mathscr{T}_k}
    \sum_{i=1}^n
    \left(
        \frac{\partial\omega_I}{\partial x^i}
    \right)^2.
\]
Consequently, after cancelling the diagonal terms, the remaining
cross terms can be written as
\[
\begin{aligned}
|d\omega|^2+|\delta\omega|^2-|\nabla\omega|^2
={}&
2\sum_{I\in\mathscr{T}_{k+1}}
\sum_{\substack{i,j\in I\\i<j}}
\Bigg[
\operatorname{sgn}[i,\widehat{I}_i]\,
\operatorname{sgn}[j,\widehat{I}_j]\,
\frac{\partial\omega_{\widehat{I}_i}}{\partial x^i}
\frac{\partial\omega_{\widehat{I}_j}}{\partial x^j}
\\
&\qquad\qquad\qquad+
\operatorname{sgn}[i,\widehat{I}_{ij}]\,
\operatorname{sgn}[j,\widehat{I}_{ij}]\,
\frac{\partial\omega_{\widehat{I}_j}}{\partial x^i}
\frac{\partial\omega_{\widehat{I}_i}}{\partial x^j}
\Bigg].
\end{aligned}
\]

By Lemma~\ref{lem:multi_index},
\[
    \operatorname{sgn}[i,\widehat{I}_{ij}]
    \operatorname{sgn}[j,\widehat{I}_{ij}]
    =
    -
    \operatorname{sgn}[i,\widehat{I}_i]
    \operatorname{sgn}[j,\widehat{I}_j],
\]
which gives \eqref{lem23_1}.

Alternatively, regrouping the cross terms according to
$I\in\mathscr{T}_{k-1}$ and $i,j\notin I$ gives
\[
\begin{aligned}
|d\omega|^2+|\delta\omega|^2-|\nabla\omega|^2
={}&
2\sum_{I\in\mathscr{T}_{k-1}}
\sum_{\substack{i,j\notin I\\i<j}}
\Bigg[
\operatorname{sgn}[i,I]\,
\operatorname{sgn}[j,I]\,
\frac{\partial\omega_{[iI]}}{\partial x^i}
\frac{\partial\omega_{[jI]}}{\partial x^j}
\\
&\qquad\qquad\qquad+
\operatorname{sgn}[i,[jI]]\,
\operatorname{sgn}[j,[iI]]\,
\frac{\partial\omega_{[jI]}}{\partial x^i}
\frac{\partial\omega_{[iI]}}{\partial x^j}
\Bigg].
\end{aligned}
\]
The second identity of Lemma~\ref{lem:multi_index} gives
\[
    \operatorname{sgn}[i,[jI]]
    \operatorname{sgn}[j,[iI]]
    =
    -
    \operatorname{sgn}[i,I]\operatorname{sgn}[j,I],
\]
and hence \eqref{lem23_2}.
\end{proof}

\begin{lemma}
\label{lemma:boundary_identity}
Let $\Omega\subset\mathbb{R}^n$ be a bounded open Lipschitz domain and let
$\omega\in C^1(\overline{\Omega};\Lambda^k)$. Then
\begin{equation}
\label{eq:boundary_identity_wedge}
\|d\omega\|_{L^2}^2+\|\delta\omega\|_{L^2}^2-\|\nabla\omega\|_{L^2}^2
=
\int_{\partial\Omega}\langle \nu\wedge\omega,d\omega\rangle
-
\int_{\partial\Omega}
\sum_{I\in\mathscr T_{k+1}}
\sum_{\substack{i,j\in I\\ i\neq j}}
\operatorname{sgn}[i,\widehat I_i]
\operatorname{sgn}[j,\widehat I_j]
\,\omega_{\widehat I_j}\,
\nu_i
\frac{\partial\omega_{\widehat I_i}}{\partial x^j}.
\end{equation}

Similarly,
\begin{equation}
\label{eq:boundary_identity_contraction}
\|d\omega\|_{L^2}^2+\|\delta\omega\|_{L^2}^2-\|\nabla\omega\|_{L^2}^2
=
\int_{\partial\Omega}\langle \nu\lrcorner\omega,\delta\omega\rangle
-
\int_{\partial\Omega}
\sum_{I\in\mathscr T_{k-1}}
\sum_{\substack{i,j\notin I\\ i<j}}
\operatorname{sgn}[i,I]\operatorname{sgn}[j,I]\,
\omega_{[iI]}\,
\nu_j
\frac{\partial\omega_{[jI]}}{\partial x^i}.
\end{equation}
\end{lemma}

\begin{proof}
We prove \eqref{eq:boundary_identity_wedge}. The proof of
\eqref{eq:boundary_identity_contraction} is identical, starting from the second
identity of Lemma \ref{lemma:pointwise_identity}.

By Lemma \ref{lemma:pointwise_identity},
\[
|d\omega|^2+|\delta\omega|^2-|\nabla\omega|^2
=
\sum_{I\in\mathscr T_{k+1}}
\sum_{\substack{i,j\in I\\ i\neq j}}
\operatorname{sgn}[i,\widehat I_i]
\operatorname{sgn}[j,\widehat I_j]
\left(
\frac{\partial\omega_{\widehat I_i}}{\partial x^i}
\frac{\partial\omega_{\widehat I_j}}{\partial x^j}
-
\frac{\partial\omega_{\widehat I_j}}{\partial x^i}
\frac{\partial\omega_{\widehat I_i}}{\partial x^j}
\right).
\]
Integrating over $\Omega$ gives
\begin{equation}
\label{eq:integrated_pointwise_identity}
\begin{aligned}
\|d\omega\|_{L^2}^2+\|\delta\omega\|_{L^2}^2-\|\nabla\omega\|_{L^2}^2
&=
\int_\Omega
\sum_{I\in\mathscr T_{k+1}}
\sum_{\substack{i,j\in I\\ i\neq j}}
\operatorname{sgn}[i,\widehat I_i]
\operatorname{sgn}[j,\widehat I_j] \\
&\qquad\qquad\cdot
\left(
\frac{\partial\omega_{\widehat I_i}}{\partial x^i}
\frac{\partial\omega_{\widehat I_j}}{\partial x^j}
-
\frac{\partial\omega_{\widehat I_j}}{\partial x^i}
\frac{\partial\omega_{\widehat I_i}}{\partial x^j}
\right).
\end{aligned}
\end{equation}

For each pair $(I,i,j)$, the product rule gives
\[
\frac{\partial}{\partial x^j}
\left(
\omega_{\widehat I_j}
\frac{\partial\omega_{\widehat I_i}}{\partial x^i}
\right)
=
\frac{\partial\omega_{\widehat I_j}}{\partial x^j}
\frac{\partial\omega_{\widehat I_i}}{\partial x^i}
+
\omega_{\widehat I_j}
\frac{\partial^2\omega_{\widehat I_i}}{\partial x^j\partial x^i},
\]
and similarly,
\[
\frac{\partial}{\partial x^i}
\left(
\omega_{\widehat I_j}
\frac{\partial\omega_{\widehat I_i}}{\partial x^j}
\right)
=
\frac{\partial\omega_{\widehat I_j}}{\partial x^i}
\frac{\partial\omega_{\widehat I_i}}{\partial x^j}
+
\omega_{\widehat I_j}
\frac{\partial^2\omega_{\widehat I_i}}{\partial x^i\partial x^j}.
\]
Since mixed partial derivatives commute for $C^2$ functions,
\[
\frac{\partial^2\omega_{\widehat I_i}}{\partial x^j\partial x^i}
=
\frac{\partial^2\omega_{\widehat I_i}}{\partial x^i\partial x^j},
\]
we obtain
\begin{equation}
\label{eq:divergence_identity}
\frac{\partial\omega_{\widehat I_i}}{\partial x^i}
\frac{\partial\omega_{\widehat I_j}}{\partial x^j}
-
\frac{\partial\omega_{\widehat I_j}}{\partial x^i}
\frac{\partial\omega_{\widehat I_i}}{\partial x^j}
=
\frac{\partial}{\partial x^j}
\left(
\omega_{\widehat I_j}
\frac{\partial\omega_{\widehat I_i}}{\partial x^i}
\right)
-
\frac{\partial}{\partial x^i}
\left(
\omega_{\widehat I_j}
\frac{\partial\omega_{\widehat I_i}}{\partial x^j}
\right).
\end{equation}

Substituting \eqref{eq:divergence_identity} into
\eqref{eq:integrated_pointwise_identity} and applying the Gauss--Green theorem
componentwise yields
\[
\begin{aligned}
\|d\omega\|_{L^2}^2+\|\delta\omega\|_{L^2}^2-\|\nabla\omega\|_{L^2}^2
&=
\int_{\partial\Omega}
\sum_{I\in\mathscr T_{k+1}}
\sum_{\substack{i,j\in I\\ i\neq j}}
\operatorname{sgn}[i,\widehat I_i]
\operatorname{sgn}[j,\widehat I_j] \\
&\qquad\cdot
\left(
\omega_{\widehat I_j}\nu_j
\frac{\partial\omega_{\widehat I_i}}{\partial x^i}
-
\omega_{\widehat I_j}\nu_i
\frac{\partial\omega_{\widehat I_i}}{\partial x^j}
\right).
\end{aligned}
\]
The identity remains valid for $\omega\in C^1(\overline\Omega;\Lambda^k)$ by
density.

Since $\Omega$ is Lipschitz, the outward unit normal $\nu$ is defined almost
everywhere on $\partial\Omega$. Using the coordinate formula for the exterior
product,
\[
\nu\wedge\omega
=
\sum_{I\in\mathscr T_{k+1}}
\left(
\sum_{j\in I}
\operatorname{sgn}[j,\widehat I_j]\,
\nu_j\omega_{\widehat I_j}
\right)
dx^I,
\]
while the coordinate expression for the exterior derivative is
\[
d\omega
=
\sum_{I\in\mathscr T_{k+1}}
\left(
\sum_{i\in I}
\operatorname{sgn}[i,\widehat I_i]
\frac{\partial\omega_{\widehat I_i}}{\partial x^i}
\right)
dx^I.
\]
Taking the inner product of these two $(k+1)$-forms gives
\[
\langle\nu\wedge\omega,d\omega\rangle
=
\sum_{I\in\mathscr T_{k+1}}
\sum_{i,j\in I}
\operatorname{sgn}[i,\widehat I_i]
\operatorname{sgn}[j,\widehat I_j]
\,\omega_{\widehat I_j}\,
\nu_j
\frac{\partial\omega_{\widehat I_i}}{\partial x^i}.
\]
Separating the diagonal terms from the terms with $i\neq j$,
\[
\begin{aligned}
\langle\nu\wedge\omega,d\omega\rangle
&=
\sum_{I\in\mathscr T_{k+1}}
\sum_{i\in I}
\omega_{\widehat I_i}\nu_i
\frac{\partial\omega_{\widehat I_i}}{\partial x^i} \\
&\qquad+
\sum_{I\in\mathscr T_{k+1}}
\sum_{\substack{i,j\in I\\ i\neq j}}
\operatorname{sgn}[i,\widehat I_i]
\operatorname{sgn}[j,\widehat I_j]
\,\omega_{\widehat I_j}\,
\nu_j
\frac{\partial\omega_{\widehat I_i}}{\partial x^i}.
\end{aligned}
\]
Hence
\[
\begin{aligned}
&\sum_{I\in\mathscr T_{k+1}}
\sum_{\substack{i,j\in I\\ i\neq j}}
\operatorname{sgn}[i,\widehat I_i]
\operatorname{sgn}[j,\widehat I_j]
\,\omega_{\widehat I_j}\,
\nu_j
\frac{\partial\omega_{\widehat I_i}}{\partial x^i} \\
&\qquad=
\langle\nu\wedge\omega,d\omega\rangle
-
\sum_{I\in\mathscr T_{k+1}}
\sum_{i\in I}
\omega_{\widehat I_i}\nu_i
\frac{\partial\omega_{\widehat I_i}}{\partial x^i}.
\end{aligned}
\]
Substituting this identity into the boundary integral above gives
\eqref{eq:boundary_identity_wedge}. The proof of
\eqref{eq:boundary_identity_contraction} is obtained in the same way starting
from the second pointwise identity of Lemma
\ref{lemma:pointwise_identity}.
\end{proof}


\begin{theorem}[Boundary identity in terms of principal curvatures]
\label{thm:boundary_principal_curvature_identity}
Let $\Omega\subset\mathbb{R}^n$ be a bounded open Lipschitz domain whose boundary is piecewise $C^2$. Suppose
\[
\partial\Omega=\bigcup_{s=1}^N\Gamma_s,
\]
where each $\Gamma_s$ is a relatively open $C^2$ hypersurface and
\[\sigma\!\left(\partial\Omega\setminus\bigcup_{s=1}^N\Gamma_s\right)=0.
\]
Let $E_1,\ldots,E_{n-1}$ be an orthonormal frame field of principal directions on each $\Gamma_s$, with associated principal curvatures $\gamma_1,\ldots,\gamma_{n-1}$.

Then, for every $\omega\in C_T^1(\Omega;\Lambda^k)$,
\begin{equation}
\label{eq:tangential_boundary_identity}
\|d\omega\|_{L^2}^2+\|\delta\omega\|_{L^2}^2-\|\nabla\omega\|_{L^2}^2
=
\sum_{s=1}^N
\int_{\Gamma_s}
\sum_{l=1}^{n-1}
\gamma_l\,|E_l\wedge\omega|^2.
\end{equation}

Similarly, for every $\omega\in C_N^1(\Omega;\Lambda^k)$,
\begin{equation}
\label{eq:normal_boundary_identity}
\|d\omega\|_{L^2}^2+\|\delta\omega\|_{L^2}^2-\|\nabla\omega\|_{L^2}^2
=
\sum_{s=1}^N
\int_{\Gamma_s}
\sum_{l=1}^{n-1}
\gamma_l\,|E_l\lrcorner\omega|^2.
\end{equation}
\end{theorem}

\begin{remark}
If $\partial\Omega$ is $C^2$, then the identities remain valid for
$\omega\in W_T^{1,2}(\Omega;\Lambda^k)$ and
$\omega\in W_N^{1,2}(\Omega;\Lambda^k)$ by a density argument.
\end{remark}
\begin{remark}
    For a brief overview on principal directions and curvatures, see the section on shape operator \ref{sec:shape_operator} in the appendix.
\end{remark}

\begin{proof}
We prove \eqref{eq:tangential_boundary_identity}. The proof of
\eqref{eq:normal_boundary_identity} is identical, starting from the
second identity of Lemma \ref{lemma:boundary_identity}.
\\\\ Since $\partial\Omega=\bigcup_{s=1}^N\Gamma_s$ up to a set of
surface measure zero, Lemma \ref{lemma:boundary_identity} gives
\begin{equation}
\label{eq:boundary_start}
\begin{aligned}
\|d\omega\|_{L^2}^2+\|\delta\omega\|_{L^2}^2-\|\nabla\omega\|_{L^2}^2
&=
\sum_{s=1}^N
\int_{\Gamma_s}
\langle \nu\wedge\omega,d\omega\rangle     \\
&\quad-
\sum_{s=1}^N
\int_{\Gamma_s}
A,
\end{aligned}
\end{equation}
where
\[
A=
\sum_{I\in\mathscr{T}_{k+1}}
\sum_{\substack{i,j\in I\\ i\neq j}}
\operatorname{sgn}[i,\widehat I_i]
\operatorname{sgn}[j,\widehat I_j]
\,
\omega_{\widehat I_j}
\nu_i
\frac{\partial\omega_{\widehat I_i}}{\partial x^j}.
\]

It therefore suffices to rewrite $A$ on each smooth boundary piece
$\Gamma_s$. We shall do this by decomposing the coordinate derivatives
into their normal and tangential components, and then expressing the
result in terms of the exterior product $\nu\wedge\omega$.
\\\\ \textbf{Step 1. Decomposition of the boundary derivatives.}

Fix one component $\Gamma_s$. Let
$E_1,\ldots,E_{n-1}$ be an orthonormal frame of principal directions
on $\Gamma_s$, and let $\nu$ denote the outward unit normal.

Since $\{E_1,\ldots,E_{n-1},\nu\}$ is an orthonormal basis of
$\mathbb{R}^n$ at every point of $\Gamma_s$, each coordinate vector
$e_j$ can be decomposed as
\[
e_j=\sum_{l=1}^{n-1}E_l^jE_l+\nu_j\nu,
\]
where $E_l=(E_l^1,\ldots,E_l^n)$.

Consequently, applying directional derivatives to the coefficient
$\omega_{\widehat I_i}$ gives
\begin{equation}
\label{eq:directional_decomposition}
\frac{\partial\omega_{\widehat I_i}}{\partial x^j}
=
\sum_{l=1}^{n-1}
E_l^j
\frac{\partial\omega_{\widehat I_i}}{\partial E_l}
+
\nu_j
\frac{\partial\omega_{\widehat I_i}}{\partial\nu}.
\end{equation}

Substituting \eqref{eq:directional_decomposition} into $A$, we
separate the contributions involving normal derivatives from those
involving tangential derivatives. Thus, writing $A=B+C$, we have
\[
\begin{aligned}
B={}&
\sum_{I\in\mathscr{T}_{k+1}}
\sum_{i,j\in I}
\operatorname{sgn}[i,\widehat I_i]
\operatorname{sgn}[j,\widehat I_j]
\,
\omega_{\widehat I_j}
\nu_i\nu_j
\frac{\partial\omega_{\widehat I_i}}{\partial\nu},
\end{aligned}
\]
and
\[
\begin{aligned}
C={}&
\sum_{I\in\mathscr{T}_{k+1}}
\sum_{i,j\in I}
\operatorname{sgn}[i,\widehat I_i]
\operatorname{sgn}[j,\widehat I_j]
\,
\omega_{\widehat I_j}
\nu_i
\sum_{l=1}^{n-1}
E_l^j
\frac{\partial\omega_{\widehat I_i}}{\partial E_l}.
\end{aligned}
\]
\textbf{Step 2. The normal part.}

We first rewrite $B$ in terms of the exterior product. Indeed, recall that
the coordinate expression for the exterior product gives
\[
(\nu\wedge\omega)_I
=
\sum_{j\in I}
\operatorname{sgn}[j,\widehat I_j]
\nu_j
\omega_{\widehat I_j}.
\]
Similarly, applying the same formula to the form
$\frac{\partial\omega}{\partial\nu}$ yields
\[
\left(\nu\wedge\frac{\partial\omega}{\partial\nu}\right)_I
=
\sum_{i\in I}
\operatorname{sgn}[i,\widehat I_i]
\nu_i
\frac{\partial\omega_{\widehat I_i}}{\partial\nu}.
\]
Comparing these expressions with the definition of $B$, we obtain
\[
\begin{aligned}
B
&=
\sum_{I\in\mathscr{T}_{k+1}}
(\nu\wedge\omega)_I
\left(\nu\wedge\frac{\partial\omega}{\partial\nu}\right)_I\\
&=
\left\langle
\nu\wedge\omega,
\nu\wedge\frac{\partial\omega}{\partial\nu}
\right\rangle.
\end{aligned}
\]
Hence,
\begin{equation}
\label{eq:B_identity}
B=
\left\langle
\nu\wedge\omega,
\nu\wedge\frac{\partial\omega}{\partial\nu}
\right\rangle.
\end{equation}
\textbf{Step 3. The tangential part.}

We now consider $C$. Fix $I\in\mathscr{T}_{k+1}$ and denote by
$C_I$ the contribution corresponding to this index. 
\\\\ To rewrite $C_I$, we differentiate the coefficient of
$\nu\wedge\omega$ along a principal direction $E_l$. By the product
rule,
\[
\begin{aligned}
\frac{\partial}{\partial E_l}
(\nu\wedge\omega)_I
={}&
\sum_{i\in I}
\operatorname{sgn}[i,\widehat I_i]
\nu_i
\frac{\partial\omega_{\widehat I_i}}{\partial E_l}
\\
&+
\sum_{i\in I}
\operatorname{sgn}[i,\widehat I_i]
\omega_{\widehat I_i}
\frac{\partial\nu_i}{\partial E_l}.
\end{aligned}
\]

Multiplying this identity by
$\operatorname{sgn}[j,\widehat I_j]\omega_{\widehat I_j}E_l^j$
and summing over $j\in I$, we obtain
\[
\begin{aligned}
C_I
&=
\sum_{l=1}^{n-1}
( E_l\wedge\omega )_I
\frac{\partial}{\partial E_l}
(\nu\wedge\omega)_I
\\
&\quad-
\sum_{i,j\in I}
\operatorname{sgn}[i,\widehat I_i]
\operatorname{sgn}[j,\widehat I_j]
\omega_{\widehat I_i}
\omega_{\widehat I_j}
\sum_{l=1}^{n-1}
E_l^j
\frac{\partial\nu_i}{\partial E_l}.
\end{aligned}
\]
Since $E_l$ is a principal direction, we have
\[
\frac{\partial\nu}{\partial E_l}
=
\gamma_lE_l,
\]
and hence
\[
\frac{\partial\nu_i}{\partial E_l}
=
\gamma_lE_l^i.
\]
Substituting this into the preceding expression for $C_I$ gives
\[
\begin{aligned}
C_I
&=
\sum_{l=1}^{n-1}
( E_l\wedge\omega )_I
\frac{\partial}{\partial E_l}
(\nu\wedge\omega)_I
\\
&\quad-
\sum_{l=1}^{n-1}
\gamma_l
\left(
\sum_{i\in I}
\operatorname{sgn}[i,\widehat I_i]
E_l^i
\omega_{\widehat I_i}
\right)
\left(
\sum_{j\in I}
\operatorname{sgn}[j,\widehat I_j]
E_l^j
\omega_{\widehat I_j}
\right).
\end{aligned}
\]
Observe that the expression inside each parenthesis is precisely
the $I$-th coefficient of $E_l\wedge\omega$. Therefore,
\[
\begin{aligned}
C_I
&=
\sum_{l=1}^{n-1}
( E_l\wedge\omega )_I
\frac{\partial}{\partial E_l}
(\nu\wedge\omega)_I
\\
&\quad-
\sum_{l=1}^{n-1}
\gamma_l
( E_l\wedge\omega )_I^2.
\end{aligned}
\]
Summing over all $I\in\mathscr{T}_{k+1}$, we arrive at
\begin{equation}
\label{eq:C_identity}
\begin{aligned}
C={}&
\sum_{l=1}^{n-1}
\left\langle
E_l\wedge\omega,
\frac{\partial}{\partial E_l}
(\nu\wedge\omega)
\right\rangle
\\
&-
\sum_{l=1}^{n-1}
\gamma_l|E_l\wedge\omega|^2.
\end{aligned}
\end{equation}
\textbf{Step 4. Combining our results.}

Combining \eqref{eq:B_identity} and \eqref{eq:C_identity}, we obtain
the following expression for $A$ on $\Gamma_s$:
\[
\begin{aligned}
A={}&
\left\langle
\nu\wedge\omega,
\nu\wedge\frac{\partial\omega}{\partial\nu}
\right\rangle
\\
&+
\sum_{l=1}^{n-1}
\left\langle
E_l\wedge\omega,
\frac{\partial}{\partial E_l}
(\nu\wedge\omega)
\right\rangle
\\
&-
\sum_{l=1}^{n-1}
\gamma_l|E_l\wedge\omega|^2.
\end{aligned}
\]

Substituting this expression into \eqref{eq:boundary_start} gives
\[
\begin{aligned}
\|d\omega\|_{L^2}^2+\|\delta\omega\|_{L^2}^2-\|\nabla\omega\|_{L^2}^2
&=
\sum_{s=1}^N
\int_{\Gamma_s}
\Bigg[
\langle\nu\wedge\omega,d\omega\rangle
\\
&\qquad-
\left\langle
\nu\wedge\omega,
\nu\wedge\frac{\partial\omega}{\partial\nu}
\right\rangle
\\
&\qquad-
\sum_{l=1}^{n-1}
\left\langle
E_l\wedge\omega,
\frac{\partial}{\partial E_l}
(\nu\wedge\omega)
\right\rangle
\\
&\qquad+
\sum_{l=1}^{n-1}
\gamma_l|E_l\wedge\omega|^2
\Bigg].
\end{aligned}
\]

If $\omega\in C_T^1(\Omega;\Lambda^k)$, then
\[
\nu\wedge\omega=0
\qquad\text{on }\Gamma_s
\]
for every $s$. Since the tangential derivative of a vanishing form
also vanishes, we have
\[
\frac{\partial}{\partial E_l}(\nu\wedge\omega)=0
\qquad\text{on }\Gamma_s.
\]
Consequently, the first three boundary terms vanish identically, and
we obtain
\[
\begin{aligned}
&\|d\omega\|_{L^2}^2+\|\delta\omega\|_{L^2}^2-\|\nabla\omega\|_{L^2}^2
\\
&\qquad=
\sum_{s=1}^N
\int_{\Gamma_s}
\sum_{l=1}^{n-1}
\gamma_l|E_l\wedge\omega|^2,
\end{aligned}
\]
which is \eqref{eq:tangential_boundary_identity}.
\\\\ The proof of \eqref{eq:normal_boundary_identity} is obtained in
exactly the same way, starting from
\eqref{eq:boundary_identity_contraction}. One replaces the exterior
product by the interior product throughout and uses the boundary
condition $\nu\lrcorner\omega=0$ on each $\Gamma_s$.
\end{proof}

\begin{proposition}[Trace estimate]
\label{prop:trace_estimate}
Let $\Omega\subset\mathbb{R}^n$ be a bounded open set of class $C^2$.
Then there exists a constant $C=C(\Omega)>0$ such that, for every
$0<\varepsilon<1$,
\[
    \int_{\partial\Omega}u^2
    \leq
    \varepsilon\int_\Omega |\nabla u|^2
    +
    \frac{C}{\varepsilon}\int_\Omega |u|^2
\]
for every $u\in W^{1,2}(\Omega)$.
\end{proposition}

\begin{proof}
By the density of $C^1(\overline{\Omega})$ in $W^{1,2}(\Omega)$ and
the continuous trace embedding
\[
    W^{1,2}(\Omega)\hookrightarrow L^2(\partial\Omega),
\]
it suffices to prove the estimate for
$u\in C^1(\overline{\Omega})$ (since boundary values viably get approximated by the continuous trace map).
\\\\ Since $\Omega$ is of class $C^2$, the exterior unit normal
\[
    \nu:\partial\Omega\to\mathbb{R}^n
\]
admits a $C^1$ extension to $\overline{\Omega}$, which we continue
to denote by $\nu$. In particular, both $|\nu|$ and
$|\operatorname{div}\nu|$ are bounded in $\Omega$. Thus, for some
constant $C>0$ depending only on $\Omega$,
\[
    |\nu|\leq C,
    \qquad
    |\operatorname{div}\nu|\leq C
    \qquad\text{in }\Omega.
\]

On $\partial\Omega$, $\nu$ is the unit exterior normal, and hence
$|\nu|^2=1$. Therefore,
\[
\begin{aligned}
    \int_{\partial\Omega}u^2
    &=
    \int_{\partial\Omega}u^2|\nu|^2\\
    &=
    \int_{\partial\Omega}\langle u^2\nu,\nu\rangle.
\end{aligned}
\]
Applying the divergence theorem to the vector field $u^2\nu$, we
obtain
\[
\begin{aligned}
    \int_{\partial\Omega}u^2
    &=
    \int_\Omega\operatorname{div}(u^2\nu)\\
    &=
    \int_\Omega u^2\operatorname{div}\nu
    +
    \int_\Omega\langle\nu,\nabla(u^2)\rangle.
\end{aligned}
\]
Since $\nabla(u^2)=2u\nabla u$, it follows that
\[
\begin{aligned}
    \int_{\partial\Omega}u^2
    &\leq
    \int_\Omega|\operatorname{div}\nu|\,u^2
    +
    2\int_\Omega|\nu|\,|u|\,|\nabla u|\\
    &\leq
    C\int_\Omega u^2
    +
    2C\int_\Omega|u|\,|\nabla u|.
\end{aligned}
\]
We now estimate the mixed term using Young's inequality. For every
$\varepsilon>0$,
\[
    2C|u||\nabla u|
    \leq
    \varepsilon|\nabla u|^2
    +
    \frac{C^2}{\varepsilon}u^2.
\]
Consequently,
\[
\begin{aligned}
    \int_{\partial\Omega}u^2
    &\leq
    \varepsilon\int_\Omega|\nabla u|^2
    +
    \left(
        C+\frac{C^2}{\varepsilon}
    \right)
    \int_\Omega u^2.
\end{aligned}
\]
Since $0<\varepsilon<1$, we have
\[
    C+\frac{C^2}{\varepsilon}
    \leq
    \frac{C'}{\varepsilon}
\]
for some constant $C'=C'(\Omega)>0$. Thus,
\[
    \boxed{
    \int_{\partial\Omega}u^2
    \leq
    \varepsilon\int_\Omega|\nabla u|^2
    +
    \frac{C'}{\varepsilon}\int_\Omega|u|^2
    }
\]
as required. Passing the inequality to the limit yields the result for $W^{1,2}(\Omega)$.
\end{proof}
Now we are in position to present the main result of the section:
\begin{theorem}[Gaffney inequality]
\label{thm:gaffney_inequality}
Let $\Omega\subset\mathbb{R}^n$ be a bounded open set of class $C^2$.
Then there exists a constant $C=C(\Omega)>0$ such that
\[
    \|\omega\|_{W^{1,2}(\Omega;\Lambda^k)}
    \leq
    C\left(
        \|d\omega\|_{L^2(\Omega)}
        +
        \|\delta\omega\|_{L^2(\Omega)}
        +
        \|\omega\|_{L^2(\Omega)}
    \right)
\]
for every
\[
    \omega\in
    W_T^{1,2}(\Omega;\Lambda^k)
    \cup
    W_N^{1,2}(\Omega;\Lambda^k).
\]
\end{theorem}

\begin{proof}
By the density results for the tangential and normal Sobolev spaces, it
suffices to establish the estimate for forms in
\[
    C_T^1(\overline{\Omega};\Lambda^k)
    \cup
    C_N^1(\overline{\Omega};\Lambda^k).
\]
Indeed, the estimate then extends to the corresponding Sobolev spaces by
approximation and the continuity of the trace operator.
\\\\ We first consider the tangential case, namely
\[
    \omega\in C_T^1(\overline{\Omega};\Lambda^k),
\]
so that
\[
    \nu\wedge\omega=0
    \qquad\text{on }\partial\Omega.
\]
The boundary identity involving the principal curvatures gives
\[
    \|d\omega\|_{L^2}^2
    +
    \|\delta\omega\|_{L^2}^2
    -
    \|\nabla\omega\|_{L^2}^2
    =
    \int_{\partial\Omega}
    \sum_{l=1}^{n-1}
    \gamma_l|E_l\wedge\omega|^2,
\]
where $E_1,\ldots,E_{n-1}$ are principal directions and
$\gamma_1,\ldots,\gamma_{n-1}$ are the corresponding principal
curvatures. Rearranging, we obtain
\begin{equation}
\label{eq:gaffney_start}
    \|\nabla\omega\|_{L^2}^2
    =
    \|d\omega\|_{L^2}^2
    +
    \|\delta\omega\|_{L^2}^2
    -
    \int_{\partial\Omega}
    \sum_{l=1}^{n-1}
    \gamma_l|E_l\wedge\omega|^2.
\end{equation}
Since $\partial\Omega$ is compact and of class $C^2$, its principal
curvatures are bounded. Hence there exists a constant
$C_1=C_1(\Omega)>0$ such that
\[
    |\gamma_l(x)|\leq C_1
\]
for every $x\in\partial\Omega$ and every $1\leq l\leq n-1$.
Moreover, since each $E_l$ is a unit vector, we have
\[
    |E_l\wedge\omega|
    \leq
    |E_l|\,|\omega|
    =
    |\omega|.
\]
It follows that
\[
\begin{aligned}
\left|
    \sum_{l=1}^{n-1}
    \gamma_l|E_l\wedge\omega|^2
\right|
&\leq
    \sum_{l=1}^{n-1}
    |\gamma_l|\,|E_l\wedge\omega|^2\\
&\leq
    C_1\sum_{l=1}^{n-1}|\omega|^2\\
&=
    C_1(n-1)|\omega|^2.
\end{aligned}
\]
Thus, for some constant $C_2=C_2(\Omega)>0$,
\[
    \left|
    \sum_{l=1}^{n-1}
    \gamma_l|E_l\wedge\omega|^2
    \right|
    \leq C_2|\omega|^2
    \qquad\text{on }\partial\Omega.
\]
Substituting this bound into \eqref{eq:gaffney_start}, we obtain
\begin{equation}
\label{eq:gaffney_boundary_control}
    \|\nabla\omega\|_{L^2}^2
    \leq
    \|d\omega\|_{L^2}^2
    +
    \|\delta\omega\|_{L^2}^2
    +
    C_2\int_{\partial\Omega}|\omega|^2.
\end{equation}
We now apply the trace estimate. For every $0<\varepsilon<1$,
\[
    \int_{\partial\Omega}|\omega|^2
    \leq
    \varepsilon\|\nabla\omega\|_{L^2}^2
    +
    \frac{C_3}{\varepsilon}
    \|\omega\|_{L^2}^2,
\]
where $C_3=C_3(\Omega)>0$. Combining this with
\eqref{eq:gaffney_boundary_control}, we obtain
\[
\begin{aligned}
    \|\nabla\omega\|_{L^2}^2
    \leq{}&
    \|d\omega\|_{L^2}^2
    +
    \|\delta\omega\|_{L^2}^2\\
    &+
    C_2\varepsilon\|\nabla\omega\|_{L^2}^2
    +
    \frac{C_2C_3}{\varepsilon}
    \|\omega\|_{L^2}^2.
\end{aligned}
\]
Choose $\varepsilon>0$ sufficiently small that
\[
    C_2\varepsilon\leq\frac12.
\]
Then
\[
\begin{aligned}
    \frac12\|\nabla\omega\|_{L^2}^2
    \leq{}&
    \|d\omega\|_{L^2}^2
    +
    \|\delta\omega\|_{L^2}^2\\
    &+
    \frac{C_2C_3}{\varepsilon}
    \|\omega\|_{L^2}^2.
\end{aligned}
\]
Consequently, there exists a constant $C_4=C_4(\Omega)>0$ such that
\begin{equation}
\label{eq:gradient_estimate}
    \|\nabla\omega\|_{L^2}^2
    \leq
    C_4
    \left(
        \|d\omega\|_{L^2}^2
        +
        \|\delta\omega\|_{L^2}^2
        +
        \|\omega\|_{L^2}^2
    \right).
\end{equation}
By the definition of the $W^{1,2}$ norm,
\[
    \|\omega\|_{W^{1,2}}^2
    =
    \|\omega\|_{L^2}^2
    +
    \|\nabla\omega\|_{L^2}^2.
\]
Combining this with \eqref{eq:gradient_estimate}, we obtain
\[
    \|\omega\|_{W^{1,2}}^2
    \leq
    C_5
    \left(
        \|d\omega\|_{L^2}^2
        +
        \|\delta\omega\|_{L^2}^2
        +
        \|\omega\|_{L^2}^2
    \right)
\]
for some constant $C_5=C_5(\Omega)>0$. Taking square roots and using
\[
    \sqrt{a^2+b^2+c^2}\leq a+b+c
    \qquad\text{for }a,b,c\geq0,
\]
we conclude that
\[
    \|\omega\|_{W^{1,2}}
    \leq
    C
    \left(
        \|d\omega\|_{L^2}
        +
        \|\delta\omega\|_{L^2}
        +
        \|\omega\|_{L^2}
    \right)
\]
for some constant $C=C(\Omega)>0$.
\\\\ We now consider the normal case,
\[
    \omega\in C_N^1(\overline{\Omega};\Lambda^k),
\]
for which
\[
    \nu\lrcorner\omega=0
    \qquad\text{on }\partial\Omega.
\]
The corresponding boundary identity gives
\[
    \|d\omega\|_{L^2}^2
    +
    \|\delta\omega\|_{L^2}^2
    -
    \|\nabla\omega\|_{L^2}^2
    =
    \int_{\partial\Omega}
    \sum_{l=1}^{n-1}
    \gamma_l|E_l\lrcorner\omega|^2.
\]
Since each $E_l$ is a unit vector, we have
\[
    |E_l\lrcorner\omega|
    \leq
    |E_l|\,|\omega|
    =
    |\omega|.
\]
The boundedness of the principal curvatures therefore yields the same
boundary estimate as in the tangential case. Applying the trace
inequality and absorbing the resulting gradient term, we obtain
\[
    \|\nabla\omega\|_{L^2}^2
    \leq
    C
    \left(
        \|d\omega\|_{L^2}^2
        +
        \|\delta\omega\|_{L^2}^2
        +
        \|\omega\|_{L^2}^2
    \right).
\]
Consequently, the same $W^{1,2}$ estimate holds for normal forms.
\\\\ Finally, by the density of $C_T^1(\overline{\Omega};\Lambda^k)$ in
$W_T^{1,2}(\Omega;\Lambda^k)$ and of
$C_N^1(\overline{\Omega};\Lambda^k)$ in
$W_N^{1,2}(\Omega;\Lambda^k)$, together with the continuity of the trace
operator, the estimate extends to all
\[
    \omega\in
    W_T^{1,2}(\Omega;\Lambda^k)
    \cup
    W_N^{1,2}(\Omega;\Lambda^k).
\]
\end{proof}

\section{The Hodge Laplacian}
We recall the definition of harmonic fields. Let
$0\leq k\leq n$ be an integer.

\begin{definition}
The set of harmonic fields is defined by
    \[
        \mathcal{H}(\Omega;\Lambda^k)
        =
        \left\{
            \omega\in W^{1,2}(\Omega;\Lambda^k)
            : d\omega=0\text{ and }\delta\omega=0
        \right\}.
    \]
    We shall write
    \[
        \mathcal{H}_T(\Omega;\Lambda^k)
        =
        \mathcal{H}(\Omega;\Lambda^k)
        \cap
        W_T^{1,2}(\Omega;\Lambda^k),
    \]
    and
    \[
        \mathcal{H}_N(\Omega;\Lambda^k)
        =
        \mathcal{H}(\Omega;\Lambda^k)
        \cap
        W_N^{1,2}(\Omega;\Lambda^k).
    \]
    That is,
    \[
        \mathcal{H}_T(\Omega;\Lambda^k)
        =
        \left\{
            \omega\in\mathcal{H}(\Omega;\Lambda^k)
            : \nu\wedge\omega=0
            \text{ on }\partial\Omega
        \right\},
    \]
    and
    \[
        \mathcal{H}_N(\Omega;\Lambda^k)
        =
        \left\{
            \omega\in\mathcal{H}(\Omega;\Lambda^k)
            : \nu\lrcorner\omega=0
            \text{ on }\partial\Omega
        \right\}.\]

\end{definition}

\begin{remark}
The space $\mathcal{H}(\Omega;\Lambda^k)$ can be equivalently
    defined as
    \[
        \mathcal{H}(\Omega;\Lambda^k)
        =
        \left\{
            \omega\in L_{\mathrm{loc}}^1(\Omega;\Lambda^k)
            : d\omega=0\text{ and }\delta\omega=0
        \right\},
    \]
    where the equations $d\omega=0$ and $\delta\omega=0$ are understood
    in the sense of distributions, namely,
    \[
        \int_\Omega\langle\omega,\delta\varphi\rangle=0,
        \qquad
        \text{for every }
        \varphi\in C_0^\infty(\Omega;\Lambda^{k+1}),
    \]
    and
    \[
        \int_\Omega\langle\omega,d\varphi\rangle=0,
        \qquad
        \text{for every }
        \varphi\in C_0^\infty(\Omega;\Lambda^{k-1}).
    \]

\end{remark}
\subsection{Existence and uniqueness of minimizers}
We draw primarily from the source \cite{CsatoDacorognaKneuss2012} for this section. cf. chapter 6. 
\\\\Some motivations are in order. 

We consider the problem of solving the Hodge Laplacian equation
\[
    \Delta_H\omega=-g
\]
subject to the tangential boundary condition
\[
    \nu\wedge\omega=0
    \qquad\text{on }\partial\Omega,
\]
where $\Delta_H=d\delta+\delta d$ is the Hodge Laplacian and
$g\in L^2(\Omega;\Lambda^k)$.
\\\\ The corresponding weak formulation is obtained by testing the equation
against forms $u\in W_T^{1,2}(\Omega;\Lambda^k)$. Formally, using
integration by parts, we obtain
\[
    \int_\Omega\langle d\omega,du\rangle
    +
    \int_\Omega\langle\delta\omega,\delta u\rangle
    =
    -\int_\Omega\langle g,u\rangle.
\]
This suggests considering the functional
\[
    D_g(\omega)
    =
    \int_\Omega
    \left(
        \frac12|d\omega|^2
        +
        \frac12|\delta\omega|^2
        +
        \langle g,\omega\rangle
    \right),
\]
whose Euler--Lagrange equation is precisely the weak formulation of
the Hodge Laplacian equation above.
\\\\ The idea is that, we use the Gaffney inequality, together with the orthogonality condition to
$\mathcal{H}_T(\Omega;\Lambda^k)$, to get coercivity of the quadratic
part of $D_g$ on $X$. Consequently, the direct method in the calculus
of variations (cf. \cite{Dacorogna2008}) can be applied to obtain a unique minimizer of $D_g$
over $X$. The minimizer satisfies the weak Euler--Lagrange equation
\[
    \int_\Omega\langle d\omega,du\rangle
    +
    \int_\Omega\langle\delta\omega,\delta u\rangle
    =
    -\int_\Omega\langle g,u\rangle
    \qquad\text{for every }u\in X.
\]

This provides the existence step in the $W^{1,2}$ setting for the
Hodge--Morrey decomposition.
\\\\ Let $0\leq k\leq n$. We now present the existence and uniqueness in
$W^{1,2}$ for the Hodge Laplacian equation. 
\\\\ Recall that
\[
    W_T^{1,2}(\Omega;\Lambda^k)
    =
    \left\{
        f\in W^{1,2}(\Omega;\Lambda^k)
        : \nu\wedge f=0\text{ on }\partial\Omega
    \right\}.
\]

\begin{theorem}[Existence of minimizer]
\label{thm:existence_minimizer}
Let $\Omega\subset\mathbb{R}^n$ be a bounded open set of class $C^2$.
Let $g\in L^2(\Omega;\Lambda^k)$, and define
\[
    D_g(\omega)
    =
    \int_\Omega
    \left(
        \frac12|d\omega|^2
        +
        \frac12|\delta\omega|^2
        +
        \langle g,\omega\rangle
    \right).
\]
Set
\[
    X
    =
    W_T^{1,2}(\Omega;\Lambda^k)
    \cap
    \mathcal{H}_T(\Omega;\Lambda^k)^\perp.
\]
Then there exists a unique $\tilde{\omega}\in X$ such that
\[
    D_g(\tilde{\omega})\leq D_g(\omega)
    \qquad\text{for every }\omega\in X,
\]
and $\tilde{\omega}$ satisfies the weak form of the Euler--Lagrange equation,
namely,
\begin{equation}
\label{eq:6.4}
    \int_\Omega\langle d\tilde\omega,du\rangle
    +
    \int_\Omega\langle\delta\tilde\omega,\delta u\rangle
    =
    -\int_\Omega\langle g,u\rangle
    \qquad\text{for every }u\in X.
\end{equation}

Moreover, if
\[
    g\in\mathcal{H}_T(\Omega;\Lambda^k)^\perp,
\]
then $\tilde\omega$ satisfies
\begin{equation}
\label{eq:6.5}
    \int_\Omega\langle d\tilde\omega,du\rangle
    +
    \int_\Omega\langle\delta\tilde\omega,\delta u\rangle
    =
    -\int_\Omega\langle g,u\rangle
    \qquad\text{for every }
    u\in W_T^{1,2}(\Omega;\Lambda^k).
\end{equation}

The same result holds upon replacing $T$ everywhere
by $N$.
\\\\ In particular, taking $g=0$ gives us direct existnce in $W^{1,2}(\Omega;\Lambda^k)$, for the \emph{Hodge Laplace's equation}. 
\end{theorem}
\begin{remark}\textbf{(On the Euler--Lagrange equation)}
\\\\ The Euler--Lagrange equation associated with the functional $D_g$ is
obtained by considering variations within the admissible space $X$.
Indeed, let $\omega\in X$ be a minimizer of $D_g$, and let $u\in X$
be arbitrary. Since $X$ is a vector space, the form
\[
    \omega_t=\omega+tu
\]
belongs to $X$ for every $t\in\mathbb{R}$. By the minimality of
$\omega$, the function
\[
    \Phi(t)=D_g(\omega+tu)
\]
attains a minimum at $t=0$. Hence $\Phi'(0)=0$.

Expanding the functional, we obtain
\[
\begin{aligned}
    \Phi(t)
    =\int_\Omega\bigg(
        &\frac12|d\omega+tdu|^2
        +\frac12|\delta\omega+t\delta u|^2\\
        &+\langle g,\omega+tu\rangle
    \bigg).
\end{aligned}
\]
Differentiating with respect to $t$ at $t=0$ gives
\[
\begin{aligned}
    0=\Phi'(0)
    ={}&
    \int_\Omega\langle d\omega,du\rangle
    +
    \int_\Omega\langle\delta\omega,\delta u\rangle\\
    &+
    \int_\Omega\langle g,u\rangle.
\end{aligned}
\]
Therefore, the minimizer $\omega$ satisfies
\[
    \int_\Omega\langle d\omega,du\rangle
    +
    \int_\Omega\langle\delta\omega,\delta u\rangle
    =
    -\int_\Omega\langle g,u\rangle
    \qquad\text{for every }u\in X.
\]
This is the weak form of the Euler--Lagrange equation associated
with $D_g$.
    
\end{remark}

\begin{proof}
We divide the proof into five steps.

\medskip
\noindent\textbf{Step 1. Coercivity of the quadratic part of $D_g$.}

We first claim that there exists a constant $C_1>0$ such that
\begin{equation}\label{eq:coercivity_X}
    \|\omega\|_{W^{1,2}}^2
    \leq C_1\int_\Omega
    \bigl(|d\omega|^2+|\delta\omega|^2\bigr)
\end{equation}
for every $\omega\in X$.

The point here is that the Gaffney inequality alone does not immediately
give this estimate, since it contains an additional $L^2$-term on the
right-hand side. We shall use the fact that $\omega$ is orthogonal to
$\mathcal H_T$ to eliminate this term.
\\\\ Suppose that \eqref{eq:coercivity_X} were false. Then, for every
$m\in\mathbb N$, there would exist $\omega_m\in X$ such that
\[
    \|\omega_m\|_{W^{1,2}}^2
    \geq
    m\int_\Omega
    \bigl(|d\omega_m|^2+|\delta\omega_m|^2\bigr).
\]
Since this inequality is homogeneous in $\omega_m$, we may normalize
$\omega_m$ so that
\[
    \|\omega_m\|_{L^2}=1
\]
for every $m$.
\\\\ By the Gaffney inequality, there exists a constant
$C_2>0$ such that
\[
    \|\omega_m\|_{W^{1,2}}^2
    \leq
    C_2\int_\Omega
    \bigl(|d\omega_m|^2+|\delta\omega_m|^2
    +|\omega_m|^2\bigr).
\]
Combining this with the preceding inequality and the normalization
$\|\omega_m\|_{L^2}=1$, we obtain
\[
    \|\omega_m\|_{W^{1,2}}^2
    \leq
    \frac{C_2}{m}\|\omega_m\|_{W^{1,2}}^2+C_2.
\]
For $m$ sufficiently large, we have $C_2/m\leq 1/2$, and hence
\[
    \frac12\|\omega_m\|_{W^{1,2}}^2\leq C_2.
\]
Consequently, the sequence $\{\omega_m\}$ is uniformly bounded in
$W^{1,2}(\Omega;\Lambda^k)$.
\\\\ Since $W^{1,2}(\Omega;\Lambda^k)$ is reflexive, we may extract a
subsequence (From \v{S}mulian--Eberlein), which we do not relabel, such that there exists
$u\in W^{1,2}(\Omega;\Lambda^k)$ satisfying
\[
    \omega_m\rightharpoonup u
    \qquad\text{in }W^{1,2}(\Omega;\Lambda^k).
\]
Moreover, by the Rellich--Kondrachov theorem, we have a compact inclusion $$W^{1,2}(\Omega)\hookrightarrow L^2(\Omega) $$
therefore, every bounded sequence has a strongly convergent subsequence. We can therefore (upon relabelling which we do not do) write
\[
    \omega_m\longrightarrow u
    \qquad\text{in }L^2(\Omega;\Lambda^k).
\]
In particular, since $\|\omega_m\|_{L^2}=1$ for every $m$, we obtain
\[
    \|u\|_{L^2}=1.
\]

Furthermore, since $d$ and $\delta$ are bounded linear operators from
$W^{1,2}(\Omega;\Lambda^k)$ into the appropriate $L^2$-spaces, we have
\[
    d\omega_m\rightharpoonup du
    \qquad\text{in }L^2(\Omega;\Lambda^{k+1}),
\]
and
\[
    \delta\omega_m\rightharpoonup\delta u
    \qquad\text{in }L^2(\Omega;\Lambda^{k-1}).
\]
Finally, since each $\omega_m\in\mathcal H_T^\perp$ and
$\mathcal H_T^\perp$ is closed in $L^2(\Omega;\Lambda^k)$, the strong
$L^2$-convergence implies that
\[
    u\in\mathcal H_T^\perp.
\]
We now show that
\[
    du=0\quad\text{in }\Omega,\qquad
    \delta u=0\quad\text{in }\Omega,
\]
and that
\[
    \nu\wedge u=0\quad\text{on }\partial\Omega.
\]
Indeed, from the defining inequality for $\omega_m$ and the uniform
bound on $\|\omega_m\|_{W^{1,2}}$, we have
\[
    \int_\Omega
    \bigl(|d\omega_m|^2+|\delta\omega_m|^2\bigr)
    \leq
    \frac1m\|\omega_m\|_{W^{1,2}}^2
    \longrightarrow 0.
\]
Therefore,
\[
    d\omega_m\longrightarrow 0
    \qquad\text{in }L^2(\Omega;\Lambda^{k+1}),
\]
and
\[
    \delta\omega_m\longrightarrow 0
    \qquad\text{in }L^2(\Omega;\Lambda^{k-1}).
\]
On the other hand, we already know that
$d\omega_m\rightharpoonup du$ and
$\delta\omega_m\rightharpoonup\delta u$ in $L^2$. It follows that
\[
    du=0\quad\text{in }\Omega,
    \qquad
    \delta u=0\quad\text{in }\Omega.
\]
It remains to verify the tangential boundary condition. Since
$\omega_m\in W_T^{1,2}(\Omega;\Lambda^k)$, we have
\[
    \nu\wedge\omega_m=0
    \qquad\text{on }\partial\Omega.
\]
The continuity of the trace embedding
\[
    W^{1,2}(\Omega;\Lambda^k)
    \hookrightarrow L^2(\partial\Omega;\Lambda^k)
\]
along with the weak convergence of $\omega_m$ $$\omega_m \rightharpoonup u \text{ in }W^{1,2}(\Omega,\Lambda^k) $$
gives us $$\mathrm{Tr}(\omega_m)\rightharpoonup \mathrm{Tr}(u) \text{ in }L^2(\partial\Omega,\Lambda^k). $$
Consequently,
\[
    \nu\wedge u=0
    \qquad\text{on }\partial\Omega.
\]
since $\omega_m$ satisfies the boundary condition for each $m$.
\\\\ Thus $u\in\mathcal H_T$. Since we also have
$u\in\mathcal H_T^\perp$, it follows that
\[
    u\in\mathcal H_T\cap\mathcal H_T^\perp=\{0\}.
\]
This contradicts $\|u\|_{L^2}=1$. The claim
\eqref{eq:coercivity_X} is therefore proved.
\\\\ \textbf{Step 2. A bound in terms of the functional $D_g$.}
\\\\ We next show that there exists a constant $C_4>0$ such that
\begin{equation}\label{eq:bound_Dg}
    \|\omega\|_{W^{1,2}}^2
    \leq
    4C_1D_g(\omega)+C_4
\end{equation}
for every $\omega\in X$.
\\\\ Recall that
\[
    D_g(\omega)
    =
    \frac12\int_\Omega
    \bigl(|d\omega|^2+|\delta\omega|^2\bigr)
    +\int_\Omega\langle g,\omega\rangle.
\]
By \eqref{eq:coercivity_X}, we have
\[
    \begin{aligned}
    \|\omega\|_{W^{1,2}}^2
    &\leq
    C_1\int_\Omega
    \bigl(|d\omega|^2+|\delta\omega|^2\bigr)\\
    &=2C_1D_g(\omega)
    -2C_1\int_\Omega\langle g,\omega\rangle.
    \end{aligned}
\]
Using the Cauchy--Schwarz inequality, we obtain
\[
    \|\omega\|_{W^{1,2}}^2
    \leq
    2C_1D_g(\omega)
    +2C_1\|g\|_{L^2}\|\omega\|_{L^2}.
\]
Since $\|\omega\|_{L^2}\leq\|\omega\|_{W^{1,2}}$, Young's inequality
gives
\[
    2C_1\|g\|_{L^2}\|\omega\|_{W^{1,2}}
    \leq
    \frac12\|\omega\|_{W^{1,2}}^2
    +2C_1^2\|g\|_{L^2}^2.
\]
Hence
\[
    \frac12\|\omega\|_{W^{1,2}}^2
    \leq
    2C_1D_g(\omega)+2C_1^2\|g\|_{L^2}^2,
\]
and therefore
\[
    \|\omega\|_{W^{1,2}}^2
    \leq
    4C_1D_g(\omega)+4C_1^2\|g\|_{L^2}^2.
\]
Taking
\[
    C_4=4C_1^2\|g\|_{L^2}^2,
\]
we obtain \eqref{eq:bound_Dg}.
\\\\ The important idea here is that the estimate above shows that the functional
$D_g$ controls the $W^{1,2}$-norm on $X$, up to an additive constant.
In particular, any sequence in $X$ along which $D_g$ remains bounded
must also remain bounded in $W^{1,2}$.
\\\\ \textbf{Step 3. Existence of a minimizer.}

Recall that a function $f$ from a metric space $X$ to $\bbR$ is \emph{lower semicontinuous} if for every sequence $x_n\to x \text{ in }X$, we have $$f(x)\leq \liminf_{n\to \infty} f(x_n) $$
The functional $D_g$ is weakly lower semicontinuous in
$W^{1,2}(\Omega;\Lambda^k)$. Indeed, the maps
\[
    \omega\longmapsto d\omega,
    \qquad
    \omega\longmapsto\delta\omega
\]
are bounded linear maps into $L^2$, and the squared $L^2$-norm is
weakly lower semicontinuous. Moreover, the linear term
\[
    \omega\longmapsto\int_\Omega\langle g,\omega\rangle
\]
is weakly continuous ($\omega_n\rightharpoonup \omega$ gives us $ \int_{\Omega}\langle g,\omega_n\rangle \to \int_{\Omega}\langle g,\omega\rangle$).
\\\\ Due to lower semicontinuity of $D_g$, we need only a \emph{weakly convergent} minimizing sequence whose limit still belongs in $X$, instead of requiring strong convergence. The resultant extracted weak subsequence shall be our \emph{minimizer} (This is a technique called \emph{Direct method} cf. \cite{Dacorogna2008} \emph{"Introduction to The Calculus of Variations"}). 
\\\\ Let $\{\omega_m\}\subset X$ be a minimizing sequence for $D_g$. By
\eqref{eq:bound_Dg}, the sequence $\{\omega_m\}$ is bounded in
$W^{1,2}(\Omega;\Lambda^k)$. Since this space is reflexive, we may
extract a subsequence, which we do not relabel, such that
\[
    \omega_m\rightharpoonup\omega
    \qquad\text{in }W^{1,2}(\Omega;\Lambda^k).
\]
By the Rellich--Kondrachov theorem, we also have
\[
    \omega_m\longrightarrow\omega
    \qquad\text{in }L^2(\Omega;\Lambda^k).
\]
We now verify that $\omega\in X$.
\\\\ First, since $\omega_m\in W_T^{1,2}(\Omega;\Lambda^k)$, we have
\[
    \nu\wedge\omega_m=0
    \qquad\text{on }\partial\Omega.
\]
The continuity of the trace operator, together with the weak
convergence in $W^{1,2}$, implies that
\[
    \nu\wedge\omega=0
    \qquad\text{on }\partial\Omega.
\]
Thus
\[
    \omega\in W_T^{1,2}(\Omega;\Lambda^k).
\]
Second, since $\omega_m\in\mathcal H_T^\perp$ for every $m$ and
$\omega_m\to\omega$ strongly in $L^2$, we obtain
\[
    \omega\in\mathcal H_T^\perp.
\]
Consequently,
\[
    \omega\in W_T^{1,2}(\Omega;\Lambda^k)
    \cap\mathcal H_T^\perp=X.
\]
We may therefore apply weak lower semicontinuity to conclude that
\[
    D_g(\omega)
    \leq\liminf_{m\to\infty}D_g(\omega_m).
\]
Since $\{\omega_m\}$ was a minimizing sequence, $\omega$ is a minimizer
of $D_g$ on $X$. This proves the existence assertion. Moreover, the Euler--Lagrange equation obtained from the minimization
problem holds for every $u\in X$, namely
\begin{equation}\label{eq:EL_X}
    \int_\Omega\langle d\omega,du\rangle
    +\int_\Omega\langle\delta\omega,\delta u\rangle
    =
    -\int_\Omega\langle g,u\rangle.
\end{equation}
\\\\ \textbf{Step 4. Uniqueness of the minimizer.}

Let $u,v\in X$ be two minimizers of $D_g$. Since the quadratic part
of $D_g$ is strictly convex in the pair $(d\omega,\delta\omega)$,
we obtain
\[
    du=dv
    \qquad\text{and}\qquad
    \delta u=\delta v.
\]
Equivalently,
\[
    d(u-v)=0,\qquad
    \delta(u-v)=0.
\]
Since $u-v\in X$, the coercivity estimate
\eqref{eq:coercivity_X} gives
\[
    \begin{aligned}
    \|u-v\|_{W^{1,2}}^2
    &\leq
    C_1\int_\Omega
    \bigl(|d(u-v)|^2+|\delta(u-v)|^2\bigr)\\
    &=0.
    \end{aligned}
\]
Hence $u=v$, and the minimizer is unique.

\medskip
\noindent\textbf{Step 5. Extension of the Euler--Lagrange equation to
all of $W_T^{1,2}(\Omega;\Lambda^k)$.}

It remains to establish \eqref{eq:6.5}. Let
\[
    u\in W_T^{1,2}(\Omega;\Lambda^k).
\]
Decompose $u$ into $w$ and $v$:
\[
    u=w+v,
    \qquad
    w\in\mathcal H_T,\quad
    v\in\mathcal H_T^\perp.
\]
Since $w\in\mathcal H_T$, we have
\[
    dw=0,\qquad\delta w=0.
\]
Moreover, by the assumption $g\in\mathcal H_T^\perp$,
\[
    \int_\Omega\langle g,w\rangle=0.
\]

We also have $v=u-w\in W_T^{1,2}(\Omega;\Lambda^k)$, since both
$u$ and $w$ satisfy the tangential boundary condition. In addition,
$v\in\mathcal H_T^\perp$. Thus
\[
    v\in W_T^{1,2}(\Omega;\Lambda^k)
    \cap\mathcal H_T^\perp=X.
\]
We may therefore apply \eqref{eq:EL_X} with $v$ in place of $u$,
obtaining
\[
    \int_\Omega\langle d\omega,dv\rangle
    +\int_\Omega\langle\delta\omega,\delta v\rangle
    =
    -\int_\Omega\langle g,v\rangle.
\]

Finally, since $u=v+w$ and $dw=\delta w=0$, we have
\[
    du=dv,\qquad\delta u=\delta v.
\]
Also,
\[
    \int_\Omega\langle g,u\rangle
    =
    \int_\Omega\langle g,v\rangle
    +\int_\Omega\langle g,w\rangle
    =
    \int_\Omega\langle g,v\rangle.
\]
Consequently,
\[
    \begin{aligned}
    &\int_\Omega\langle d\omega,du\rangle
    +\int_\Omega\langle\delta\omega,\delta u\rangle
    +\int_\Omega\langle g,u\rangle\\
    &\qquad=
    \int_\Omega\langle d\omega,dv\rangle
    +\int_\Omega\langle\delta\omega,\delta v\rangle
    +\int_\Omega\langle g,v\rangle\\
    &\qquad=0.
    \end{aligned}
\]
This is precisely the desired identity \eqref{eq:6.5}.
\end{proof}

We now turn to the main result of the present chapter, which gives the Hodge--Morrey decomposition of \(L^2\)-differential forms on bounded domains. We present the result without proof.

\begin{theorem}[Hodge--Morrey decomposition]
    Let \(\Omega\subset\mathbb{R}^n\) be a bounded open \(C^3\) set with exterior unit normal \(\nu\). Let \(0\leq k\leq n\) and \(f\in L^2(\Omega;\Lambda^k)\). Then the following hold.

    \begin{enumerate}
        \item There exist
        \[
            \alpha\in W_T^{1,2}(\Omega;\Lambda^{k-1}),
            \qquad
            \beta\in W_T^{1,2}(\Omega;\Lambda^{k+1}),
        \]
        \[
            h\in\mathcal{H}_T(\Omega;\Lambda^k),
            \qquad
            \omega\in W_T^{2,2}(\Omega;\Lambda^k),
        \]
        such that, in \(\Omega\),
        \[
            f=d\alpha+\delta\beta+h,
        \]
        where
        \[
            \alpha=\delta\omega,
            \qquad
            \beta=d\omega.
        \]

        \item There exist
        \[
            \alpha\in W_N^{1,2}(\Omega;\Lambda^{k-1}),
            \qquad
            \beta\in W_N^{1,2}(\Omega;\Lambda^{k+1}),
        \]
        \[
            h\in\mathcal{H}_N(\Omega;\Lambda^k),
            \qquad
            \omega\in W_N^{2,2}(\Omega;\Lambda^k),
        \]
        such that, in \(\Omega\),
        \[
            f=d\alpha+\delta\beta+h,
        \]
        where
        \[
            \alpha=\delta\omega,
            \qquad
            \beta=d\omega.
        \]

        \item There exist
        \[
            \alpha\in W_T^{1,2}(\Omega;\Lambda^{k-1}),
            \qquad
            \beta\in W_N^{1,2}(\Omega;\Lambda^{k+1}),
        \]
        \[
            h\in\mathcal{H}(\Omega;\Lambda^k),
        \]
        and
        \[
            \omega_1\in W_T^{2,2}(\Omega;\Lambda^k),
            \qquad
            \omega_2\in W_N^{2,2}(\Omega;\Lambda^k),
        \]
        such that, in \(\Omega\),
        \[
            f=d\alpha+\delta\beta+h,
        \]
        where
        \[
            \alpha=\delta\omega_1,
            \qquad
            \beta=d\omega_2.
        \]
    \end{enumerate}

    Moreover, in each of the three cases, the forms \(d\alpha\), \(\delta\beta\), and \(h\) are mutually orthogonal with respect to the \(L^2\)-inner product. Furthermore, in each case, there exists a positive constant \(C=C(\Omega)\) such that
    \[
        \|\omega\|_{W^{2,2}}+\|h\|_{L^2}
        \leq C\|f\|_{L^2},
    \]
    with the corresponding interpretation of \(\omega\) in cases \textup{(i)} and \textup{(ii)}, and of \(\omega_1,\omega_2\) in case \textup{(iii)}.
\end{theorem}
