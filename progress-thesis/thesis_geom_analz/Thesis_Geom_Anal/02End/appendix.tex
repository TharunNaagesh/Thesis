\thmp{\label{Taylor}(Taylor's Theorem)}{Let $f:U\to \bbR$ be $C^1$ on an open set $U\ni p$ that is star shaped about $p$. Then there exist $g_1,g_2,\cdots,g_n:U\to\bbR$ such that
$$f(x)=f(p)+\sum_{i=1}^n g_i(x)(x^i-p^i)$$
where $g_i(p)=\frac{\partial f}{\partial x^i}(p)$. Moreover, if $f$ is $C^{\infty}$, so are the $g_i$.}{Let $x\in U$. Since $U$ is star shaped, $\gamma(t)=p+t(x-p)\in U$ for $t\in[0,1]$. By the chain rule,
$$\frac{d}{dt}(f\circ \gamma)(t)=\sum_{i=1}^n \frac{\partial f}{\partial x^i}(\gamma(t))(x^i-p^i)$$
Integrating both sides from $0$ to $1$ gives
$$f(x)-f(p)=\sum_{i=1}^n (x^i-p^i)\int_0^1\frac{\partial f}{\partial x^i}(p+t(x-p))\,dt$$
Let $g_i(x):=\int_0^1 \frac{\partial f}{\partial x^i}(p+t(x-p))\,dt$, whence it is easy to verify that $g_i(p)=\frac{\partial f}{\partial x^i}(p)$. If $f$ is $C^{\infty}$, we can differentiate under the integral sign as many times as we like, so $g_i$ is $C^{\infty}$.
}

\begin{theorem} (\textbf{Inverse Function Theorem}) Let $f:\bbR^n\to \bbR^n$ be continuously differentiable in an open set containing $a$, and suppose $\det Df(a)\neq 0$.
Then there exist open sets $U,V\subset\bbR^n$ with $a\in U$, $f(a)\in V$ such that $f|_{U}:U\to V$ is a diffeomorphism.
\end{theorem}

\thmp{\label{thm:implicit}(Implicit Function Theorem)}{Let $f:\bbR^n \times \bbR^m \to \bbR^m$ be continuously differentiable in an open set containing $(a,b)$, with $f(a,b)=0$. Let $M$ be the $m \times m$ matrix $$M:=\big(D_{n+j}f^i(a,b)\big)_{1 \leq i,j\leq m}$$ and suppose $\det M\neq 0$. Then there exist open sets $A \subset \bbR^n$, $B \subset \bbR^m$ with $a\in A$, $b\in B$ such that for every $x\in A$ there is a unique $g(x)\in B$ with $f(x,g(x))=0$, and the $g$ so obtained is differentiable.}{Define $F:\bbR^n\times \bbR^m \to \bbR^n \times \bbR^m$ by $F(x,y):=(x,f(x,y))$. This is $C^1$, and
$$DF=\begin{pmatrix} I_n & 0 \\ D_xf & D_yf \end{pmatrix}$$
so $\det DF(a,b)=\det M\neq 0$. By the Inverse Function Theorem there is an open $A\times B\ni(a,b)$ such that $F:A\times B\to W$ is a diffeomorphism onto an open $W\ni(a,0)$ (shrink to a product neighbourhood if needed). Shrinking $A$ if necessary, we can also assume $(x,0)\in W$ for every $x\in A$.

Since $F$ preserves the first coordinate, $F^{-1}(u,v)=(u,k(u,v))$ for some differentiable $k$. Look at $f\circ F^{-1}(u,v)=\pi_2(u,v)=v$ (where $\pi_2\circ F\equiv f$), so $f(x,k(x,0))=0$ for $x\in A$. Our $g$ is $g(x):=k(x,0)$, which is differentiable as a composition of differentiable functions. For uniqueness, if $f(x,y)=0=f(x,y')$ with $y,y'\in B$ then $F(x,y)=(x,0)=F(x,y')$, and injectivity of $F$ on $A\times B$ gives $y=y'$.
}

\thmp{(A corollary of \ref{thm:implicit})}{Let $f:\bbR^n\to \bbR^p$ ($p\leq n$) be continuously differentiable in an open set containing $a\in \bbR^n$. If $f(a)=0$ and the $p\times n$ matrix $\big(D_jf^i(a)\big)$ has rank $p$, then there exist an open set $W\subset\bbR^n$ and a diffeomorphism $h:W\to h(W)\subset \bbR^n$ with $a\in h(W)$ such that
$$f\circ h(x_1,x_2,\cdots,x_n)=(x_{n-p+1},x_{n-p+2},\cdots,x_n)$$}{\textbf{Case 1:} Suppose the $p\times p$ matrix $Q:=\big(D_{n-p+j}f^i(a)\big)_{1\leq i,j\leq p}$ is invertible. Write points of $\bbR^n$ as $(u,v)\in\bbR^{n-p}\times\bbR^p$ and let $F(u,v):=(u,f(u,v))$. Exactly as before $\det DF(a)=\det Q\neq 0$, so there is an open neighbourhood $A$ of $a$ such that $F:A\to W$ is a diffeomorphism onto an open $W\subset\bbR^n$. Let $h:=F^{-1}$. Since $\pi_2\circ F\equiv f$, we have $$f\circ h=\pi_2\circ F\circ F^{-1}=\pi_2$$ on $W$, which is the map $(x_1,\cdots,x_n)\mapsto(x_{n-p+1},\cdots,x_n)$. Also $a=h(F(a))\in h(W)$.
\\\\ \textbf{Case 2:} Since the matrix has rank $p$, some $p$ columns $i_1,\cdots,i_p$ of it are linearly independent. Let $\operatorname{Swap}:\bbR^n\to\bbR^n$ be the linear map that permutes the coordinates so that these $p$ indices go to the last $p$ indices. It is invertible since it takes basis elements to basis elements. Then $f\circ\operatorname{Swap}$ is a function as in Case 1 (at the point $\operatorname{Swap}^{-1}(a)$), so there is a diffeomorphism $h$ with $f\circ\operatorname{Swap}\circ h=\pi_2$, and our desired function is $\operatorname{Swap}\circ h$.}

\defn{Locally finite collection}{Let $X$ be a topological space. We say a collection of subsets $\{A_{\alpha}\}_{\alpha\in \Lambda}$ is \emph{locally finite} if for every $x\in X$, there exists a neighbourhood $V_x$ such that $V_x \cap A_{\alpha}=\varnothing$ for all but finitely many $\alpha\in \Lambda$. }

\begin{theorem}\textbf{(Partition of unity in $\mathbf{\bbR^n}$)\label{thm:partition of unity}}
\cite{Spivak1965} Let $A\subset \bbR^n$ and $\mathscr O$ be an open cover of $A$. Then there exists a collection $\Phi$ of $C^{\infty}$ functions $\phi$ (defined on an open set containing $A$) such that: 
\begin{enumerate}
    \item For each $x\in A$, we have $0\leq\varphi(x)\leq1$.

    \item For each $x\in A$, there is an open set $V$ containing $x$
    such that all but finitely many $\varphi\in\Phi$ vanish on $V$.

    \item For each $x\in A$,
    \[
    \sum_{\varphi\in\Phi}\varphi(x)=1.
    \]

    \item For each $\varphi\in\Phi$, there is an open set
    $U\in\mathscr O$ such that $\varphi$ vanishes outside some
    closed set contained in $U$.
\end{enumerate}
\end{theorem}

\begin{theorem}\cite{Spivak1965}
    \label{thm:change_of_variable}
Let $A\subset \bbR^n$ be an open set and $g:A\to \bbR^n$ a $C^1$ function such that $\det (g')\neq 0$ for all $x\in A$. If $f:g(A)\to \bbR$ is integable, then $$\int_{g(A)} f=\int_A f\circ g |\det(g')| $$   
\end{theorem}

\section{Brief Look at Riemannian Manifolds}\label{app:riemannian}
\defn{Smooth Manifold}{\label{defn:smooth_manifold}
A smooth manifold of dimension $n$ is a topological space $M$ that is \emph{Hausdorff} and \emph{second countable}, together with a collection of pairs
\[
\{(U_\alpha,\varphi_\alpha)\}_{\alpha\in\Lambda},
\]
where the sets $U_\alpha\subseteq M$ are open and cover $M$, and each
\[
\varphi_\alpha:U_\alpha\longrightarrow \varphi_\alpha(U_\alpha)\subseteq\mathbb{R}^n
\]
is a homeomorphism onto an open subset of $\mathbb{R}^n$.

Furthermore, whenever $U_\alpha\cap U_\beta\neq\varnothing$, the transition map
\[
\varphi_\beta\circ\varphi_\alpha^{-1}:
\varphi_\alpha(U_\alpha\cap U_\beta)
\longrightarrow
\varphi_\beta(U_\alpha\cap U_\beta)
\]
is smooth.}

\defn{Riemannian Manifold}{\label{defn:Riemann_Manifold_and_metric}A Smooth $n-$dimensional manifold $(M,\mathscr A)$ is said to be a \emph{Riemannian Manifold} of dimension $n$ if for every $p\in M$ there exists an inner product $g_p:T_pM\times T_pM\to \bbR$ (smoothly depending on $p$) that is positive definite. Such a $g$ is called the \emph{metric}.}

We recall the fact (\ref{basis for tensors}) that the space of covariant--$2$ tensors has a basis given by $\{dx^i\otimes dx^j: i,j\leq n\}$. This then lets us write the metric in the following useful form: 
\begin{equation}
    g_p=\sum_{i,j=1}^n g_{ij}dx^i\otimes dx^j
\end{equation}
Where $g_{ij}=g(\frac{\partial}{\partial x^i},\frac{\partial}{\partial x^j})$.

We also recall the musical isomorphisms (\ref{section:flat_map}) here:

\subsubsection{The $\flat$ map}
Let $X$ be a vector field on $M$. Define the $\flat$ map that takes vector fields to $1-$forms as follows: $$X^{\flat}:=g(X,\cdot) $$
Let $X\equiv \frac{\partial}{\partial x^q}$, then \begin{equation}(\frac{\partial}{\partial x^q})^{\flat}=\sum_{i\leq n}\sum_{j\leq n} g_{ij} dx^i(\frac{\partial}{\partial x^q})dx^j =\sum_{j\leq n} g_{qj} dx^j\end{equation}
And for $\omega=\sum_{q=1}^n \omega^q \frac{\partial}{\partial x^q}$ we have \begin{equation}
    (\omega)^{\flat}=\sum_{j\leq n} (\sum_{q=1}^n g_{qj} \omega^q )dx^j
\end{equation}

\subsubsection{The $\sharp$ map}
Since $g$ was positive--definite, we have that the matrix $g_{ij}$ is invertible. Hence, we have the inverse of the $\flat$ map, called $\sharp$ map that takes $1-$forms to vector fields as follows: Let $\alpha=\sum_{i=1}^n \alpha^i dx^i$. Then \begin{equation}
    \alpha^{\sharp}=\sum_{j=1}^n(\sum_{q=1}^n g^{qj}\alpha^q)\frac{\partial}{\partial x^j}
\end{equation}
Where $g^{ij}=\bigg.([g_{ij}]^{-1}\bigg.)_{ij}$

\subsection{Natural Inner Product on Forms, Hodge Star and Interior Product}

Likewise let $\beta=w_1\wedge w_2\wedge \cdots \wedge w_k$. We define the inner product on $k-$forms as follows: \begin{equation}
    \langle  \alpha,\beta\rangle_g:=\det\Bigg{(}\begin{bmatrix}
    g(v_1^{\sharp},w_1^{\sharp}) & g(v_1^{\sharp},w_2^{\sharp}) & \cdots & g(v_1^{\sharp},w_k^{\sharp})\\  g(v_2^{\sharp},w_1^{\sharp}) & g(v_2^{\sharp},w_2^{\sharp}) & \cdots & g(v_2^{\sharp},w_k^{\sharp}) \\ \vdots & \vdots & \ddots & \vdots \\     g(v_k^{\sharp},w_1^{\sharp}) & g(v_k^{\sharp},w_2^{\sharp}) & \cdots & g(v_k^{\sharp},w_k^{\sharp}) 
    \end{bmatrix}\Bigg{)}
\end{equation}
And we extend it bilinearly to all $k-$ forms.

\subsubsection{Orientability}
\defn{Orientable Manifold}{A differentiable manifold $M$ is said to be orientable if there exists an atlas $\mathscr A$ such that for any two charts $(U,\varphi)$ and $(V,\psi)$ in $\mathscr A$, the Jacobian determinant of the transition maps $\psi \circ \varphi^{-1}$ is positive. If we denote our charts as $x=(x^1,x^2,\cdots,x^n)$ and $y=(y^1,y^2,\cdots,y^n)$ respectively, we can suggestively then write $\det(\frac{\partial x^i}{\partial y^j})>0$ }

Note that, if $x$
and $y$ are coordinate charts around a point $p$, $T_pM$ has bases $\{\frac{\partial }{\partial x^i}\}_{i\leq n}$ and $\{\frac{\partial}{\partial y^j}\}_{j\leq n}$. Chain rule lets us write $\frac{\partial}{\partial x^i}=\sum_{j=1}^n \frac{\partial y^j}{\partial x^i} \frac{\partial}{\partial y^j}$, and hence the transformation $A$ that takes the basis $\{\frac{\partial}{\partial x^i}\}_{i\leq n}$ to $\{\frac{\partial}{\partial y^j}\}_{j\leq n}$ is of positive determinant.

\subsubsection{Orthonormal frames and co-frames}
Choose a frame $E_1,E_2,\cdots,E_n$ (from Gram-Schmidt process) that forms an orthonormal basis with respect to $g$, i.e. $g(E_i,E_j)=\delta_{ij}$. Consider the co-frame $\xi^1,\xi^2,\cdots,\xi^n$ dual to $E_1,E_2,\cdots, E_n$ in $T_p^*M$. $(\xi^i)^{\sharp}=\sum_j \xi_i^j E_j=\sum_j g({\xi^i}^{\sharp},E_j)E_j$. Note that $g({\xi^i}^{\sharp},E_j)={{\xi^i}^{\sharp}}^{\flat}(E_j)=\xi^i(E_j)=\delta_{ij}$ which gives us that ${\xi^i}^{\sharp}=E_i$, and hence we have $$\langle(\xi^i)^{\sharp},(\xi^j)^\sharp\rangle_g=\delta_{ij} $$
Which tells us that the co-frame is also orthonormal with respect to the natural inner product $\langle \cdot ,\cdot \rangle_g$. We collect this result in the following lemma:

\begin{lemma}{\label{lemma:co_frame_orthogonal} If $E_1,E_2,\cdots,E_n$ is an orthonormal frame with respect to $g$, then the dual co-frame $\xi^1,\xi^2,\cdots,\xi^n$ is also orthonormal with respect to the natural inner product $\langle\cdot,\cdot \rangle_g  $.}\end{lemma}

These $\xi^i$ in \ref{lemma:co_frame_orthogonal} form an orthonormal basis for $T_p^*M$, and hence the collection $\{\xi^{i_1}\wedge \xi^{i_2}\wedge \cdots \wedge \xi^{i_k}:1\leq i_1<i_2<\cdots<i_k\leq n\}$ forms a basis for $\Lambda^k(T_pM)$. Let $\alpha=\sum_{I\in \mathscr T_k} \alpha^I \xi^I$ and $\beta=\sum_{I\in \mathscr T_k} \beta^I \xi^I$ be two $k-$forms written in this basis. Then we have \begin{equation}\label{eq:inner_product_of_k_forms_in_ON_basis}
    \langle \alpha,\beta\rangle_g=\sum_{I\in \mathscr T_k}\sum_{J\in \mathscr T_k}\langle\xi^I,\xi^J\rangle_g\alpha^I\beta^J=\sum_{I\in \mathscr T_k}\alpha^I\beta^I
\end{equation}

\subsubsection{The Volume Form}
Let $M$ be an orientable manifold and let $E_1,E_2,\cdots,E_k$ be a positively oriented orthonormal frame with respect to $g$. We define the volume form as the unique $n$ form such that: \begin{equation}\label{def:volume_form_for_riemannian_manifold}
    \mathrm{vol}_g(E_1,E_2,\cdots,E_n)=1
\end{equation}

Let $F_1,F_2,\cdots,F_n$ be another positively oriented (not necessarily orthonormal) frame. Suppose $\{E_i\}$ are related to $\{F_j\}$ via the orthogonal transformation $A$, i.e. $F_i=\sum_j A_i^j E_j$. Then we have \begin{equation}
    \mathrm{vol}_g(F_1,F_2,\cdots,F_n)=\det(A)
\end{equation} 
$\det(A)$ can be found as follows:
$$g(F_i,F_k)=g(\sum_i A_j^i E_i,\sum_m A_k^m E_m)=\sum_i\sum_m A^i_j A^m_k \delta_{im}=\sum_i A^i_j A^i_k =(A^TA)_{jk}$$
Therefore, $$\sqrt{\det(\{g(F_i,F_k)\}_{i,k\leq n})}=\det(A)$$
Let $x=(x^1,x^2,\cdots,x^n)$ a coordinate chart around $p\in M$, such that $\{\frac{\partial}{\partial x^i}\}_{i\leq n}$ is positively oriented. We then have that \begin{equation}
    \mathrm{vol}_g(\frac{\partial}{\partial x^1},\frac{\partial}{\partial x^2},\cdots,\frac{\partial}{\partial x^n})=\sqrt{ \det \Big{(} \bigg.{(} g(\frac{\partial}{\partial x^i},\frac{\partial}{\partial x^j})  \bigg.{)}\Big{)} }
\end{equation}

\end{equation}
\subsubsection{Hodge Star on Riemannian Manifolds}
\defn{Hodge Star for Riemannian Manifolds}{Let $\beta\in \Lambda^k$ be a $k-$form on an orientable Riemannian manifold $(M,g)$. We define the Hodge Star $\star \beta$ as the \emph{unique} $n-k$ form such that for every test $k$ form $\alpha$ the following holds: \begin{equation}
    \alpha \wedge \star \beta=\langle \alpha,\beta\rangle_g \mathrm{vol}_g
\end{equation} }

\lemp{}{Let $E_1,E_2,\cdots,E_n$ be a given positively oriented orthonormal frame. Let $\xi^1,\xi^2,\cdots,\xi^n$ be the dual co-frame. Then \begin{equation}
    \star (\xi^{i_1}\wedge \xi^{i_2}\wedge \cdots \wedge \xi^{i_k})=\operatorname{sign}(R(I,I^c)) \xi^{j_1}\wedge \xi^{j_2}\wedge \cdots \wedge \xi^{j_{n-k}}
\end{equation} Where $R(I,I^c)$ is the perumtation that rearranges $\{I,I^c\}$ to its natural order, and $j_1,j_2,\cdots,j_{n-k}$ are the canonically ordered incides of $I^c$.}{The proof follows from the following observations: 
\begin{enumerate}
    \item The volume form is $\xi^1\wedge \xi^2\wedge \cdots \wedge \xi^n$
    \item  $\operatorname{sign}(R(I,I^c))\xi^{i_1}\wedge \xi^{i_2}\cdots \wedge \xi^{i_k} \wedge \xi^{j_1}\wedge \cdots \wedge \xi^{j_{n-k}}=\mathrm{vol}_g$.
    \item $\langle \xi^{i_1}\wedge \cdots \wedge \xi^{i_k},\xi^{i_1}\wedge \cdots \wedge \xi^{i_k}\rangle_g=1$
\end{enumerate}}
\thmp{(Double Star Identity)}{\label{thm:double_star}Let $f$ be a $k-$form. Then \begin{equation}
    \star \star f=(-1)^{k(n-k)f}
\end{equation}}{Because the star is linear, it is enough to prove the identity on basis elements. Let $E_1,\cdots,E_n$ be a positively oriented orthonormal frame and $\xi^1,\cdots,\xi^n$ be its dual co-frame. $\star \star (\xi^{i_1}\wedge \xi^{i_2}\cdots \wedge \xi^{i_k})=\mathrm{sign}(R(I,I^c))\star (\xi^{I^c})=\mathrm{sign}(R(I,I^c))\mathrm{sign}(R(I^c,I))\xi^I$. The proof is done once we see that the product $\mathrm{sign}(R(I,I^c))\mathrm{sign}(R(I^c,I))$ is the sign of the permutation that takes swaps the first $k$ terms in $[I,I^c]$ behind the last $n-k$ terms, which amounts to a sign of $(-1)^{k(n-k)}$. }
\thmp{(Hodge Star is an isometry)}{The Hodge Star $\star:\Lambda^k\to \Lambda^{n-k}$ is an isometric isomorphism between Normed Linear Spaces.}{Note that the Hodge Star is linear. Since the dimensions of $\Lambda^k$ and $\Lambda^{n-k}$ are the same, it is enough to show that $\star$ is an isometry, i.e. $\langle \star \alpha,\star \beta \rangle_{g,n-k}=\langle \alpha,\beta \rangle_{g,k}$. 
Let $\xi^1,\xi^2,\cdots,\xi^n$ be the co-frame of the orthonormal frame $E_1,E_2,\cdots,E_n$. We saw in \ref{lemma:co_frame_orthogonal} that these too, are orthonormal with respect to the natural inner product. Let $\alpha=\sum_{I\in \mathscr T_k} \alpha^I \xi^I$ and $\beta=\sum_{I\in \mathscr T_k} \beta^I \xi^I$. From \ref{eq:inner_product_of_k_forms_in_ON_basis} we have $\langle \alpha,\beta\rangle_g=\sum_{I\in \mathscr T_k} \alpha^I\beta^I$. $\langle \star \alpha,\star \beta\rangle= \star \alpha \wedge \star \star \beta=(-1)^{k(n-k)}\star \star \beta \wedge \star \alpha =(-1)^{k(n-k)}\langle \alpha,\star\star \beta\rangle_g=\langle \alpha,\beta\rangle_g$ from the double star identity \ref{thm:double_star}.

}

\end{equation}}

\subsection{The shape operator and principal curvatures}\label{sec:shape_operator}
This section develops the shape operator and principal curvatures and directions, and will be used in the proof of the Gaffney inequality. Throughout, $\Omega\subset\bbR^n$ is a bounded open $C^2$ set and $\nu$ is the exterior unit normal. Since $\Omega$ is $C^2$, $\nu\in C^1(\partial\Omega;\bbR^n)$, and this is precisely the regularity needed to differentiate $\nu$ along the boundary. (For a merely $C^1$ set, $\nu$ is only continuous and none of what follows makes sense pointwise.)

\defn{Shape operator (Weingarten map)}{For $x\in\partial\Omega$ and $X\in T_x(\partial\Omega)$, define
\begin{equation}
    W_x(X):=\nabla_X\nu=\frac{d}{dt}\Big|_{t=0}\nu(\gamma(t)),
\end{equation}
where $\gamma$ is any $C^1$ curve in $\partial\Omega$ with $\gamma(0)=x$ and $\gamma'(0)=X$. (Only values of $\nu$ on $\partial\Omega$ are used, so no extension of $\nu$ is needed.)}
Note that $W_x$ maps $T_x(\partial\Omega)$ to itself. Indeed, since $\langle\nu,\nu\rangle=1$ on $\partial\Omega$, differentiating along $X$ gives $2\langle\nabla_X\nu,\nu\rangle=0$, i.e.\ $W_x(X)\perp\nu(x)$, so $W_x(X)\in T_x(\partial\Omega)$. 

\defn{Second fundamental form}{$\mathrm{II}_x(X,Y):=\langle W_x(X),Y\rangle$ for $X,Y\in T_x(\partial\Omega)$.}

\thmp{}{$\mathrm{II}_x$ is a symmetric bilinear form on $T_x(\partial\Omega)$. Equivalently, $W_x$ is self-adjoint.}{Bilinearity is clear. For symmetry, let $\varphi(y')$ be a $C^2$ local parametrization of $\partial\Omega$ (e.g.\ $\varphi(y')=\phi(y',0)$ for a boundary chart) and write $\nu(y')=\nu(\varphi(y'))$. Differentiating $\langle\nu,\partial_j\varphi\rangle=0$ in $y^i$ gives
\[
\langle\partial_i\nu,\partial_j\varphi\rangle=-\langle\nu,\partial_{ij}\varphi\rangle.
\]
The right side is symmetric in $(i,j)$ since $\varphi\in C^2$. Since $\partial_i\nu=W(\partial_i\varphi)$, this reads $\mathrm{II}(\partial_i\varphi,\partial_j\varphi)=\mathrm{II}(\partial_j\varphi,\partial_i\varphi)$. The $\partial_i\varphi$ form a basis of $T_x(\partial\Omega)$ and $\mathrm{II}$ is bilinear, so symmetry holds on every pair of tangent vectors, in particular for any orthonormal frame.}

The identity in the proof also shows that $\mathrm{II}(X,Y)=-\langle\nu,D_XY\rangle$ for tangent fields $X,Y$, so $\mathrm{II}$ measures how fast the boundary bends away from its tangent plane in the direction of $\nu$.

\defn{Principal curvatures and directions}{Since $W_x$ is a self-adjoint endomorphism of the $(n-1)$-dimensional inner product space $T_x(\partial\Omega)$, the spectral theorem gives an orthonormal basis $e_1,\dots,e_{n-1}$ of $T_x(\partial\Omega)$ and real numbers $\kappa_1\leq\cdots\leq\kappa_{n-1}$ with
\begin{equation}
    \nabla_{e_j}\nu=W_x(e_j)=\kappa_j e_j .
\end{equation}
The $\kappa_j$ are the \emph{principal curvatures} and the $e_j$ the \emph{principal directions} at $x$. The \emph{mean curvature} is $H=\operatorname{tr}W=\sum_j\kappa_j$ and the \emph{Gauss curvature} is $K=\det W=\prod_j\kappa_j$.}

\section{Pullback of Forms}\label{app:pullback}
Let $F:U\to V$, where $U\subseteq \bbR^n$ and $V\subseteq \bbR^m$. If $F$ is smooth, the Frechet derivative at $p$ is defined as the unique linear map $DF_p:\bbR^n\to\bbR^m$ satisfying $$\lim_{|h|\to 0} \frac{||F(p+h)-F(p)-DF_p(h)||_m}{||h||_n}=0 $$
We have $DF_p:T_pU\to T_{f(p)}V$ where $DF_p(v)\in T_{f(p)}V$ such that $$DF_p(v) (g)=v(g \circ F)$$ Where $g\in C^{\infty}_{f(p)}(V)$. 
In coordinates we have \begin{equation}\label{DF_p of basis}
DF_p(\frac{\partial}{\partial x^i}|_p)=\sum_{j=1}^m DF_p(\frac{\partial}{\partial x^i}|_p)(y^j) \frac{\partial}{\partial y^j}|_{F(p)}     
\end{equation} 
which gives $$DF_p(v)=DF_p(\sum_{i=1}^n v^i \frac{\partial}{\partial x^i}|_p)=\sum_{i=1}^n \sum_{j=1}^m DF_p(\frac{\partial}{\partial x^i}|_p)(y^j)v^i \frac{\partial}{\partial y^j}|_{F(p)}= \sum_{i=1}^n\sum_{j=1}^m \frac{\partial}{\partial x^i}|_p (y^j \circ F) v^i \frac{\partial}{\partial y^j}|_{F(p)} $$
If we define $F^j=y^j \circ F$ (The $j-$th coordinate of $F$) we get $$DF_p(v)\equiv \sum_{i=1}^n\sum_{j=1}^m \frac{\partial F^j}{\partial x^i}|_p v^i \frac{\partial}{\partial y^j}|_{F(p)} $$
Where $\{\frac{\partial F^j}{\partial x^i}|_p: 1\leq i\leq n, 1\leq j\leq m\}$ is the Jacobian Matrix.

\defn{Pullback of a $k-$form}{Let $\omega:V\to \amalg_{q\in V}\Lambda^{k}(T_q(V)) $ be a $k-$form on $V\subset \bbR^m$. We can get a $k-$form on $U\subseteq \bbR^n$ via $F$ as $F^*\omega_p \in \Lambda^k(T_p(U)) $ given by $$F^*\omega_p (v_1,v_2,\cdots,v_k)=\omega_{f(p)}\big{(}DF_p(v_1),DF_p(v_2),\cdots,DF_p(v_k)\big{)} $$}

\subsection{Pullback of a $1$-form}

Let $\omega=dy^q$ be a basis $1$-form on $V\subset\bbR^m$. By definition,
\[
F^*(dy^q)_p(v)
=
dy^q_{F(p)}\bigl(DF_p(v)\bigr).
\]
Writing
\[
v=\sum_{i=1}^n v^i
\frac{\partial}{\partial x^i}\Big|_p,
\]
we have
\[
DF_p(v)
=
\sum_{i=1}^n\sum_{j=1}^m
\frac{\partial F^j}{\partial x^i}\Big|_p
v^i
\frac{\partial}{\partial y^j}\Big|_{F(p)}.
\]
Therefore,
\[
\begin{aligned}
F^*(dy^q)_p(v)
&=
dy^q_{F(p)}
\left(
\sum_{i=1}^n\sum_{j=1}^m
\frac{\partial F^j}{\partial x^i}\Big|_p
v^i
\frac{\partial}{\partial y^j}\Big|_{F(p)}
\right)
\\
&=
\sum_{i=1}^n\sum_{j=1}^m
\frac{\partial F^j}{\partial x^i}\Big|_p
v^i
\delta_{jq}
\\
&=
\sum_{i=1}^n
\frac{\partial F^q}{\partial x^i}\Big|_p
v^i
\\
&=
\sum_{i=1}^n
\frac{\partial F^q}{\partial x^i}\Big|_p
dx^i_p(v).
\end{aligned}
\]
Hence,
\begin{equation}
\label{pullback_of_1_form}
F^*(dy^q)_p
=
\sum_{i=1}^n
\frac{\partial F^q}{\partial x^i}\Big|_p
dx^i_p.
\end{equation}

Note that
\[
F^*(y^q)=y^q\circ F=F^q.
\]
Taking the exterior derivative, we obtain
\[
d(F^*(y^q))
=
dF^q
=
\sum_{i=1}^n
\frac{\partial F^q}{\partial x^i}\Big|_p
dx^i_p.
\]
Comparing this with \eqref{pullback_of_1_form}, we see that, for the
basis $1$-forms,
\[
d(F^*(y^q))=F^*(dy^q).
\]

Now let $\omega$ be a general $1$-form on $V$, given by
\[
\omega
=
\sum_{j=1}^m \omega^j\,dy^j.
\]
Then
\[
F^*(\omega)_p(v)
:=
\omega_{F(p)}\bigl(DF_p(v)\bigr)
=
\sum_{j=1}^m
\omega^j(F(p))
dy^j_{F(p)}\bigl(DF_p(v)\bigr).
\]
Using \eqref{pullback_of_1_form},
\[
\begin{aligned}
F^*(\omega)_p(v)
&=
\sum_{j=1}^m
\omega^j(F(p))
\sum_{i=1}^n
\frac{\partial F^j}{\partial x^i}\Big|_p
dx^i_p(v)
\\
&=
\sum_{j=1}^m
\omega^j(F(p))\,dF^j_p(v).
\end{aligned}
\]
Hence,
\[
F^*(\omega)_p
=
\sum_{i=1}^n\sum_{j=1}^m
\frac{\partial F^j}{\partial x^i}\Big|_p
\omega^j(F(p))\,dx^i_p
=
\sum_{j=1}^m
\omega^j(F(p))\,dF^j_p.
\]

Yet another suggestive way to write this is
\[
F^*(\omega)_p
=
F^*\left(\sum_{j=1}^m\omega^j\,dy^j\right)_p
=
\sum_{j=1}^m
(\omega^j\circ F)(p)\,dF^j_p.
\]

\subsection{Pullback of a $k$-form}

Let $\omega=dy^I=dy^{i_1}\wedge dy^{i_2}\cdots\wedge dy^{i_k}\in\Omega(V,\Lambda^k)$ be a $k$-form on $V\subset\bbR^m$. Then
\begin{equation}\label{pullback_of_wedge_form}
F^*(dy^I)_p(v_1,\ldots,v_k)=\sum_{1\leq j_1,\ldots,j_k\leq n}(v_1^{j_1}\cdots v_k^{j_k})F^*(dy^I)_p\left(\frac{\partial}{\partial x^{j_1}}\Big|_p,\ldots,\frac{\partial}{\partial x^{j_k}}\Big|_p\right).
\end{equation}
By definition of the pullback,
\[
F^*(dy^I)_p\left(\frac{\partial}{\partial x^{j_1}}\Big|_p,\ldots,\frac{\partial}{\partial x^{j_k}}\Big|_p\right)
=dy^I_{F(p)}\left(DF_p\left(\frac{\partial}{\partial x^{j_1}}\Big|_p\right),\ldots,DF_p\left(\frac{\partial}{\partial x^{j_k}}\Big|_p\right)\right).
\]
From \ref{DF_p of basis}, we have
\[
DF_p\left(\frac{\partial}{\partial x^{j_r}}\Big|_p\right)=\sum_{q_r=1}^m\frac{\partial F^{q_r}}{\partial x^{j_r}}\Big|_p\frac{\partial}{\partial y^{q_r}}\Big|_{F(p)}.
\]
Hence
\[
F^*(dy^I)_p\left(\frac{\partial}{\partial x^{j_1}}\Big|_p,\ldots,\frac{\partial}{\partial x^{j_k}}\Big|_p\right)
=\sum_{1\leq q_1,\ldots,q_k\leq m}\left(\prod_{r=1}^k\frac{\partial F^{q_r}}{\partial x^{j_r}}\Big|_p\right)dy^I_{F(p)}\left(\frac{\partial}{\partial y^{q_1}}\Big|_{F(p)},\ldots,\frac{\partial}{\partial y^{q_k}}\Big|_{F(p)}\right).
\]
Since $dy^I=dy^{i_1}\wedge\cdots\wedge dy^{i_k}$,
\[
dy^I\left(\frac{\partial}{\partial y^{q_1}},\ldots,\frac{\partial}{\partial y^{q_k}}\right)=\det\Big((\delta_{i_s}^{q_t})_{1\leq s,t\leq k}\Big).
\]
Combining this with \ref{pullback_of_wedge_form} gives
\begin{equation}\label{pullback_of_basis_k_form_expansion}
F^*(dy^I)_p(v_1,\ldots,v_k)=\sum_{\substack{1\leq j_1,\ldots,j_k\leq n\\1\leq q_1,\ldots,q_k\leq m}}(v_1^{j_1}\cdots v_k^{j_k})\left(\prod_{r=1}^k\frac{\partial F^{q_r}}{\partial x^{j_r}}\Big|_p\right)\det\Big((\delta_{i_s}^{q_t})_{1\leq s,t\leq k}\Big).
\end{equation}
The determinant is nonzero precisely when $(q_1,\ldots,q_k)=(i_{\sigma(1)},\ldots,i_{\sigma(k)})$ for some $\sigma\in\operatorname{Sym}(k)$. Therefore,
\[
F^*(dy^I)_p(v_1,\ldots,v_k)=\sum_{1\leq j_1,\ldots,j_k\leq n}(v_1^{j_1}\cdots v_k^{j_k})\sum_{\sigma\in\operatorname{Sym}(k)}\operatorname{sign}(\sigma)\prod_{r=1}^k\frac{\partial F^{i_{\sigma(r)}}}{\partial x^{j_r}}\Big|_p.
\]
Since $dx^{j_r}_p(v_r)=v_r^{j_r}$, this gives
\[
F^*(dy^I)_p(v_1,\ldots,v_k)=\sum_{1\leq j_1,\ldots,j_k\leq n}\left(\prod_{r=1}^kdx^{j_r}_p(v_r)\right)\det\Bigg(\left(\frac{\partial F^{i_r}}{\partial x^{j_s}}\Big|_p\right)_{1\leq r,s\leq k}\Bigg).
\]
Collecting the terms corresponding to $1\leq j_1<\cdots<j_k\leq n$, we obtain
\[
\begin{aligned}
F^*(dy^I)_p(v_1,\ldots,v_k)
&=\sum_{1\leq j_1<\cdots<j_k\leq n}\det\Bigg(\left(\frac{\partial F^{i_r}}{\partial x^{j_s}}\Big|_p\right)_{1\leq r,s\leq k}\Bigg)\\
&\qquad\cdot\sum_{\sigma\in\operatorname{Sym}(k)}\operatorname{sign}(\sigma)\prod_{r=1}^kdx^{j_{\sigma(r)}}_p(v_r).
\end{aligned}
\]
The second sum is precisely $(dx^{j_1}\wedge\cdots\wedge dx^{j_k})_p(v_1,\ldots,v_k)$. Hence
\begin{equation}\label{pullback_of_basis_k_form_determinant_form}
F^*(dy^I)_p=\sum_{1\leq j_1<\cdots<j_k\leq n}\det\Bigg(\left(\frac{\partial F^{i_r}}{\partial x^{j_s}}\Big|_p\right)_{1\leq r,s\leq k}\Bigg)dx^{j_1}_p\wedge dx^{j_2}_p\cdots\wedge dx^{j_k}_p.
\end{equation}

We now look at
\[
d(F^*y^{i_1})_p\wedge d(F^*y^{i_2})_p\wedge\cdots\wedge d(F^*y^{i_k})_p=dF^{i_1}_p\wedge dF^{i_2}_p\wedge\cdots\wedge dF^{i_k}_p.
\]
Using
\[
dF^{i_r}_p=\sum_{j_r=1}^n\frac{\partial F^{i_r}}{\partial x^{j_r}}\Big|_pdx^{j_r}_p,
\]
we get
\[
dF^{i_1}_p\wedge\cdots\wedge dF^{i_k}_p=\sum_{1\leq j_1,\ldots,j_k\leq n}\left(\prod_{r=1}^k\frac{\partial F^{i_r}}{\partial x^{j_r}}\Big|_p\right)dx^{j_1}_p\wedge\cdots\wedge dx^{j_k}_p.
\]
Collecting terms with $j_1<\cdots<j_k$ gives
\[
dF^{i_1}_p\wedge\cdots\wedge dF^{i_k}_p=\sum_{1\leq j_1<\cdots<j_k\leq n}\det\Bigg(\left(\frac{\partial F^{i_r}}{\partial x^{j_s}}\Big|_p\right)_{1\leq r,s\leq k}\Bigg)dx^{j_1}_p\wedge\cdots\wedge dx^{j_k}_p.
\]
This agrees with \ref{pullback_of_basis_k_form_determinant_form}. Hence,
\begin{equation}
F^*(dy^{i_1}\wedge dy^{i_2}\wedge\cdots\wedge dy^{i_k})=d(F^*y^{i_1})\wedge d(F^*y^{i_2})\wedge\cdots\wedge d(F^*y^{i_k}).
\end{equation}

\begin{corollary}\label{Change of variables for top forms}
If $F:U_{\subset\bbR^n}\to V_{\subset\bbR^n}$ is a smooth diffeomorphism, with coordinates $x^1,\ldots,x^n$ on $U$ and $y^1,\ldots,y^n$ on $V$, then
\[
F^*(dy^1\wedge dy^2\wedge\cdots\wedge dy^n)_p=\det\Bigg(\left(\frac{\partial F^j}{\partial x^i}\Big|_p\right)_{1\leq i,j\leq n}\Bigg)dx^1_p\wedge dx^2_p\wedge\cdots\wedge dx^n_p.
\]
\end{corollary}

\begin{remark}
The proof of \ref{Change of variables for top forms} is simply an application of \ref{pullback_of_basis_k_form_determinant_form}. This is the \textbf{change of variables formula} for \emph{top forms}.
\end{remark}

If $\omega=\sum_{I\in\mathscr T_k}\omega^I dy^I$ is a general $k$-form on $V\subseteq\bbR^m$, then
\begin{equation}\label{pullback_of_arbitrary_k_form}
F^*(\omega)_p=\sum_{I\in\mathscr T_k}\omega^I(F(p))\,dF^I_p,
\end{equation}
where $dF^I_p=dF^{i_1}_p\wedge dF^{i_2}_p\wedge\cdots\wedge dF^{i_k}_p$.

\lemp{($F^*$ commutes with $d$)}
{\label{pullback_commutes_with_derivative}
Let $F:U_{\subset\bbR^n}\to V_{\subset\bbR^m}$ be smooth, and let $\omega$ be a $k$-form on $V$. We then have
\[
F^*(d\omega)=d(F^*\omega).
\]}
{If $\omega=\sum_{I\in\mathscr T_k}\omega^I dy^I$, then from \ref{pullback_of_arbitrary_k_form},
\[
F^*(\omega)=\sum_I(\omega^I\circ F)dF^I.
\]
Applying the exterior derivative and using the antiderivation property,
\[
d(F^*\omega)=\sum_I d(\omega^I\circ F)\wedge dF^I+\sum_I(\omega^I\circ F)d(dF^I).
\]
The second term vanishes because $d^2=0$. Hence
\[
d(F^*\omega)=\sum_I d(\omega^I\circ F)\wedge dF^I.
\]
On the other hand,
\[
d\omega=\sum_I d\omega^I\wedge dy^I.
\]
Pulling this back and using the pullback-of-wedge property gives
\[
F^*(d\omega)=\sum_I F^*(d\omega^I)\wedge F^*(dy^I)=\sum_I d(\omega^I\circ F)\wedge dF^I.
\]
Therefore,
\[
F^*(d\omega)=d(F^*\omega).
\]
}


\subsubsection{Pullback and Interior Product}
\begin{remark}
    For a smooth map $F \colon M \to N$, the identity $i_X (F^*\omega) = F^* (i_Y \omega)$ is ill-defined for generic vector fields since $X$ is a vector field from $M$ and $Y$ is a vector field on $N$. Naturality under pullbacks holds precisely when $X$ and $Y$ are $F$-related, which motivates the next definition.
\end{remark}
\defn{$F-$related Vector Fields}{Let $F:M\to N$ be smooth. Vector fields $X$ on $M$ is said to be $\mathbf{F}$-\textbf{related} to a vector field $Y$ on $N$ if $$dF_p(X_p)=Y_{F(p)}$$ for every $p\in M$.}
\thmp{}{
Let $F \colon M \to N$ be a smooth map. If $X$ is a vector field on $M$ and $Y$ on $N$ are $F$-related, then for any differential form $\omega \in \Omega^k(N)$,
\[
i_X (F^*\omega) = F^* (i_Y \omega).
\]
}{
Evaluating the left-hand side at $p \in M$ on vectors $v_1, \dots, v_{k-1} \in T_p M$ yields:
\begin{align*}
(i_X F^*\omega)_p (v_1, \dots, v_{k-1}) 
&= (F^*\omega)_p (X_p, v_1, \dots, v_{k-1}) \\
&= \omega_{F(p)} (dF_p X_p, dF_p v_1, \dots, dF_p v_{k-1}) \\
&= \omega_{F(p)} (Y_{F(p)}, dF_p v_1, \dots, dF_p v_{k-1}) \\
&= (F^* i_Y \omega)_p (v_1, \dots, v_{k-1}). \qedhere
\end{align*}
}

\lemp{}{Let $U\subseteq \bbR^n$ be an open connected set and $f:U\to \bbR$ be a smooth function. Then $df=0$ if and only if $f$ is constant on $U$.}{$\implies$) Let $f$ be constant on $U$. Then $\frac{\partial f}{\partial x^i}=0$ for all $i\leq n$, and hence $df_p=\sum_{i=1}^n \frac{\partial f}{\partial x^i}|_p dx^i_p=0$.
\\\\ $\impliedby)$ Say $df_p=0$ for all $p\in U$. Let $x$ and $y$ be arbitrary fixed points in $U$ and let $c:[a,b]\to U $ be any smooth curve connecting $x$ and $y$ (Such a curve exists because $U$ is path-connected, and hence continuity is guarenteed. Smoothening of the polygon edges with smooth curves guarentees smoothness). Note that $c^*f\equiv f \circ c:[a,b]\to \bbR$ is a smooth $0$ form on $[a,b]$. $d(c^*f)=d(f\circ c)= \frac{d (f\circ c (t))}{dt} dt=c^* df=0$ (from \ref{pullback_commutes_with_derivative}). Therefore $d(c^*f)=0$, so $f\circ c(a)=f\circ c(b)$ which gives $f(x)=f(y)$.  }

\section{Manifolds with Boundary}
\label{app:manifold_with_bd}The following section discusses manifolds with boundary, which is crucial for the sequel on integration. Much of the discussion is taken from \cite{Tu2011}.

We ask the question: What does it mean for a function to be smooth on an \emph{arbitrary} set $S\subset \bbR^n$? We define it in the following way: 
\defn{Smooth functions for non-open sets}{Let $S\subset \bbR^n$. We say a function $f:S\to \bbR^m$ is \emph{smooth} at $p$ if there is a neighbourhood $U$ of $p$ and a $C^{\infty}$ function $\tilde{f}:U\to \bbR^m$ such that $f\equiv \tilde{f}$ on $U\cap S$. The function is \emph{smooth} on $S$ if it is smooth at every point of $S$. }
\thmp{(Invariance of domain)}{Let $U\subset \bbR^n$ be an open set, $S\subset \bbR^n$ an arbitrary set and $f:U\to S$ a diffeomorphism. Then $S$ is open in $\bbR^n$.}{Let $p\in U$. There is an open set $V$ containing $S$ and a $C^{\infty}$ map $g:V\to \bbR^n$ such that $g|_{S}\equiv f^{-1}$. Since $f$ maps $U$ onto $S$, we have $g \circ f \equiv \mathrm{1}_U$. From chain rule, we have $Dg(f(p)) \circ Df(p)\equiv \mathrm{1}_{T_pU}$. We have $Df$ to be of non zero determinant at $p$. From the inverse function theorem, $f$ is locally invertible at $p$. $f:U_p\to V_{f(p)}\subset V$ is a diffeomorphism for neighbourhoods $U_p$ of $p$ and $V_{f(p)}$ of $f(p)$. This gives us that $V_{f(p)}\subset f(U)=S$, and hence we have $S$ to be open in $V$, and hence in $\bbR^n$. }
\begin{lemma}(\textbf{\cite{Tu2011} pg 212, Proposition 21.4.})
    Let $U $ and $V$ be open subsets of $\bbH^n:=\{(x^1,x^2,\cdots,x^n): x^n\geq 0\}$ and $f:U\to V$ a diffeomorphism. Then $f$ maps interior points to interior points and boundary points to boundary points. 
\end{lemma}
\defn{Manifold with boundary}{A topological n-manifold with boundary is a second countable Hausdorff space locally homeomorphic to $\bbH^n$.

For each point $p\in M$ we have an open set $U\subset M$ and a homeomorphism $$\phi:U \to \phi(U)\subset \bbH^n$$ with $\phi(U)$ open in $\bbH^n$.

A collection of charts $\{U,\phi\}$ is an atlas if for any two cahrts $(U,\phi)$, $(V,\psi)$
$$\psi\circ \phi^{-1}:\phi(U\cap V) \to \psi(V\cap U)\subset \bbH^n $$ is $C^{\infty}$ (in the sense of smooth functions on non open sets).

A point $p$ is \emph{interior} if in a chart $(U,\phi)$, $\phi(p)$ is interior in $\bbH^n$. Similarly, $p$ is a boundary point if $\phi(p)$ is boundary point in $\bbH^n$. The set of all boundary points is defined as $\partial M$. 
}
\begin{remark}
    \begin{enumerate}
        \item A function $f:M\to \bbR$ is $C^{\infty}$ at a boundary point $p\in \partial M$ if there is a chart $(U,\phi)$ such that $f\circ \phi^{-1}$ is $C^{\infty}$ at $p\in \bbH^n$, which implies that $f\circ \phi^{-1}$  extends as a $C^{\infty}$ function to an open neighbourhood of $\phi(p)$ in $\bbR^n$.
        \item The topological boundary ($cls(M)-int(M)$) is different from the manifold boundary $\partial M$, though, when the context is clear, we will use the same notation for both. For example, the open unit disc in $\bbR^2$ has empty manifold bonudary, but non empty topological boundary.
    \end{enumerate}
\end{remark}
\subsection{Boundary of a manifold}
Let $M$ be an $n$ dimensional manifold with boundary $\partial M$. If $(U,\phi)$ a chart, let $\phi'\equiv \phi|_{U\cap \partial M}$. $$\phi':\partial M \cap U\to \partial \bbH^n\cong \bbR^{n-1} $$
and if $(U,\phi)$ and $(V,\psi)$ are two charts for $M$, then $$\psi' \circ (\phi')^{-1}: \phi'(U\cap V\cap \partial M)\to \psi'(U\cap V\cap \partial M) $$ is $C^{\infty}$. Therefore, an atlas $\{(U,\phi)\}$ for $M$ gives rise to an induced atlas for $\partial M$ given by $\{(U\cap \partial M, \phi|_{U\cap \partial M})\}$ for $\partial M$, making $\partial M$ a manifold of $n-1$ dimension. 

We define tangent spaces as we did previously, as the space of all derivations on the algebra $C^{\infty}_p$ of germs of smooth functions.
\\\\ For a manifold $M$ with boundary, we say a vector $X_p$ is \emph{inward pointing} if $X_p \not \in T_p(\partial M)$ and there is a curve $c:[0,1)\to M$ such that $c(0)=p$, $c((0,1))\subset M^{\circ}$ (in the manifold sense) and $\frac{dc}{dt}|_{t=0}=X_p$.

\subsection{Integration of forms}
Let $U\subseteq \bbR^n$ be an open set and $\alpha$ be a $1-$form. 

\defn{Parametrized Paths}{Any map $c:[a,b]\to U\subset \bbR^n$ that is smooth is said to be a \emph{path} from $c(a)$ to $c(b)$. }

The pullback of $\alpha$ given by $c^* \alpha$ is a $1-$form on $[a,b]$. Suppose $c^*\alpha=h dt$ where $t$ is the coordinate function of $[a,b]$ and $h$ a real valued smooth function.
\defn{Integral of a $1-$form over a curve}{\begin{equation}\int_c \alpha= \int_{[a,b]}c^*\alpha=\int_a^b h(t)dt \end{equation} where the right most equality makes sense in the standard way. Moreover, if $\alpha=\sum_{i=1}^n f^i dx^i$, we have $c^*(\sum_{i=1}^n f^i dx^i)=\sum_{i=1}^nc^*(f^i)d(c^* x^i)=\sum_{i=1}^n (f^i \circ c)\frac{dc^i}{dt}dt$ where $c^i=x^i \circ c$. Its integral would then be \begin{equation}
 \int_c \alpha=\int_a^b c^*\alpha=\sum_{i=1}^n\int_a^b f^i\circ c(t) \frac{d c^i}{dt}(t)dt=\int_a^b \sum_{i=1}^n f^i(c(t))(c^i)'(t)dt   
\end{equation}
Which can be suggestively written as 
\begin{equation}
    \int_a^b \alpha^{\sharp}(c(t))\cdot c'(t)dt
\end{equation} where $\alpha^{\sharp}$ is the vector field associated with the $1-$form $\alpha$ via the inner product. 
}

\defn{Integral of $n-$forms in $\bbR^n$}{Let $U\subset \bbR^n $. Let $\omega=f(x)dx^1\wedge dx^2 \cdots dx^n$ be a top form. We define its integral as \begin{equation}\label{eq:defn_integral_forms_Rn}
 \int_{U}\omega:=\int_U f(x) dx^1 dx^2 \cdots dx^n   
\end{equation} in the usual Riemann sense. }

Let $T:U\subset \bbR^n \to V\subset \bbR^n$ be a diffeomorphism and $\omega=f(y)dy^1\wedge \cdots \wedge dy^n$ a top form on $V$. Let $x^1,\cdots,x^n$ be the coordinates of $U$, and $y^1,\cdots, y^n$ that of $V$. Defining $T^i$ as $y^i\circ T$, we have from \ref{Change of variables for top forms} \begin{equation}
    dT^1\wedge dT^2 \cdots \wedge dT^n=\det \bigg{(}\frac{\partial T^i}{\partial x^j}_{1\leq i,j\leq n} \bigg{)} dx^1\wedge dx^2 \cdots \wedge dx^n
\end{equation}
Therefore, \begin{align}\label{eq:discussion_invariance_pullback_1}
    \int_U T^*\omega &=\int_U T^*(f) T^*(dy^1)\wedge T^*(dy^2)\cdots \wedge T^*(dy^n)
    \nonumber\\ &= \int_U f\circ T(x) \det (J(T)) dx^1\wedge \cdots \wedge dx^n
    \nonumber \\ &=\int_U f(T(x))dx^1 dx^2 \cdots dx^n
\end{align}
Meanwhile, \begin{align}\label{eq:discussion_invariance_pullback_2}
    \int_V \omega:=\int_V f(y)dy^1\wedge \cdots dy^n&=\int_V f(y) dy^1dy^2 \cdots dy^n
    \nonumber \\&=\int_U f(T(x))|\det(J(T))|dx^1dx^2\cdots dx^n
\end{align}
If we have that $T$ is \emph{orientation-preserving}, we must have that $\det(J(T))>0$, whence we see equality in \ref{eq:discussion_invariance_pullback_1} and \ref{eq:discussion_invariance_pullback_2}. This motivates us to the following construction:
Let $M$ be an oriented $n-$manifold, and $\omega$ a compactly supported top form. 
Given a chart $U,\phi$ and supposing that $\mathrm{supp}(\omega)\subset U$, we can define \begin{equation}
    \int_U \omega:=\int_{\phi(U)}(\phi^{-1})^* \omega 
\end{equation}
That this is independent of the chart is clear from the fact that for any other chart $(U,\psi)$, $\phi\circ \psi^{-1}:\psi(U)\to \phi(U)$ is an orientation preserving diffeomorphism (assuming that the charts are taken from the \emph{oriented atlas}). 
\\\\ Let $\omega$ be a top form compactly supported in $M$. Let $\rho_{\alpha}$ be a partition of unity subordinate to $U_{\alpha}$ coming from the oriented atlas. One can then write \begin{equation}
    \omega=\sum_{\alpha} \rho_{\alpha} \omega
\end{equation} since this is always a finite sum (due to the finite intersection property of partitions of unity).
Since $\mathrm{supp}(\rho_{\alpha} \omega)\subset U_{\alpha}$, we can define 
\begin{equation}\label{eq:defn:integral_of_n_forms}
    \int_M \omega:=\sum_{\alpha} \int_{U_{\alpha}}\rho_{\alpha}\omega 
\end{equation}
We do not show that this is independent of the choice of the partition of unity. For details, refer pg. 266, \cite{Tu2011}.

Writing the boundary (locally) as $f(x)=x_n-g(x^1,x^2,\cdots,x^{n-1})=0$, we get the unit normal at $x_0$ as \begin{equation}\nu(x_0)=\frac{(-\nabla'g (x'),1)}{\sqrt{1+|\nabla'g (x')|^2}}. \end{equation}
For Lipschitz boundaries, the normal is still defined almost everywhere (Theorem 3.2 (\textbf{Rademacher}), \cite{EvansGariepy2015}), but even so, we lose out on the continuity of $\nu$.

    \int_M d\beta=\int_{\partial M}\beta
\end{equation} }{We need only consider the case when $M$ is $\bbR^n$ (or $\bbH^n$). Indeed, if the theorem were to be true for $\bbR^n$(or $\bbH^n$), then it is true for charts $U_{\alpha}$ (since $\phi_{\alpha}$ are diffeomorphisms) and one can invoke the partition of unity to prove the result for $M$. 

To this end, let $M=\{(x^1,x^2,\cdots,x^n):x^n\leq 0\}$ and hence $\partial M=\{(x^1,\cdots,x^{n-1})\}$. (Note that it is enough to consider forms of the kind \ref{throwaway_for_proof} by linearity).

When $j<n$, we have $$\int_M d\beta:=\int dx^1 dx^2 \widehat{dx^j}\cdots dx^n \int_{-\infty}^{\infty} \frac{\partial b_j(x)}{dx^j} dx^j =0 $$

When $j=n$, we have $d\beta=(-1)^{n-1} \partial_n b_n(x) dx^1 \wedge \cdots \wedge dx^n$, while the positively oriented boundary volume form takes the form $(-1)^{n-1} dx^1 \wedge \cdots \wedge dx^{n-1}$(Recall that the boundary orientation form comes from the interior product of the orientation form for $M$ with the outward unit normal). We thus have \begin{align*}\int_M d\beta=(-1)^{n-1}\int dx^1dx^2 \cdots dx^{n-1}\int_{-\infty}^{0} \frac{\partial b_n(x)}{\partial x^n}dx^n&=(-1)^{n-1}\int b_n(x',0)dx'\\&=\int_{\partial M} \beta \end{align*}
which proves the result.

\section{Topological Vector Spaces}
The following development is loosely based on \cite{RudFA}.
\defn{Topological Vector Space}{A Vector Space $X$ over $\bbK (\bbR \text{ or }\bbC)$ is a \emph{Topological Vector Space}(TVS, from now on) if it has a topology $\mathscr T_X$ such that:
\begin{enumerate}
    \item Point sets are closed ($T_1$ axiom).
    \item The vector space operations $(v,u)\mapsto v_u$ and $(\alpha,x)\mapsto \alpha x$ are continuous as maps from $X\times X\to X$ and $\bbK \times X\to X$ respectively. 
\end{enumerate}}
\begin{lemma}
    \label{theorem_TVS:translation_and_multiplication}
    The translation map $T_a:X\to X$ given by $T_a(x)=a+x$, and the multiplication map $M_{\lambda}:X\to X$ given by $M_{\lambda}(x):=\lambda x$ are Homeomorphisms.

\end{lemma}

\ref{theorem_TVS:translation_and_multiplication} tells us that $U$ is open in $X$ if and only if $U+a$ is open in $X$. This implies that the topology of a TVS is completely determined by a local base at $x=0$. 

\defn{Convex Set}{A set $C\subset X$ is said to be \emph{convex} if for any $x,y\in X$, $t\in [0,1]$, $$tx+(1-t)y\in X $$ In other words, $tC+(1-t)C\subset C$. }


\defn{Locally Convex TVS (LCTVS)}{A TVS $X$ is \emph{Locally Convex} if there exists a local basis of $0$ that consists of convex sets.}
\defn{Invariant Metric}{A metric $d$ on a TVS $X$ is \emph{invariant} if $d(x+h,y+h)=d(x,y)$}


\subsection{Banach and Frechet Topologies}
\defn{Seminorm}{A function $p:X\to \bbR$ is a seminorm if it satisfies the following:
\begin{enumerate}
    \item $p(x)\geq 0, \forall x\in X$
    \item $p(x+y)\leq p(x)+p(y)$
    \item $p(\lambda x)=|\lambda| p(x)$ for $\lambda\in \bbK$.
\end{enumerate}
}
\begin{example}\label{Example:Seminorm_topology_generated}
    (\textbf{Topology generated by a family of seminorms}) Let $\{p_i\}$ be a countable family of seminorms on a vector space $X$ which \emph{separates points}, i.e. if $p_i(x)=0$ for all $i\in \bbN$, then $x=0$. We make it a TVS in the following way: Let $U_{i,\delta}(z):=\{x\in X: p_i(x-z)<\delta\}$. This collection $\{U_{i,\delta}(z):i\in \bbN,\delta>0,z\in X\}$ forms a sub-basis, and generates the topology of $X$. 
\end{example}
We can get a description of the topology so generated: $V_{\epsilon,k}(z):=\{y\in X: p_i(y-z)<\epsilon \ \forall i\leq k\}$ forms a basis for the TVS $X$ as we vary $\epsilon>0, k\in \bbN$ and $z \in X$.
One can, in fact, show that the topology generated above is \emph{metrizable} and the metric is given by \begin{equation}
    d(x,y):=\sum_{i=1}^{\infty} \frac{p_i(x-y)}{2^i(1+p_i(x-y))}
\end{equation}
Let $B_{\epsilon}(0)$ be the open ball of radius $\epsilon$ at $0$ coming form the metric above. Let $\epsilon$ be so that $\sum_{i=N}^{\infty}\frac{p_i(x)}{2^i(1+p_i(x))}<\frac{\epsilon}{2}$. Let $x$ be so that $x\in V_{\delta=\frac{\epsilon}{2}(1-1/2^N), N}(0)$, i.e. $p_i(x)<\frac{\epsilon}{2}(1-1/2^N)$ for all $i\leq N$. Then we have $$d(x,0)< \epsilon $$ which means that $V_{\delta,N}(0)\subset B_{\epsilon}(0)$.
For the reverse inclusion: Let $d(x,0)=\sum_{i=1}^{\infty}\frac{p_i(x)}{2^i(p_i(x)+1)}<\delta$ which means for every $i\in \bbN$, $\frac{p_i(x)}{(p_i(x)+1)}<\delta 2^i$. Let $\delta$ be so that $2^K \delta<\epsilon/1+\epsilon$. Then for all $i\leq K$, $p_i(x)<\epsilon$ which implies $x\in \cap_{i=1}^K V_{\epsilon,i}(0)$ which is a basis element of the Frechet space. Hence, they form equivalent topologies. 
\defn{Frechet Space}{A TVS $X$ is a Frechet Space if it is a complete metric space with an invariant metric, but not necessarily \emph{normable}. Its topology comes from a countable family of seminorms $\{p_i\}$ as illustrated in \ref{Example:Seminorm_topology_generated}.}

\begin{theorem} (\textbf{Characterization of Frechet spaces})
A Hausdorff locally convex topological vector space is metrizable iff its topology is generated by a countable family of seminorms.
    
\end{theorem}


\begin{example}\label{example:C^kK}
    \textbf{(The Banach topology of $C^k(K)$)} Let $K$ be a compact set in $\bbR^n$. We  define the norm on $C^k(K)$ as \begin{equation}
        ||u||_{C^k(K)}:=\sum_{|\alpha|=0}^k ||D^{\alpha}(u)||_{L^\infty(K)}
    \end{equation}
This is easily seen to be a genuine \emph{norm}. Moreover, this space is complete with respect to this norm; See Theorem 8.2.3 in \cite{BartleSherbert2011}.

\end{example}
\begin{remark}\label{remark:C^k_on_compact}
    \textbf{A word of caution:} \emph{What does it mean for a function to be \textbf{differentiable} on a \textbf{compact set}?} We consider the following notion of differentiability: We say a function $f$ is of class $C^k$ on a compact set $K$ if there is an open set $U\subseteq \bbR^n$ with $K\subset U$ and a function $\tilde f:U\to \bbR^m$ of class $C^k$ such that $\tilde f=f$ when restricted on $K$.
\end{remark}
\begin{remark}
    There is, however, a different notion by which we mean $C^k(\bar{\Omega})$. This is given below:
\end{remark}
\defn{$C^{k}(\bar{\Omega})$}{\label{def:C^kOmega}Let $\Omega$ be a bounded domain. The collection of all functions $f$ of class $C^{k}$ on $\Omega$ whose derivatives $D^{\alpha}f$ extends continuously to the boundary of $\Omega$. }
\begin{remark}
    Note that, even though $\bar\Omega$ is compact for a bounded domain in $\bbR^n$, the two notions \ref{def:C^kOmega} and \ref{remark:C^k_on_compact} are \emph{not the same}. We require additional regularity on the set $\Omega$ for the two notions to be equal. 
\end{remark}


\subsubsection{Frechet Topology of $C^k(\Omega)$}
Let $\Omega$ be an open set in $\bbR^n$. We consider an exhaustion $\{K_i\}$ of $\Omega$ consisting of compact sets (e.g. $\{x\in \Omega:||x||\leq n \text{ and } \mathrm{dist}(x,\Omega^c)\geq 1/n \}$). We define $p_i(u):=||u||_{C^k(K_i)}$ (as defined in \ref{example:C^kK}). This forms a countable collection of semi-norms, and hence can be topologised as in \ref{Example:Seminorm_topology_generated}. A cauchy sequence is such that for every $i\in \bbN$, $$p_i(u_n-u_m)\to 0$$ as $n,m \to \infty$. This means that in $C^k(K_i)$, $u_n|_{K_i}$ is cauchy and thus convergent (as $C^k(K_i)$ is Banach).  Therefore, $C^k(\Omega)$ is a Frechet space. 
\begin{remark} 1. It is easy to see that the collection $\{p_i\}$ separates points.

2. It should be noted that, the topology is independent of our choice of exhaustion $\{K_i\}$ as for any other exhaustion $\{F_j\}$, we see that they are \emph{mutually dominant} (for every $i$, $\exists j(i)$ such that $K_i\subset F_{j(i)}$ and vice versa). 
\end{remark}
\textbf{What it means to be Cauchy:} A sequence $u_n$ is cauchy in $C^k(\Omega)$ if given any neighbourhood $V_{\varepsilon,k}$ of $0$, there exists $n_0$ such that $u_n-u_m \in V_{\varepsilon,k}$ for $n,m\geq n_0$. This can be recharacterised in the following way: 
\begin{center}
\emph{$u_n$ is Cauchy if and only if for every $i\in \bbN$, $p_i(u_n-u_m)\to 0$ as $n,m \to \infty$.}
    
\end{center}

\textbf{What it means to converge:} A sequence $u_n$ converges to $u$ in $C^k(\Omega)$ if for every neighbourhood $V_{\varepsilon,s}$ of $0$, there exists $n_0$ such that $u_n-u \in V_{\varepsilon,s}$. This can be recharacterised in the following way: 
\begin{center}
    \emph{$u_n\to u$ if and only if for every $i\in \bbN$, $p_i(u_n-u)\to 0$ as $n\to \infty$. }
\end{center}
\begin{remark}
    The above characterisation is true in general for any Frechet space. 
\end{remark}

\subsubsection{Frechet Topology of $C^{\infty}(K)$}
Let $K$ be a compact subset of $\bbR^n$. Let $$p_j(u):=||u||_{C^{j-1}(K)}$$
This forms a countable family of seminorms that separate points, and hence $C^{\infty}(\Omega)$ is a Frechet space under this topology.

\subsubsection{Frechet Topology of $C^{\infty}(\Omega)$}
Let $\{K_i\}$ be an exhaustion of $\Omega\subset \bbR^n$. Let $$p_{i,j}(u):=||u||_{C^{j-1}(K_i)}$$
This yields a countable family of seminorms that separate points. With this topology, $C^{\infty}(\Omega)$ is a Frechet space (we omit the proof of completeness). 

\subsubsection{Frechet Topology of $L^p_{\mathrm{loc}}(\Omega)$}
Let $\Omega$ be an open set and let $\{K_i\}_{i\in \bbN}$ be an exhaustion. We define $L^p_{\mathrm{loc}}(\Omega)$ to be the space of all measureable functions $f: \Omega\to \bbK$ such that \begin{equation}
    \big.(\int_{\Omega'} |f|^p d\mu(x)\big.)^{\frac{1}{p}}<\infty
\end{equation}
We define $p_m(u):=\Bigg.(\int_{K_m}|u|^{p} d\mu \Bigg.)^{\frac{1}{p}} $ which gives us a sequence of seminorms for $L^p_{\mathrm{loc}}(\Omega)$, making it a Frechet space.
\subsection{Inductive Limit Topologies}
\defn{Inductive Limit of Frechet Spaces (LF spaces)}{These are complete topological vector spaces (in the sense of cauchy nets or cauchy filters. cf.\cite{SchaeferWolff1999}) that are a union of of an increasing sequence $E_n$ of Frechet spaces such that $E_n \hookrightarrow E_{n+1} $ is an isomorphism of topological vector spaces. $X=\cup_i E_i$ and we say a set $E$ in $X$ is open if and only if $E_i \cap E$ is open in $E_i$.}
\begin{remark}
    1. \textbf{LF spaces are not \emph{in general} metrizable.}
    
    2. The above point implies that it is \emph{\textbf{not enough}} to consider \emph{sequences} alone to try to characetrise topologies, as we did in the case of Frechet spaces (which were metrizable).
\end{remark}
\subsubsection{LF Topology on $C_c(\Omega)$}
Let $\{K_i\}$ be an exhaustion of $\Omega$, and define $$E_j:=\{u\in C_c(\Omega):\mathrm{supp}(u)\subset K_j\}$$
We see that $E_j$ is a vector space and we can give $E_j$ a Banach space structure with the norm $||u||_{E_j}:=||u||_{L^{\infty}(K_j)}$. Moreover, $E_1\subset E_2\subset \cdots \subset \Omega$ and the inclusions are isomorphisms of Banach spaces. 

\subsubsection{LF Topology on $C^k_c(\Omega)$}
Let $\{K_i\}$ be an exhaustion and $$E_j:=\{u\in C^k_c(\Omega): \mathrm{supp}(u)\subset K_j \}$$
This is a linear subspace of $C^k_c(\Omega)$, with the norm $$||u||_{E_j}:=||u||_{C^k(K_j)}=\sum_{|\alpha|=1}^k ||D^{\alpha}u||_{L^{\infty}(K_j)} $$ which makes it a Banach Space. The inclusions $E_1 \hookrightarrow E_2 \hookrightarrow \cdots$ are also TVS isomorphisms, making $C^k_c(\Omega)$ an LF space.
\subsubsection{LF Topology on $C^{\infty}_c(\Omega)$}
Let $$E_j:=\{u\in C^{\infty}_c(\Omega):\mathrm{supp}(u)\subset K_j\}$$ and define $$p^j_i(u):=||u||_{C^i(K_j)}$$
This is a collection of semi-norms on $E_j$, making it a Frechet space. 
We then have $$C^{\infty}_c(\Omega)=\cup_j E_j$$ making $C_c^{\infty}(\Omega)$ an LF space. 

\subsection{Criteria for convergence}
In the Banach space setting, the criteria for convergence is quite simple: $u_n\to u$ if and only if $||u_n-u||\to 0$. 

The following proposition characterises convergence in Frechet spaces:
\begin{proposition}
Let $E$ be a Frechet space with a countable collection of seminorms $\{p_i\}$. A sequence $u_n$ converges to $u$ if and only if for each $i\in \bbN$, $$p_i(u_n-u)\to 0$$ as $n\to\infty$.    
\end{proposition}

The following proposition characeterises convergence in LF spaces:

\begin{proposition}\label{defn:convergence_in_LF}{Let $E=\cup_i E_i$ be the inductive limit of Frechet spaces $E_i$ with $E_1\hookrightarrow E_2 \hookrightarrow \cdots \hookrightarrow E$. A sequence $u_n$ converges to $u$ in $E$ if and only if there exists an integer $k\in \bbN$ such that $$\{u_n\}_{n\in \bbN}\subset E_k, \ u\in E_k \ \text{and} \ u_n\to u \in E_k (\text{in the Frechet sense})$$}\end{proposition}


\begin{example}
Consider the space $C^k_c(\Omega)$ (with an exhaustion $\{K_j\}$ of $\Omega$), which is an LF space, with $E_j:=\{u\in C^k_c(\Omega):\mathrm{supp}(u)\subset K_j\}$, equipped with the norm $$||u||_{E_j}:=||u||_{C^k(K_j)}$$
Let $u_n$ be a sequence converging to $u$ in $E=C^k_c(\Omega)$. This means that (from the characterisation in \ref{defn:convergence_in_LF}) there exists a large enough $q$ such that $\{u_n\}_{n\in \bbN} \subset E_q$, or rather, the supports of $u_n$ are in $K_q$, and $u_n\to u$ in the norm sense, in $E_q$. This means, $$||u_n-u||_{C^k(K_q)}=\sum_{|\alpha|=0}^k |D^{\alpha}(u_n)-D^{\alpha}(u)|_{L^{\infty}(K_q)}\to 0$$ which amounts to (since $E_q$ is Banach) $D^{\alpha}(u)|_{K_q}\to D^{\alpha}u|_{K_q}$ uniformly in $K_q$ for all $\alpha\in \{0,1,\cdots,k\}$.    
\end{example}

\subsection{Continuous linear maps on LCTVS}
\defn{Sequentially Continuous}{A map $f:X\to Y$ between topological spaces is \emph{sequentially continuous} if for every $x_n\to x$ in $X$, we have $f(x_n)\to f(x)$ in $Y$.}

\begin{proposition}
    A map from a \emph{first countable} topological space $X$ to a topological space $Y$ is continuous iff it is sequentially continuous.
\end{proposition}

We now describe what it means for a linear map between Frechet spaces to be continuous:
\subsubsection{Continuity on Frechet space}
\begin{proposition}
    Let $E$ and $F$ be Frechet spaces and $\{p_i\}_{i\in \bbN}$ and $\{q_i\}_{i\in \bbN}$ the associated countable family of seminorms. A linear map $T:E\to F$ is continuous if and only if for every $i\in \bbN$, there exists $j\in \bbN$, and a constant $C=C(i)$ such that $q_i(Tx)\leq Cp_j(x)$ for every $x\in E$.
\end{proposition}

\begin{proposition}
    Let $E$ be a Frechet space with a countable family $\{p_i\}$ of seminorms generating its topology. A linear map $T:E\to \bbK$ is continuous iff for each integer $i\in \bbN$, there exists a constant $C=C(i)$ such that $$|Tx|\leq Cp_i(x)\ \text{for all }x\in E$$ 
\end{proposition}

\subsubsection{Continuity on LF space}
Let $E=\cup_i E_i$ where $\{E_i\}$ is an increasing collection of Frechet spaces, with $E_1\hookrightarrow E_2 \hookrightarrow \cdots \hookrightarrow E $ being isomorphisms of TVS. We begin with an observation that the Frechet topology on $E_i$ is the same as the topology induced from $E$. This amounts to the following:

\begin{proposition}
    Let $E$ be an LF space with $E=\cup_i E_i$. Then any convex subset $U\subset E$ containing the origin is a neighbourhood of $0$ if and only if for each $n\in \bbN$, $U\cap E_n$ is a neighbourhood of $0$ in $E_n$.
\end{proposition}

We can now characterise continuous linear maps, with the following proposition: 

\begin{proposition}
    Let $E$ be an $LF$ space with $E=\cup_i E_i$, an increasing limit of a sequence of Frechet spaces. Let $F$ be any LCTVS. A linear map $T:E\to F$ is continuous if and only if for each $n\in \bbN$, $$ T|_{E_n}\to F$$ is continuous in the Frechet sense.
\end{proposition}


\section{Dual Spaces, Weak and Weak* Topologies of Normed Linear Spaces}
\subsection{Weak topology}
\defn{Dual space $X^*$}{Let $X$ be a normed linear space over $\bbK$. The \emph{topological dual space}(generally referred to as \emph{the dual space}) is defined as the space of all \emph{continuous linear functionals on }$X$. }

We set up some notation: Let $X$ be a normed linear space and $X^*$ its topological dual. We can make a bilinear pairing $$
    \langle,\rangle: X^*\times X\to \bbK
$$
by \begin{equation}
    \langle f,x\rangle:=f(x)
\end{equation} and we will choose to write $\langle f,x\rangle$ in place of $f(x)$. 

\defn{The weak topology on $X$ ($\sigma(X,X^*)$)}{The weak topology on $X$, denoted $\sigma(X,X^*)$ is the \emph{coarsest} topology that makes the collection $\{f:X\to \bbK: f\in X^*\}$ continuous.}
\begin{theorem}\label{thm:Basis_for_weak}\textbf{(Basis for Weak Topology)}
    $\sigma(X,X^{*})$ is generated by the collection $$\{\cap_{i=1}^n f^{-1}_i(U_i): f_i \in X^*, U_i \text{ open in }\bbK, n\in \bbN\}$$
\end{theorem}

\begin{proposition}
    Let $X$ be a normed linear space. $x_n\in X$ converges to $x\in X$ in the weak topology if and only if $\phi(x_n)\to\phi(x)$ for all $\phi\in X^*$ 
\end{proposition}
Convergence in the weak topology is called as \emph{weak convergence}. 

We gather a few facts about weak topology $\sigma(X,X^*)$ (without proof):
\begin{lemma}
    The weak topology $\sigma(X,X^*)$ is Hausdorff. 
\end{lemma}
The proof comes from a direct application of the geometric form of the \emph{Hahn-Banach Theorem}.
\begin{proposition}
    Let $X$ be a NLS and $\sigma(X,X^*)$ the weak topology on $X$. This has a basis of the form $V(f_1,f_2,\cdots,f_k;x_0,\varepsilon)$ (where $\{f_i\}_{i=1}^k$ are elements of $X^*$, $x_0\in X$, and $\varepsilon>0$) given by $$V(f_1,f_2,\cdots,f_k;x_0,\varepsilon):=\{x\in X: |\langle f,x-x_0\rangle|<\varepsilon \text{ for all }i\in \{1,\cdots,k\}\} $$
\end{proposition}
\begin{remark} We make some important remarks:
    \begin{enumerate}
        \item In general, weakly closed implies weakly sequentially closed.
        \item $C$ being weakly closed \emph{always} implies that $C$ is strongly closed (recall that that the weak topology is indeed weaker than the strong norm topology), but \emph{the converse is not always true}. For convex sets however, the coverse holds true.
    \end{enumerate}
\end{remark}

\begin{lemma}
    Let $C\subset X$ be a convex set that is strongly closed. Then $C$ is weakly closed.
\end{lemma}
\emph{The proof is again an application of the geomeric form of Hahn-Banach theorem}.

\begin{theorem}{\label{thm:characterisation_of_weak_convergence}
\textbf{Characterisation of weak convergence \cite{Brezis2011}}    
\begin{enumerate}
    \item $x_n \rightharpoonup x$ if and only if $f(x_n)\to f(x)$ for all $f\in X^*$.
    \item Strong convergence in $X$ implies weak convergence in $X,\sigma(X,X^*)$.
    \item If $x_n \rightharpoonup x$, then $||x_n||$ is bounded and $||x||\leq \liminf ||x_n||$.
    \item If $x_n\rightharpoonup x$ and $f_n\to f$ strongly in $X^*$, then $\langle f_n,x_n\rangle \to \langle f,x\rangle$.
\end{enumerate}}\end{theorem}

\subsection{Weak* topology}
\subsubsection{The $J$ map}
Consider for every $x$ a map $J_x \in X^{**}$ whose action on a given linear functional $f\in X^*$ is given by \begin{equation}
    J_x(f):=\langle f,x\rangle
\end{equation}
The map $x\mapsto J_x$ as a map from $X\to X^{**}$ is easily seen to be linear. Moreover, if $J_x=J_y$, then we would have $\langle f,x\rangle=\langle f,y\rangle$ for all $f\in X^*$ which implies $x=y$. Therefore, $J:X\to X^{**}$ is \emph{injectlve}. Give $X^{**}$ the operator norm topology, from which we get $$||J_x||:= \sup_{f\in X^*, ||f||_{op}=1} |J_x(f)|= \sup_{f\in X^*, ||f||_{op}=1} |\langle f,x\rangle|=||x||_{X}$$ from Hahn-Banach.
Therefore, $J$ is actually an \emph{isometry} of Normed Linear Spaces, which means we can see $X$ as an isometric subspace of $X^{**}$. We collect this result in the following theorem:
\begin{theorem}\label{thm:X_isometry_to_X**}
Let $X$ be a normed linear space and $J:X\to X^{**}$ be given by $J_x(f):=\langle f,x\rangle$ for all $f\in X^*$. Then $x\mapsto J_x$ is linear, continuous and an isometry of normed linear spaces.    
\end{theorem}

\defn{Weak* topology on $X^*$($\sigma(X^*,X)$)}{We define the \emph{weak*} topology on $X^*$ to be the coarsest topology on $X^*$ that makes each $J_x:X^*\to \bbK$ continuous. }
\begin{remark}
 This is easily seen to be \emph{weaker} than the weak topology on $X^*$, as we require less maps to be continuous.

\end{remark}
We now describe a useful basis for the weak* topology on $X^*$.
\begin{proposition}
    Let $X$ be a NLS. $X^*$ with the topology $\sigma(X^*,X)$ has a basis of the following form (with $x_1,x_2,\cdots,x_k \in X$, $\varepsilon>0$ and $f_0\in X^*$): $$V(x_1,x_2,\cdots,x_k;\varepsilon,f_0):=\{f\in X^*: |\langle f-f_0,x_i\rangle|<\varepsilon, \text{ with }i\in\{1,\cdots,k\}\}$$
\end{proposition}

\textbf{Notation for weak* convergence:} When a sequence $f_n\in X^*$ converges to $f$ in the \emph{weak* topology}, we denote it by: $$f_n \overset{\ast}{\rightharpoonup} f $$

\begin{theorem}
    \textbf{Characterisations of weak* convergence\cite{Brezis2011}}
    \begin{enumerate}
        \item $f_n\overset{\ast}{\rightharpoonup} f$ iff $\langle f_n,x\rangle\to \langle f,x\rangle$ for all $x\in X$. 
        \item $f_n\to f$ in the operator norm implies $f_n \overset{\ast}{\rightharpoonup} f$ in $\sigma(X^*,X)$. Moreover, if $f_n \rightharpoonup f$ in $\sigma(X^*,X^{**})$, then $f\overset{\ast}{\rightharpoonup} f$ in $\sigma(X^*,X)$.
        \item If $f_n \overset{\ast}{\rightharpoonup} f$, then $||f_n||$ is bounded and $||f||\leq \liminf ||f_n||$.
        \item If $f_n\overset{\ast}{\rightharpoonup}f$ and $x_x\to x$ strongly, then $\langle f_n,x_n\rangle \to \langle f,x\rangle$.
    \end{enumerate}
\end{theorem}

\section{Distributions}
\defn{Distributions}{Let $\Omega\subset \bbR^n$ be open. A \emph{distribution} is a \textbf{continuous linear functional} on the LCTVS $C^{\infty}_c(\Omega)$. The space of all distributions is denoted $\mathscr D'(\Omega)$.}
\begin{theorem}\label{thm:characterise_distributions}
    \textbf{Characterisation theorem for distributions:}

    TFAE:
    \begin{enumerate}
        \item $T:C^{\infty}_c(\Omega)\to \bbR$ is a \emph{distribution}.
        \item For every compact set $K\subset \Omega$, there exists a constant $C=C(K)$ and an integer $m=m(K)$ such that $$|T(\psi)|\leq C||\psi||_{C^m(K)} \text{ for all }\psi \in C^{\infty}(\Omega)\text{ with }\mathrm{supp}(\psi) \subset K.$$
        \item $\mathbf{T}$ \textbf{is sequentially continuous}. For a sequence $\psi_n$ converging to $\psi$ in $C^{\infty}_c(\Omega)$ (which is tantamount to saying the following: For a fixed exhaustion $\{K_i\}$, there is an $E_j:=\{\psi\in C^{\infty}_c(\Omega): \mathrm{supp}(\psi) \subset K_j \}$ such that $\psi_n$ is in $E_j$, and converges to $\psi$ in the Frechet sense.), we have $$T(\psi_n)\to T(\psi).$$
    \end{enumerate}

\end{theorem}

\defn{$L^1_{\mathrm{loc}}(\Omega)$}{The space of all Lebesgue measurable functions $f:\Omega\to \bbR$ that \emph{locally integrable},i.e. functions for which we have for every compact set $K\subset \Omega$ $$\int_K |f(x)|d\mu(x)<\infty $$}

Given $f\in L^1_{\mathrm{loc}}(\Omega)$, we define \begin{equation}\label{eq:T_f}
    T_f(\phi):=\int_{\Omega} f(x)\phi(x)d\mu(x) \text{ for all }\phi\in C^{\infty}_c(\Omega)
\end{equation}
We immediately see that $T_f$ is a linear operator on $C^{\infty}_c(\Omega)$.
That it is continuous is evident from the following: Let $K$ be a compact set. $$|T_f(\varphi)|\leq \int_K |f(x)||\varphi(x)|d\mu(x)\leq \Big(\int_K |f(x)|d\mu(x)\Big)||\varphi||_{C^0(K)}$$

\subsection{Convolutions and mollifiers}
\defn{Convolution}{Let $f,g:\bbR^n\to \bbR$ be measurable. The \emph{convolution} of $f$ and $g$, denoted $f\ast g$ is defined as: \begin{equation}
    f\ast g (z):=\int_{\bbR^n} f(z-t)g(t)dt
\end{equation}}
We list without proof some properties of convolutions: 
\begin{enumerate}
    \item $\ast$ is commutative, i.e. $f\ast g=g\ast f$.
    \item $\ast$ is associative. $(f\ast g)\ast h=f \ast (g\ast h)$.
    \item $\mathrm{supp}(f\ast g)\subset \overline{\mathrm{supp}(f)+\mathrm{supp}(g)}$.
\end{enumerate}
\begin{theorem}\textbf{(General H\"{o}lder inequality)} \label{thm:Holder_for_many_functions}
Let $f_1,f_2,\cdots,f_k:\Omega \to \bbR$ be measurable functions in $L^{p_1}(\Omega),L^{p_2}(\Omega),\cdots,L^{p_k}(\Omega)$ respectively, such that $$1/p_1+1/p_2+\cdots+1/p_k=1.$$ Then we have $$\int_{\Omega}|f_1||f_2|\cdots|f_k|d\mu \leq ||f_1||_{p_1}||f_2||_{p_2}\cdots||f_k||_{p_k}  $$   
\end{theorem}

\thmp{(Young's inequality for convolutions)}{Let $p,q \geq 1$ with $f\in L^p(\bbR^n)$, $g\in L^q(\bbR^n)$ and let $r\geq 1$ be such that $$1/p+1/q=1+1/r. $$ Then \begin{equation}
    f\ast g \in L^r(\bbR^n) \text{ and }||f\ast g||_r \leq ||f||_p ||g||_q
\end{equation} }{Without loss of generality, assume $f$ and $g$ are positive. 
We have
$$f(z-t)g(t)=\Big(f(z-t)^p g(t)^q\Big)^{1/r}(f(z-t)^p)^{1/p-1/r}(g(t)^q)^{1/q-1/r}$$ 
Since the exponents $r$, $rp/r-p$ and $rq/r-q$ are conjugate, we can apply \ref{thm:Holder_for_many_functions}
to get $$f\ast g (z)=\int_{\bbR^n} f(z-t)g(t)dt \leq \Big(\int_{\bbR^n}f(z-t)^p g(t)^q dt\Big)^{1/r}  ||f||_p^{1-p/r}||g||_q^{1-q/r}  $$
\begin{align*}||f\ast g||_r&=\Big(\int_{\bbR^n} (f\ast g(z))^r dz\Big)^{1/r} \\ &\leq \Big(\int_{\bbR^n} \int_{\bbR^n} ||f||_p^{r-p}||g||_q^{r-q} f(z-t)^p g(t)^q dt dz\Big)^{1/r}\\ &=\Big(||f||_p^{r-p}||g||_q^{r-q}||f||_p^p||g||_q^q\Big)^{1/r} \leq ||f||_p||g||_q \end{align*}
And we have the result.
}
\begin{proposition}
    If $f\in L^{1}(\bbR^n)$ and $g\in C^{k}(\bbR^n)$ with $D^{\alpha}g \in L^{\infty}(\bbR^n)$ for all $|\alpha|\leq k$, then $f\ast g \in C^{k}(\bbR^n)$ and $$D^{\alpha}(f\ast g)(x)=f \ast D^{\alpha}g (x) $$ for each multiindex $\alpha$ with $|\alpha|\leq k$.
\end{proposition}
\vspace{2pt}

A slightly different version of the proposition is given below:
\begin{proposition}
    If $f\in L^1_{\mathrm{loc}}(\bbR^n)$ and $g\in C^k_c(\bbR^n)$, then $f\ast g\in C^k(\bbR^n)$ and $$D^{\alpha}(f\ast g)=f\ast D^{\alpha}g $$ for each multiindex $\alpha$ with $|\alpha|\leq k$. 
\end{proposition}

\subsubsection{Mollifiers}
\defn{Mollifying kernel}{A function $\phi:\bbR^n\to \bbR$ is a \emph{mollifying kernel} if \begin{enumerate}
    \item $\phi\in C^{\infty}_c(\bbR^n)$.
    \item $\mathrm{supp}(\phi)\subset \overline{B_1(0)}$.
    \item $\phi\geq 0$ and non-vanishing.
    \item For some finite real number $a\in \bbR$ with $a> 0$ we have $$\int_{\bbR^n} \phi(x) dx=a $$
\end{enumerate}}
Given a Mollifying kernel $\phi$ such that $$\int_{\bbR^n} \phi(x)dx =1 \text{ and }\mathrm{supp}(\phi)=\overline{B_1(0)},$$  we can construct a \emph{sequence of mollifiers} $\phi_{\epsilon}\in C^{\infty}_c(\bbR^n)$ corresponding to $\phi$ given by \begin{equation}\label{eqn:mollifiers}
    \phi_{\epsilon}(x)=1/\epsilon^n \phi(x/\epsilon) \text{ for all }x\in \bbR^n
\end{equation}  
for every $\epsilon>0$.
For this sequence of mollifiers, the following holds: 
\begin{proposition}
    For every $\epsilon>0$, $\phi_{\epsilon}$ defined as in \ref{eqn:mollifiers} satisfies $$\int_{\bbR^n}\phi_{\epsilon}(x)dx =1 \text{ and }\mathrm{supp}(\phi_{\epsilon})=\overline{B_{\epsilon}(0)}.$$ Moreover, $$\lim_{\epsilon\to 0}\phi_{\epsilon}(x)=0 \text{ whenever }x\neq 0.$$ 
\end{proposition}

\begin{theorem}\textbf{(Mollification theorem)}\label{thm:mollification}
    Let $\phi$ be a mollifying kernel (assuming $a=1$) and $\{\phi_k\}$ be a sequence of mollifiers given by $$\phi_k(x)=k^n \phi(kx).$$ Then, the following holds:
    \begin{enumerate}
        \item For $1\leq p<\infty$, let $f\in L^p(\bbR^n)$. Then we have that $f_k(x):=\phi_k \ast f \in C^{\infty}(\bbR^n)$ for each $k \in \bbN$, and $$||f_k-f||_p \to 0 \text{ as }k\to \infty. $$
        \item Let $f\in C(\bbR^n)$. Then $f_k:=\phi_k \ast f\in C^{\infty}(\bbR^n)$ for all $k\in \bbN$, and $$f_k\to f \text{ uniformly on any compact subset }K\subset \bbR^n$$
        \item Let $f\in C^k(\bbR^n)$. Then $f_s:=\phi_s \ast f \in C^{\infty}(\bbR^n)$ for each $s\in \bbN$ and $$D^{\alpha}f_s \to D^{\alpha}f \text{ uniformly on every compact set }K \subset \bbR^n, \text{ for all }\alpha \text{ such that }|\alpha|\leq k. $$ 
    \end{enumerate}

\end{theorem}

We present now a crucial result which can be proved with the tools we have developed thus far: 

\thmp{\label{thm:fundamental_lemma_of_CV}(Fundamental lemma of calculus of variations)}{Let $\Omega\subset \bbR^n$ be open and $u\in L^1_{\mathrm{loc}}(\Omega)$ be such that for every $\phi \in C^{\infty}_c(\Omega)$ $$\int_{\Omega} u(x)\phi(x)dx=0 $$ Then $u=0 \ a.e. \ x \in \Omega$.}{Let $K$ be an arbitrary compact set. Let $$\nu:=\begin{cases}
    \mathrm{sgn} (u) & \text{if }x\in K \\
    0 & \text{else} 
\end{cases}$$
Note that $\nu$ is in $L^1_{\mathrm{loc}}(\Omega)$, and $\|\nu\|_{\infty}=1$. Since we can find a compact exhaustion for $\Omega$, there is always a larger compact set $K'$ such that $K\subset K'\subset \Omega$. We mollify $\nu$ and obtain a sequence $\nu_q$. That this is in  $C^{\infty}_c(\Omega)$ is seen by the following argument: Indeed,  $$\mathrm{supp}(\nu_q)\subset \overline{\mathrm{supp}(\phi_k)+\mathrm{supp}(\nu)}\subset \overline{\mathrm{supp}(\phi_k)+K}.$$ If $k$ is large enough (and dependent only on $K$ and $\Omega$) we will eventually have $\mathrm{supp}(v_q)\subset K'$.
We have that $\|\nu_q\|_{\infty}\leq \|\nu\|_{\infty}=1$, and $\nu_q\to \nu$ in the $L^1$ sense. From the hypothesis, we have for all $s\in \bbN$: $$\int_{\Omega} u(x) \nu_s(x)dx=0 $$
From dominated convergence theorem, we can pass to the limit inside the integral to obtain $$\int_K u(x) \mathrm{sgn}(u)(x)dx=\int_K |u(x)|dx=0 $$ which gives $u(x)=0 \ a.e. x \in K$. Since $K $ was arbitrary, we have the result.
}
A simple corollary of \ref{thm:fundamental_lemma_of_CV} is the following:
\begin{proposition}
    Let $f,g\in L^1_{\mathrm{loc}}(\Omega)$, and let $T_f$ and $T_g$ be defined as \ref{eq:T_f}. If $T_f=T_g$, then $f=g \ a.e. x\in \Omega$. 
\end{proposition}
\subsubsection{Differentiation of distributions}
\defn{Differential of a distribution}{Let $T:C^{\infty}_c(\Omega)\to \bbR$ be a distribution. We define the \emph{distributional derivative} of $T$ as follows:
\begin{equation}
    \Big(D^{\alpha}T\Big)(\phi):=(-1)^{|\alpha|}T \big(D^{\alpha}\phi\big)
\end{equation}
}

The reason for defining the derivative of a distribution with a factor of $-1$ is due to the integartion by parts formula:

Indeed, if $T=T_f$ for some $C^1$ function, we have \begin{align*}\Big(\frac{\partial}{\partial x^i}T_f \Big)(\phi):=-T_f(\frac{\partial}{\partial x^i}\phi)=-\int_{\Omega} f(x)\frac{\partial \phi}{\partial x^i}(x) dx&=\int_{\Omega} \frac{\partial f}{\partial x^i}(x) \phi(x)dx \\ &= T_{\partial_i f}(\phi)  \end{align*}
And in general, $$D^{\alpha}T_f\equiv T_{D^{\alpha}f}. $$



\section{Briefly on the space of H\"{o}lder continuous functions}\label{app:holder}
Let $\Omega\subset\mathbb{R}^n$ be an open set and let
$0<\alpha\leq 1$.
\defn{$C^{0,\alpha}(\Omega)$}{
We define
\[
C^{0,\alpha}(\Omega)
=
\left\{
u\in C(\Omega)\cap L^{\infty}(\Omega):
[u]_{C^{0,\alpha}(\Omega)}<\infty
\right\},
\]
where
\[
[u]_{C^{0,\alpha}(\Omega)}
=
\sup_{\substack{x,y\in\Omega\\x\neq y}}
\frac{|u(x)-u(y)|}{|x-y|^\alpha}
\]
is the Hölder seminorm of $u$. The corresponding norm for $C^{0,\alpha}(\Omega)$ is
\[
\|u\|_{C^{0,\alpha}(\Omega)}
=
\|u\|_{L^\infty(\Omega)}
+
[u]_{C^{0,\alpha}(\Omega)}.
\]}

\defn{$C^{0,\alpha}(\overline{\Omega})$.}{
We define
\[
C^{0,\alpha}(\overline{\Omega})
=
\left\{
u\in C(\overline{\Omega})\cap L^{\infty}(\overline{\Omega}):
[u]_{C^{0,\alpha}(\overline{\Omega})}<\infty
\right\},
\]
where
\[
[u]_{C^{0,\alpha}(\overline{\Omega})}
=
\sup_{\substack{x,y\in\overline{\Omega}\\x\neq y}}
\frac{|u(x)-u(y)|}{|x-y|^\alpha}.
\] The corresponding norm for $C^{0,\alpha}(\overline{\Omega})$ is
\[
\|u\|_{C^{0,\alpha}(\overline{\Omega})}
=
\|u\|_{L^\infty(\overline{\Omega})}
+
[u]_{C^{0,\alpha}(\overline{\Omega})}.
\]
}
\begin{remark}
\begin{enumerate}
\item There is no distinction between the two spaces, as H\"{o}lder continuous functions on $\Omega$ extend uniformly to the boundary $\overline{\Omega}$. \emph{But}, this is only true when $k=0$ and not for general $C^{k,\alpha}(\Omega)$.
\item $\alpha=1$ corresponds to the \emph{Lipschitz class} of functions.
\item If $\Omega$ is connected, $C^{0,\alpha}$ for $\alpha>1$ only contains constant functions. Indeed, if $\alpha>1$, we would have that derivatives \emph{exist} at each point $x\in \Omega$ and are zero.    
\end{enumerate}
\end{remark}

\defn{$C^{k,\alpha}(\Omega)$}{
For $k\in\mathbb{N}$, we define
\[
C^{k,\alpha}(\Omega)
=
\left\{
u\in C^k(\Omega):
D^\beta u\in L^\infty(\Omega)\text{ for all }|\beta|\leq k,\
[D^\beta u]_{C^{0,\alpha}(\Omega)}<\infty
\text{ for all }|\beta|=k
\right\},
\]
where $\beta=(\beta_1,\ldots,\beta_n)$ is a multi-index and
\[
D^\beta u
=
\frac{\partial^{|\beta|}u}
{\partial x_1^{\beta_1}\cdots\partial x_n^{\beta_n}},
\qquad
|\beta|=\beta_1+\cdots+\beta_n.
\]
The standard norm on this space is
\[
\|u\|_{C^{k,\alpha}(\Omega)}
=
\sum_{|\beta|\leq k}
\|D^\beta u\|_{L^\infty(\Omega)}
+
\sum_{|\beta|=k}
[D^\beta u]_{C^{0,\alpha}(\Omega)}.
\]}

\defn{$C^{k,\alpha}(\overline{\Omega})$}{
Similarly, we define
\[
C^{k,\alpha}(\overline{\Omega})
=
\left\{
u\in C^k(\overline{\Omega}):
D^\beta u\in L^\infty(\Omega)\text{ for all }|\beta|\leq k,\
[D^\beta u]_{C^{0,\alpha}(\overline{\Omega})}<\infty
\text{ for all }|\beta|=k
\right\},
\]
with norm
\[
\|u\|_{C^{k,\alpha}(\overline{\Omega})}
=
\sum_{|\beta|\leq k}
\|D^\beta u\|_{L^\infty(\overline{\Omega})}
+
\sum_{|\beta|=k}
[D^\beta u]_{C^{0,\alpha}(\overline{\Omega})}.
\]
Here, $C^k(\overline{\Omega})$ is understood in the usual up-to-the-boundary
sense: the derivatives up to order $k$ extend continuously to
$\overline{\Omega}$.}


\defn{Local Hölder spaces}
{We also define
\[
C^{k,\alpha}_{\mathrm{loc}}(\Omega)
=
\left\{
u\in C^k(\Omega):
u\in C^{k,\alpha}(K)
\text{ for every compact }K\subset\Omega
\right\}.
\]
In other words, all derivatives of order $k$ are locally Hölder
continuous with exponent $\alpha$. The corresponding local seminorms are \[
\|u\|_{C^{k,\alpha}(K)}
=
\sum_{|\beta|\leq k}
\|D^\beta u\|_{L^\infty(K)}
+
\sum_{|\beta|=k}
[D^\beta u]_{C^{0,\alpha}(K)},
\]
where
\[
[D^\beta u]_{C^{0,\alpha}(K)}
=
\sup_{\substack{x,y\in K\\x\neq y}}
\frac{|D^\beta u(x)-D^\beta u(y)|}{|x-y|^\alpha}.
\] 
Fix a compact exhaustion $K_i$ of $\Omega$ and let $p_i(u):=\|u\|_{C^{k,\alpha}(K_i)}$ be a collection of semi-norms. This collection separates points and hence metrizes $C^{k,\alpha}_{\mathrm{loc}}(\Omega)$. With this collection of semi-norms, $C^{k,\alpha}_{\mathrm{loc}}(\Omega)$ becomes a \emph{Frechet space}. }
\begin{theorem}
    H\"{o}lder spaces are complete metric spaces.
\end{theorem}

\section{Sobolev Spaces}\label{app:sobolev}
Throughout, $\Omega\subseteq\bbR^n$ is an open set, $k\geq 1$ is an integer and $1\leq p\leq\infty$ unless stated otherwise. For a multi-index $\alpha=(\alpha_1,\ldots,\alpha_n)$ we write $|\alpha|=\sum_i\alpha_i$ and $\partial^\alpha=\partial_1^{\alpha_1}\cdots\partial_n^{\alpha_n}$. We denote by $C_c^\infty(\Omega)$ the space of \emph{test functions}, and by $p'$ the conjugate exponent, $\frac1p+\frac1{p'}=1$. Much of this section follows \cite{Evans2010} (Chapter 5), \cite{AdamsFournier2003} and \cite{GilbargTrudinger2001} (Chapter 7).

\subsection{Weak derivatives and Sobolev spaces}
\defn{Weak derivative}{\label{defn:weak_derivative}Let $u\in L^1_{loc}(\Omega)$ and let $\alpha$ be a multi-index. We say that $v\in L^1_{loc}(\Omega)$ is the \emph{$\alpha$-th weak derivative} of $u$, and write $v=\partial^\alpha u$, if
\begin{equation}\label{eq:weak_derivative}
    \int_\Omega u\,\partial^\alpha\varphi\,dx=(-1)^{|\alpha|}\int_\Omega v\,\varphi\,dx\qquad\text{for every }\varphi\in C_c^\infty(\Omega).
\end{equation}}
By the fundamental lemma of the calculus of variations, $v$ is unique up to a set of measure zero (whenever it exists). If $u\in C^{|\alpha|}(\Omega)$, then \eqref{eq:weak_derivative} is just integration by parts, so the weak derivative coincides with the classical one.

\defn{The Sobolev space $W^{k,p}(\Omega)$ (weak derivative version)}{\label{defn:sobolev_weak}
$$W^{k,p}(\Omega):=\{u\in L^p(\Omega):\ \partial^\alpha u \text{ exists in the sense of \ref{defn:weak_derivative} and }\partial^\alpha u\in L^p(\Omega)\text{ for all }|\alpha|\leq k\}.$$
We write $W^{k,p}_{loc}(\Omega)$ for the functions that lie in $W^{k,p}(U)$ for every open $U\subset\subset\Omega$.}

We now give the same definition in the language of distributions.

\defn{The Sobolev space $W^{k,p}(\Omega)$ (distributional version)}{\label{defn:sobolev_distributional}Let $\mathscr D'(\Omega)$ be the space of distributions (continuous linear functionals on $C_c^\infty(\Omega)$). Every $u\in L^1_{loc}(\Omega)$ defines a distribution $T_u(\varphi):=\int_\Omega u\varphi\,dx$, and the \emph{distributional derivative} of $T\in \mathscr D'(\Omega)$ is
$$(\partial^\alpha T)(\varphi):=(-1)^{|\alpha|}T(\partial^\alpha\varphi).$$
Then $$W^{k,p}(\Omega):=\{u\in L^p(\Omega): \text{for each }|\alpha|\leq k,\ \partial^\alpha T_u=T_{v_\alpha}\text{ for some }v_\alpha\in L^p(\Omega)\}.$$}

\defn{Norms and seminorms}{For $u\in W^{k,p}(\Omega)$ define
\begin{equation}
\|u\|_{W^{k,p}(\Omega)}:=\Big(\sum_{|\alpha|\leq k}\|\partial^\alpha u\|_{L^p(\Omega)}^p\Big)^{1/p}\ (1\leq p<\infty),\qquad \|u\|_{W^{k,\infty}(\Omega)}:=\max_{|\alpha|\leq k}\|\partial^\alpha u\|_{L^\infty(\Omega)},
\end{equation}
and the \emph{seminorms}
\begin{equation}
|u|_{W^{k,p}(\Omega)}:=\Big(\sum_{|\alpha|=k}\|\partial^\alpha u\|_{L^p(\Omega)}^p\Big)^{1/p}\ (p<\infty),\qquad |u|_{W^{k,\infty}(\Omega)}:=\max_{|\alpha|=k}\|\partial^\alpha u\|_{L^\infty(\Omega)}.
\end{equation}}

\defn{$W^{k,p}_0(\Omega)$}{For $1\leq p<\infty$, let $W^{k,p}_0(\Omega)$ be the closure of $C_c^\infty(\Omega)$ in $W^{k,p}(\Omega)$ with respect to $\|\cdot\|_{W^{k,p}(\Omega)}$.}

\subsubsection{Completeness, separability and reflexivity}
\thmp{}{For $1\leq p\leq\infty$, $W^{k,p}(\Omega)$ is a Banach space. In particular $W_0^{k,p}(\Omega)$, being a closed subspace, is also Banach.}{Let $(u_m)$ be Cauchy in $W^{k,p}(\Omega)$. Then for every $|\alpha|\leq k$, $(\partial^\alpha u_m)$ is Cauchy in $L^p(\Omega)$, and since $L^p$ is complete, $\partial^\alpha u_m\to v_\alpha$ in $L^p$ for some $v_\alpha$. Write $u:=v_0=\lim u_m$. For any $\varphi\in C_c^\infty(\Omega)$, $\varphi$ and $\partial^\alpha \varphi$ are in $L^{p'}$, so by H\"older we can pass to the limit in
$$\int_\Omega u_m\,\partial^\alpha\varphi\,dx=(-1)^{|\alpha|}\int_\Omega \partial^\alpha u_m\,\varphi\,dx$$
to get $\int_\Omega u\,\partial^\alpha\varphi=(-1)^{|\alpha|}\int_\Omega v_\alpha\varphi$. Hence $\partial^\alpha u=v_\alpha\in L^p$, so $u\in W^{k,p}(\Omega)$ and $u_m\to u$ in $W^{k,p}$.}

\thmp{}{$W^{k,p}(\Omega)$ is separable for $1\leq p<\infty$ and reflexive for $1<p<\infty$.}{Let $N$ be the number of multi-indices with $|\alpha|\leq k$. The map
$$P:W^{k,p}(\Omega)\to L^p(\Omega)^N,\qquad u\mapsto(\partial^\alpha u)_{|\alpha|\leq k}$$
is a linear isometry when $L^p(\Omega)^N$ carries the norm $\big(\sum_\alpha\|f_\alpha\|_p^p\big)^{1/p}$, and its image is closed (by completeness of $W^{k,p}$). Now $L^p(\Omega)^N$ is separable for $1\leq p<\infty$ and reflexive for $1<p<\infty$. Subspaces of separable metric spaces are separable, and closed subspaces of reflexive Banach spaces are reflexive. Transferring through $P$ gives the claim.}
\subsubsection{The Hilbert spaces $H^k$}
\defn{$H^k(\Omega)$ and $H^k_0(\Omega)$}{We write $H^k(\Omega):=W^{k,2}(\Omega)$ and $H^k_0(\Omega):=W^{k,2}_0(\Omega)$.}
\begin{proposition}
$H^k(\Omega)$ is a Hilbert space with the inner product
$$\langle u,v\rangle_{H^k(\Omega)}:=\sum_{|\alpha|\leq k}\int_\Omega\partial^\alpha u\,\partial^\alpha v\,dx,$$
whose induced norm is $\|\cdot\|_{W^{k,2}}$. The same holds for $H^k_0(\Omega)$.
\end{proposition}
This is immediate from completeness together with the parallelogram law (or because $P$ above embeds $H^k$ isometrically as a closed subspace of $L^2(\Omega)^N$, a Hilbert space). For $k=1$,
$$\langle u,v\rangle_{H^1}=\int_\Omega uv+\nabla u\cdot\nabla v\,dx,$$
and for $H^1_0(\Omega)$ with $\Omega$ bounded one may equivalently use $\int_\Omega\nabla u\cdot\nabla v\,dx$ (see the Poincar\'e inequality below), which is the form that appears in the weak formulation of the Dirichlet problem. The dual of $H^1_0(\Omega)$ is denoted $H^{-1}(\Omega)$. On $\bbR^n$ one has the Fourier characterization
$$H^k(\bbR^n)=\{u\in L^2:\ (1+|\xi|^2)^{k/2}\hat u\in L^2\},\qquad \|u\|_{H^k}\simeq\|(1+|\xi|^2)^{k/2}\hat u\|_{L^2},$$
which also allows one to define $H^s$ for non-integer $s$.

\subsection{Approximation by smooth functions}
Fix a mollifier: $\eta\in C_c^\infty(B_1(0))$, $\eta\geq0$, $\int\eta=1$, and set $\eta_\varepsilon(x):=\varepsilon^{-n}\eta(x/\varepsilon)$. For $\varepsilon>0$ put $\Omega_\varepsilon:=\{x\in\Omega:\mathrm{dist}(x,\partial\Omega)>\varepsilon\}$ and for $u\in L^1_{loc}(\Omega)$,
$$u^\varepsilon:=\eta_\varepsilon*u\in C^\infty(\Omega_\varepsilon).$$
The basic fact is that mollification commutes with weak derivatives: if $\partial^\alpha u$ exists, then $\partial^\alpha(u^\varepsilon)=\eta_\varepsilon*\partial^\alpha u$ in $\Omega_\varepsilon$.

\begin{theorem}[Local approximation by smooth functions]\label{thm:local_approx}
Let $u\in W^{k,p}(\Omega)$ with $1\leq p<\infty$. Then $u^\varepsilon\in C^\infty(\Omega_\varepsilon)$ and $u^\varepsilon\to u$ in $W^{k,p}_{loc}(\Omega)$, i.e. for every open $U\subset\subset\Omega$, $\|u^\varepsilon-u\|_{W^{k,p}(U)}\to0$ as $\varepsilon\to0$.
\end{theorem}

The previous theorem controls $u^\varepsilon$ only on compactly contained sets, and its norm may blow up as one approaches $\partial\Omega$. To get an approximation on all of $\Omega$ one patches local approximations together with a partition of unity.

\begin{theorem}[\textbf{(Meyers--Serrin)} Global approximation by smooth functions]\label{thm:meyers_serrin}
Let $\Omega\subseteq\bbR^n$ be \emph{any} open set and $1\leq p<\infty$. Then $C^\infty(\Omega)\cap W^{k,p}(\Omega)$ is dense in $W^{k,p}(\Omega)$. 
\end{theorem}

\begin{theorem}[Global approximation by functions smooth up to the boundary]\label{thm:global_approx_C1}
Let $\Omega$ be a bounded open $C^1$ set and $1\leq p<\infty$. Then $C^\infty(\overline\Omega)$ (the restrictions to $\Omega$ of $C_c^\infty(\bbR^n)$ functions) is dense in $W^{k,p}(\Omega)$.
\end{theorem}


\subsection{Calculus rules for weak derivatives}
The following rules show that weak derivatives behave like classical ones. The proofs are usually by approximation (Theorems~\ref{thm:local_approx} and \ref{thm:meyers_serrin}) and passing to the limit in the classical identity, and we omit them.
\begin{theorem}\label{thm:calculus_rules}
Let $\Omega\subseteq\bbR^n$ be open.
\begin{enumerate}
    \item \textbf{(Leibniz rule)} If $u\in W^{k,p}(\Omega)$ and $\varphi\in C^\infty(\Omega)$ with $\varphi$ and all its derivatives up to order $k$ bounded, then $\varphi u\in W^{k,p}(\Omega)$ and
    $$\partial^\alpha(\varphi u)=\sum_{\beta\leq\alpha}\binom{\alpha}{\beta}\partial^\beta\varphi\ \partial^{\alpha-\beta}u,\qquad |\alpha|\leq k.$$
    More generally, if $u\in W^{1,p}(\Omega)$, $v\in W^{1,p'}(\Omega)$, then $uv\in W^{1,1}(\Omega)$ and $\partial_i(uv)=u\,\partial_iv+v\,\partial_iu$. Also, if $u,v\in W^{1,p}(\Omega)\cap L^\infty(\Omega)$, then $uv\in W^{1,p}(\Omega)$ with the same formula.
    \item \textbf{(Chain rule)} Let $G\in C^1(\bbR)$ with $G'$ bounded and $u\in W^{1,p}(\Omega)$. Then $G(u)\in W^{1,p}_{loc}(\Omega)$ and $\partial_iG(u)=G'(u)\partial_iu$. If $|\Omega|<\infty$ or $G(0)=0$, then $G(u)\in W^{1,p}(\Omega)$. The same holds for $G$ Lipschitz and piecewise $C^1$ (with finitely many corners).
    \item \textbf{(Positive and negative parts)} If $u\in W^{1,p}(\Omega)$ then $u^+=\max(u,0)$, $u^-=\max(-u,0)$ and $|u|$ lie in $W^{1,p}(\Omega)$, and
    $$\nabla u^+=\mathbf 1_{\{u>0\}}\nabla u,\qquad\nabla u^-=-\mathbf 1_{\{u<0\}}\nabla u,\qquad\nabla|u|=\mathrm{sgn}(u)\nabla u.$$
    In particular $\nabla u=0$ a.e. on $\{u=c\}$ for every $c\in\bbR$, and $\max(u,v),\min(u,v)\in W^{1,p}$ for $u,v\in W^{1,p}$.
    \item \textbf{(Change of variables)} Let $\Phi:\Omega'\to\Omega$ be a bijection such that $\Phi$ and $\Phi^{-1}$ are $C^1$ with bounded derivatives (a bi-Lipschitz $C^1$ diffeomorphism). If $u\in W^{1,p}(\Omega)$ then $u\circ\Phi\in W^{1,p}(\Omega')$ and
    $$\partial_j(u\circ\Phi)=\sum_{i=1}^n(\partial_iu)\circ\Phi\ \partial_j\Phi^i.$$
    Moreover $\|u\circ\Phi\|_{W^{1,p}(\Omega')}\leq C\|u\|_{W^{1,p}(\Omega)}$ with $C$ depending on $n,p$ and the bounds on $D\Phi,D\Phi^{-1}$. This is what allows one to flatten the boundary with a chart (for instance an admissible boundary coordinate system) and work in $\bbH^n$.
    \item \textbf{(Lipschitz functions)} $W^{1,\infty}_{loc}(\Omega)=\mathrm{Lip}_{loc}(\Omega)$ (Rademacher). If $\Omega$ is convex, then $W^{1,\infty}(\Omega)$ is the space of bounded Lipschitz functions on $\Omega$, with $\mathrm{Lip}(u)=\|\nabla u\|_{L^\infty}$.
\end{enumerate}
\end{theorem}
Here is a weak analogue of ``zero derivative implies constant'', which we need in the sequel:
\thmp{}{Let $\Omega$ be open and connected and $u\in W^{1,p}_{loc}(\Omega)$ with $\nabla u=0$ a.e. Then $u$ is a.e. equal to a constant.}{For $U\subset\subset\Omega$ and $\varepsilon$ small, $\nabla u^\varepsilon=\eta_\varepsilon*\nabla u=0$ on $U$, so $u^\varepsilon$ is locally constant on $U$. Since $u^\varepsilon\to u$ in $L^p(U)$ (and a.e. along a subsequence), $u$ is locally a.e. constant. A locally constant function on a connected set is constant.}

\subsubsection{The Gagliardo--Nirenberg--Sobolev inequality and Sobolev embeddings}
\defn{Sobolev conjugate}{For $1\leq p<n$, the \emph{Sobolev conjugate} of $p$ is $$p^*:=\frac{np}{n-p},\qquad\text{equivalently}\qquad\frac1{p^*}=\frac1p-\frac1n.$$}
Note that $p^*>p$, $1^*=\frac{n}{n-1}$, and $p^*\to\infty$ as $p\uparrow n$.

\begin{theorem}[Gagliardo--Nirenberg--Sobolev]\label{thm:GNS}
Let $1\leq p<n$. There exists a constant $C=C(n,p)$ such that
\begin{equation}\label{eq:GNS}
\|u\|_{L^{p^*}(\bbR^n)}\leq C\,\|Du\|_{L^p(\bbR^n)}\qquad\text{for all }u\in C_c^1(\bbR^n).
\end{equation}
\end{theorem}
\begin{remark}
\begin{enumerate}
    \item Let $u_\lambda(x):=u(\lambda x)$. Then $\|u_\lambda\|_{L^q}=\lambda^{-n/q}\|u\|_{L^q}$ and $\|Du_\lambda\|_{L^p}=\lambda^{1-n/p}\|Du\|_{L^p}$. If \eqref{eq:GNS} holds with $q$ in place of $p^*$ and a constant independent of $u$, letting $\lambda\to0$ and $\lambda\to\infty$ forces $-\frac nq=1-\frac np$, i.e. $\frac1q=\frac1p-\frac1n$. So $q=p^*$. This tells us that the \emph{only choice} of exponent is $p^*$.
    \item By density, \eqref{eq:GNS} extends to all $u\in W^{1,p}(\bbR^n)$ with $1\leq p<n$ (recall $W^{1,p}(\bbR^n)=W^{1,p}_0(\bbR^n)$).
\end{enumerate}
\end{remark}

\begin{theorem}[Sobolev embedding in $\bbR^n$]\label{thm:sobolev_embedding_Rn}
Let $1\leq p<n$. Then $W^{1,p}(\bbR^n)\hookrightarrow L^q(\bbR^n)$ continuously for every $q\in[p,p^*]$, i.e.
$$\|u\|_{L^q(\bbR^n)}\leq C(n,p,q)\|u\|_{W^{1,p}(\bbR^n)}.$$
\end{theorem}
For $q=p^*$ this is Theorem~\ref{thm:GNS}. For $q\in(p,p^*)$ one interpolates, $\|u\|_q\leq\|u\|_p^{1-\theta}\|u\|_{p^*}^\theta$ with $\frac1q=\frac{1-\theta}p+\frac\theta{p^*}$. Iterating gives the higher-order version: if $kp<n$, then $W^{k,p}(\bbR^n)\hookrightarrow L^q(\bbR^n)$ for $q\in[p,q_k]$, $\frac1{q_k}=\frac1p-\frac kn$.

\begin{theorem}[Sobolev embedding on bounded domains]\label{thm:sobolev_embedding_bounded}
Let $\Omega\subset\bbR^n$ be a bounded open $C^1$ set and $1\leq p<n$. Then $W^{1,p}(\Omega)\hookrightarrow L^q(\Omega)$ continuously for every $q\in[1,p^*]$:
$$\|u\|_{L^q(\Omega)}\leq C(n,p,q,\Omega)\|u\|_{W^{1,p}(\Omega)}.$$
If we instead consider $u\in W^{1,p}_0(\Omega)$, then this holds for \emph{any} bounded open $\Omega$ (no regularity of $\partial\Omega$), and in fact $\|u\|_{L^{p^*}(\Omega)}\leq C(n,p)\|Du\|_{L^p(\Omega)}$ with a constant independent of $\Omega$.
\end{theorem}
\begin{remark}
The idea of the proof is to reduce to $\bbR^n$. For a bounded $C^1$ set there is a bounded linear \emph{extension operator} $E:W^{1,p}(\Omega)\to W^{1,p}(\bbR^n)$ with $Eu=u$ on $\Omega$ (flatten the boundary locally, reflect across $\{y^n=0\}$, and glue using a partition of unity). Applying Theorem~\ref{thm:sobolev_embedding_Rn} to $Eu$ and restricting gives the bound for $q\in[p,p^*]$, and for $q<p$ one uses $|\Omega|<\infty$ and H\"older. \end{remark}

\subsection{Poincar\'e inequalities}
Let $|A|$ denote the Lebesgue measure of a measurable set $A$. For a set $A$ with $0<|A|<\infty$ and $u\in L^1(A)$, we write
$$(u)_A:=\fint_Au\,dx=\frac1{|A|}\int_Au\,dx$$
for the \emph{mean} (average) of $u$ over $A$, and $\fint_A|f|^q\,dx:=\frac{1}{|A|}\int_A|f|^q\,dx$.

\subsubsection{Poincar\'e inequality for $W^{1,p}_0$}
\begin{theorem}[Poincar\'e inequality, zero boundary values]\label{thm:poincare_W0}
Let $\Omega$ be a bounded open set (it is enough that $\Omega$ is contained in a slab $\{a<x^n<b\}$) and $1\leq p<\infty$. There exists $C=C(n,p,\Omega)$ such that
$$\|u\|_{L^p(\Omega)}\leq C\|Du\|_{L^p(\Omega)}\qquad\text{for all }u\in W^{1,p}_0(\Omega).$$
\end{theorem}
\begin{remark}
As a consequence we have $u\mapsto\|Du\|_{L^p(\Omega)}$ as a norm on $W_0^{1,p}(\Omega)$, equivalent to $\|\cdot\|_{W^{1,p}}$. In particular, $H^1_0(\Omega)$ is a Hilbert space under $\int_\Omega\nabla u\cdot\nabla v\,dx$ as an equivalent inner product.

\end{remark}

\subsubsection{Poincar\'e--Sobolev inequality for $W^{1,p}_0$}
For $u\in W_0^{1,p}(\Omega)$ with $1\leq p<n$, Theorem~\ref{thm:sobolev_embedding_bounded} already gives the \emph{Poincar\'e--Sobolev inequality}
$$\|u\|_{L^{p^*}(\Omega)}\leq C(n,p)\|Du\|_{L^p(\Omega)},$$
which improves Theorem~\ref{thm:poincare_W0}: the left side has a better exponent and the constant does not depend on $\Omega$. The two are related by H\"older, $\|u\|_{L^p(\Omega)}\leq|\Omega|^{1/n}\|u\|_{L^{p^*}(\Omega)}$, which shows that Theorem~\ref{thm:poincare_W0} holds with $C=C(n,p)|\Omega|^{1/n}$ on any open set of finite measure.

\subsubsection{Poincar\'e inequality on balls}
We write $B_r=B_r(x_0)$. 

\begin{theorem}[Poincar\'e inequality on balls, with means]\label{thm:poincare_ball_mean}
Let $1\leq p<\infty$. There is $C=C(n,p)$ such that for every ball $B_r\subset\bbR^n$ and every $u\in W^{1,p}(B_r)$,
$$\|u-(u)_{B_r}\|_{L^p(B_r)}\leq C\,r\,\|Du\|_{L^p(B_r)},\qquad\text{equivalently}\qquad\Big(\fint_{B_r}|u-(u)_{B_r}|^p\Big)^{1/p}\leq C\,r\Big(\fint_{B_r}|Du|^p\Big)^{1/p}.$$
\end{theorem}

\subsubsection{Poincar\'e--Sobolev inequality with means on balls}
\begin{theorem}[Poincar\'e--Sobolev inequality on balls, with means]\label{thm:poincare_sobolev_ball}
Let $1\leq p<n$. There is $C=C(n,p)$ such that for every ball $B_r\subset\bbR^n$ and every $u\in W^{1,p}(B_r)$,
$$\|u-(u)_{B_r}\|_{L^{p^*}(B_r)}\leq C\,\|Du\|_{L^p(B_r)},\qquad\text{equivalently}\qquad\Big(\fint_{B_r}|u-(u)_{B_r}|^{p^*}\Big)^{1/p^*}\leq C\,r\Big(\fint_{B_r}|Du|^p\Big)^{1/p}.$$
\end{theorem}

\subsection{Compactness}
We now digress to look at a few results that pertain to compactness in various spaces of functions. 

\subsubsection{Arzelà--Ascoli compactness theorem}

\defn{Equicontinuity}{Let $X$ be a metric space. We say a collection $\mathcal F$ of functions $f:X\to \bbR$ \emph{equicontinuous} if for every $x\in X$ and every $\varepsilon>0$, there exists a $\delta(x,\varepsilon)>0$ such that $$d_X(x,y)<\delta\implies |f(x)-f(y)|<\varepsilon,\ \text{for all } f\in \mathcal F. $$ 
If $\delta$ is independent of $x$, we say $\mathcal F$ is \emph{uniformly equicontinuous} }
\begin{lemma}
    For a compact metric space $X$, equicontinuity $\implies$ uniform equicontinuity. 
\end{lemma}
\begin{lemma}\label{lem:equivalence_of_compactness_CX}
    For a compact metric space $X$ and a collection $\mathcal F \subset C(X)\equiv CB(X)$, the following are equivalent:
    \begin{enumerate}
        \item $\mathcal F$ is relatively compact (has compact closure).
        \item $\mathcal F$ is precompact (totally-bounded).
        \item For every sequence $f_n$ of continuous functions in $\mathcal F$, there is a uniformly convergent subsequence convergent to $f\in \mathcal F$.
    \end{enumerate}

\end{lemma}
\begin{remark}
    Note that the result \ref{lem:equivalence_of_compactness_CX} is simply a version of the \emph{Heine-Borel} property for metric spaces in disguise. Indeed, in metric spaces, (compact) $\iff$ (sequentially compact) $\iff$ (complete and totally bounded). The proof of the above follows quite closely to the proof of the general statement. 
\end{remark}

\begin{theorem}\label{thm:AA}
\textbf{(Arzelà--Ascoli)}
{Let $X$ be a compact metric space and $\mathcal F$ a collection of \emph{equicontinuous} functions in $C(X)$ such that \begin{equation}\label{eq:boundedness_in_AA_thm}
    \text{For every }x\in X, \ \sup_{f\in \mathcal F}|f(x)|<\infty
\end{equation}
then, $\mathcal F$ is relatively compact in $C(X)$.}
    
\end{theorem}
\begin{remark}
1. Note that, this is tantamount to saying that given a sequence of equicontinuous functions with sufficient boundedness properties (\ref{eq:boundedness_in_AA_thm}), we can always extract a uniformly convergent subseqence. 
\\\\2. We also note that the condition \ref{eq:boundedness_in_AA_thm}, along with equicontinuity, actually implies uniform boundedness. 
\end{remark}

\subsubsection{Banach--Alaoglu theorem}
\begin{theorem} \textbf{(Banach--Alaoglu)} Let $E$ be a normed linear space and $E^*$ its dual. The closed unit ball $$B_{E^*}:=\{f\in E^*: \| f\|_{op}\leq 1 \}$$
is compact in the weak* topology $\sigma(E^*,E)$.
\end{theorem}
\textbf{Sketch:} The proof is a clever application of the Tychonoff theorem, applied on the space $Y=\bbR^E$ and noting that the set $B_{E^*}$ is essentially the space (call it $K$) of all $\omega\in Y$ such that $|\omega_x|\leq \|x\|$, $\omega_{x+y}=\omega_x+\omega_y$ and $\omega_{\lambda x}=\lambda \omega_x$. Once we note that $K$ is compact, we would have the result.
\\\\ Note that, this gives us compactness of the unit ball, yet not \emph{sequential compactness}. We require additional hypothesis (separability) to ensure \emph{sequential compactness}, as the following modification suggests: 
\begin{theorem}
    \textbf{(Banach--Alaoglu for separable spaces)} Let $E$ be a separable normed linear space. Then, the closed unit ball $$B_{E^*}:=\{f\in E^*: \|f\|_{op}\leq 1\}$$ is \emph{metrizable} and \emph{sequentially compact}. In other words, every bounded sequence $f_n$ has a weak* convergent subsequence.
\end{theorem}

We provide a characterisation for \emph{reflexive spaces} (Spaces whose double duals can be identified with the space itself, or rather, spaces for which the $J:E\to E^{**}$ is surjective).
\begin{theorem}
    \textbf{(Characterization of reflexive spaces)} Let $E$ be a normed linear space. $E$ is reflexive if and only if $$B_E:=\{x\in E:\|x\|\leq 1\} $$ is compact in the weak topology $\sigma(E,E^*)$. 
\end{theorem}

\subsubsection{Eberlein--Šmulian theorem}
We try to modify the Banach--Alaoglu for separable spaces to get a result for reflexive spaces. Recall that for reflexive spaces, we have the weak compactness of the unit ball, yet we do not have its sequential compactness from that characterisation. This is the Eberlein--Šmulian theorem. We begin with a few lemmas that will prove to be important: 
\begin{lemma}
    Closed subspaces of reflexive spaces are reflexive spaces.
\end{lemma}
\begin{lemma}
    For banach spaces $E$, $E^*$ being separable implies $E$ is separable.
\end{lemma}
\begin{lemma}
    A Banach space $E$ is reflexive if and only if its dual is reflexive.
\end{lemma}
Now we are in position to state the Eberlein--Šmulian theorem: 
\begin{theorem}
    \textbf{(Eberlein--Šmulian)} Let $E$ is a reflexive banach space. Then every bounded sequence $x_n\in E$ has a weakly convergent subsequence. In other words, the closed unit ball $B_{E}$ is weakly sequentially compact.
\end{theorem}
\begin{remark}
1. Note that for reflexive spaces \emph{in general}, we have, baked in, the \emph{topological compactness} of $B_{E}$. In fact, reflexivity is equivalent to it. For reflexive Banach spaces, on the other hand, we have weakly sequential compactness.
\\\\ 2. We have, in fact, the converse to be true as well, which is also called the Eberlein--Šmulian theorem: 
\emph{Let $E$ be a Banach space such that every bounded sequence in $E$ admits a weakly convergent subsequence. Then, $E$ is reflexive.}

\end{remark}
We now come back to $L^p$ spaces.
\subsubsection{Compactness in $L^p$ spaces}
We shall provide a criterion for compactness in $L^p(\Omega)$.
\begin{theorem}
    \textbf{(Kolmogorov--Reisz--Frechet)} Let $\mathcal F$ be a bounded subset of $L^p(\bbR^n)$ with $1\leq p<\infty$ such that $$\lim_{|h|\to 0} \|u(\cdot - h)-u(\cdot)\|_{L^p(\bbR^n)}=0 \text{ uniformly in }u\in \mathcal F. $$ Then the closure of $\mathcal F|_{\Omega}$ is compact in $L^p(\Omega)$ for any measurable set $\Omega\subset \bbR^n$ of finite measure. 
\end{theorem}

\subsubsection{Rellich-Kondarachov compact embeddings}
This is the result of primary interest to us:
\begin{theorem}\label{thm:rellich}
    \textbf{(Rellich-Kondarachov)} Let $\Omega\subset \bbR^n$ be open, bounded and smooth. Then, the following injections are \emph{compact}: 
    \begin{alignat*}{3}
W^{1,p}(\Omega) &\hookrightarrow L^q(\Omega) &\quad& \text{for all } 1\leq q<p^*, &\quad& \text{for } 1\leq p<n,\\
W^{1,p}(\Omega) &\hookrightarrow L^q(\Omega) &\quad& \text{for all } 1\leq q<\infty, &\quad& \text{for } p=n,\\
W^{1,p}(\Omega) &\hookrightarrow C(\overline\Omega) &\quad& &\quad& \text{for } p>n.
\end{alignat*}
\end{theorem}

\subsection{Traces}\label{app:trace}
We want to define 'boundary values' for functions in Sobolev spaces, so that we can viably tackle boundary value problems. The main obstacle is that, $\partial\Omega$ has Lebesgue measure zero, and hence there is no sense in which we can talk about the \emph{"value"} of an $L^p$ function on $\partial \Omega$. Indeed, this is meaningless for $L^p$ functions, but we can still precisely define a notion of a \emph{boundary value} for \emph{\textbf{Sobolev functions}}. 
\begin{theorem}
    (\textbf{Existence of Trace operator}) Let $\Omega$ be open, bounded and smooth, and let $1\\leq p<\infty$. There exists a bounded linear operator $$\mathrm{Tr}:W^{1,p}(\Omega)\to L^p(\partial\Omega)$$ such that $$\mathrm{Tr}u=u|_{\partial\Omega}$$
whenever $u\in W^{1,p}(\Omega)\cap C^\infty{\overline\Omega}$.
\end{theorem}
For any $u\in W^{1,p}(\Omega)$, we call $\mathrm{Tr}u$ as the \emph{zeroth order trace} on the boundary and us usually denoted as $u|_{\partial\Omega}$.
\begin{remark} We collect here some remarks and generalities regarding the trace operator:
\\\\ 1. $L^p(\partial\Omega)$ is defined with respect to the surface measure $d\sigma$ on $\partial\Omega$.
\\\\ 2. The trace map actually restricts to the boundary value for $u\in C^0(\overline\Omega)$, not just $C^{\infty}(\overline\Omega)$.
\\\\ 3. The theorem holds for \emph{Lipschitz} boundaries as well, not just for $C^k$ boundaries. 
\end{remark}
Note that, this operator is easy to define on smooth functions, by simply assigning the boundary values. This operation is clearly linear. Since $C^{\infty}(\overline\Omega)$ is dense in $W^{1,p}(\Omega)$, it is enough to show that $$ \|u|_{\partial\Omega}\|_{L^p(\partial\Omega)}\leq C \|u\|_{W^{1,p}(\Omega)} \text{ for every }u\in C^{\infty}(\overline\Omega).$$
After localizing, flattening the boundary (with an admissible boundary coordinate system) and patching up (using partitions of unity), the theorem can be reduced to the following inequality (which we shall not prove): 
\begin{proposition}
    There exists a constant $C>0$ such that $$\Big{(}\int_{\bbR^{n-1}} |u(x',0)^p dx'|\Big{)}^{1/p}\leq C \|u\|_{W^{1,p}(\bbR^n_+)} \text{ for every }u\in C^{\infty}_c(\bbR^n).$$
\end{proposition}
\begin{remark} The kernel of the trace map is $$\ker(\mathrm{Tr}):=W^{1,p}_0(\Omega).$$

\end{remark}

\subsubsection{Integration by parts formula}
The classical integration by parts formula reads: $$\int_{\Omega} \partial_{x_i}(u) v dx=-\int_{\Omega}u\partial_{x_i}(v)dx +\int_{\partial\Omega} \nu_i (uv) dS $$ for $u,v\in C^1(\overline\Omega)$. An identical result holds for Sobolev functions, with the boundary values coming from the trace map: 
\begin{theorem}
    \textbf{(Integration by parts formula for Sobolev functions)} Let $\Omega$ be open, bounded and $C^1$. $u\in W^{1,p}(\Omega)$ and $v\in W^{1,p'}(\Omega)$ (Ho\"lder conjugate). We then have $$\int_{\Omega} \partial_{x_i}u \cdot v dx =-\int_{\Omega} u\cdot \partial_{x_i}v dx +\int_{\partial\Omega} \nu_i (\mathrm{Tr}u \mathrm{Tr}v)dS. $$
\end{theorem}
The proof follows by approximation by smooth functions and the continuity of the trace operator.
